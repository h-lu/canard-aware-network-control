\section{Critical-layer geometry and a history-selection obstruction}
\label{sec:supplementary-outer-skeleton}

\subsection{Proofs for the preparation-independent critical layer}

\begin{proof}[Proof of \cref{prop:unprepared-outer-skeleton}]
Put \(a_*=D\gamma\).  The estimates
\cref{eq:poisson-bound,eq:shared-products} imply, with constants independent
of \(N\),
\begin{align}
 \|\mathcal N_N(x)\|_N
 &\le C_2\|x\|_N^2+C_3\|x\|_N^3,\notag\\
 \|D\mathcal N_N(x)\|
 &\le C_4\|x\|_N+C_5\|x\|_N^2.                     \label{eq:outer-N-bounds}
\end{align}
For fixed \(r\), consider on \(E_N\)
\begin{equation}
 \mathcal T_{N,r}(z)=A_N^{-1}P_{\perp,N}
       \mathcal N_N(r\mathbf1+z).                   \label{eq:outer-critical-contraction}
\end{equation}
Choose \(M_z\) larger than twice the common quadratic bound in
\cref{eq:outer-N-bounds}.  If \(r_0\) is sufficiently small, uniformly in
\(N\), the map \(\mathcal T_{N,r}\) sends
\(\{\|z\|_N\le M_zr^2\}\) into itself and has Lipschitz constant at most
one half.  Its fixed point is the unique fixed point in that ball and is the
solution \(z_{0,N}(r)\) of \cref{eq:outer-critical-transverse}.

On this ball,
\begin{equation}
 A_N-P_{\perp,N}D\mathcal N_N(r\mathbf1+z_{0,N}(r))
                                                               \label{eq:outer-critical-derivative}
\end{equation}
has inverse norm at most \(2a_*^{-1}\).  Because \(\mathcal N_N\) is a
polynomial with uniform multilinear bounds, induction in the differentiated
fixed-point equation gives, for every fixed finite \(s\), uniform \(C^s\)
bounds.  This proves \cref{eq:outer-critical-high-regularity}; in particular
the \(C^4\) bounds are uniform.  Substituting
\(z_{0,N}(r)=z_{2,N}r^2+z_{3,N}r^3+O_{\rm unif}(r^4)\) into
\cref{eq:outer-critical-transverse} gives
\begin{align*}
 A_Nz_{2,N}&=P_{\perp,N}c_N,\\
 A_Nz_{3,N}&=P_{\perp,N}
       \left(2c_N\circ g_N+\frac\beta3\mathbf1\right).
\end{align*}
Since \(P_{\perp,N}\mathbf1=0\), these identities give
\(z_{2,N}=g_N\) and \cref{eq:outer-g3}.  Taylor's formula applied after
the uniform \(C^4\) estimate yields the first line of
\cref{eq:outer-critical-expansion}.  Stationary projection of
\(\mathcal N_N(r\mathbf1+z_{0,N}(r))\) gives
\begin{equation*}
 \alpha r^2+\left\{2\pi_N^T(c_N\circ g_N)+\frac\beta3\right\}r^3
 +O_{\rm unif}(r^4),
\end{equation*}
which proves the second line and identifies \(\kappa_N\).

The voltage field at the constant history \(V_{0,N}(r)\) is
\begin{equation*}
 (2/3-w_{0,N}(r))\mathbf1
 -\mathcal N_N(x_{0,N}(r))+A_Nz_{0,N}(r).
\end{equation*}
Its stationary and transverse projections vanish by
\cref{eq:outer-critical-transverse,eq:outer-critical-curve}.  Moreover,
every delayed difference is exactly
\(V_{0,N}(r)-V_{0,N}(r)=0\).  This proves the preparation- and
redistribution-free assertion for every \(\delta\).  The recovery law is
irrelevant to this frozen voltage equation.

We next construct the layer splitting without assuming that \(P_N\) is
normal or diagonalizable.  Write
\begin{equation}
 d_N(r)=2c_N\circ x_{0,N}(r)
          +\beta x_{0,N}(r)^{\circ2}.                \label{eq:outer-diagonal-coefficient}
\end{equation}
The decomposition
\(\R^N=\operatorname{span}\{\mathbf1\}\oplus E_N\),
\(x=u\mathbf1+h\), is isometric in the sense that
\(\|x\|_N=|u|+\|h\|_N\).  In these coordinates
\begin{equation}
 \mathcal J^{\rm fv}_{N}(r)=
 \begin{pmatrix}a(r)&b(r)\\c(r)&S(r)\end{pmatrix},             \label{eq:outer-J-blocks}
\end{equation}
where
\begin{align}
 a&=-\pi_N^Td_N,&
 bh&=-\pi_N^T(d_N\circ h),\notag\\
 c&=-P_{\perp,N}d_N,&
 S&=A_N-P_{\perp,N}\diag(d_N)|_{E_N}.               \label{eq:outer-block-formulas}
\end{align}
Because \(x_{0,N}(r)=r\mathbf1+O_{\rm unif}(r^2)\),
\begin{equation}
 |a|+\|b\|+\|c\|+\|S-A_N\|\le C_J|r|              \label{eq:outer-block-smallness}
\end{equation}
with one \(C_J\).  Decrease \(r_0\) until
\(C_Jr_0\le a_*/64\).  Bounded-perturbation variation of constants then
gives
\begin{equation}
 \|e^{S(r)t}\|\le e^{-(a_*-C_J|r|)t},\qquad t\ge0. \label{eq:outer-S-semigroup}
\end{equation}

Seek the center and stable invariant graphs in the forms \(h=H(r)u\) and
\(u=L(r)h\).  Their equations are
\begin{align}
 c+SH&=H(a+bH),\notag\\
 aL+b&=L(cL+S).                                      \label{eq:outer-block-graph-equations}
\end{align}
The linear Sylvester operators at the origin are
\(H\mapsto A_NH\) and \(L\mapsto LA_N\).  Their inverse norms are at most
\(a_*^{-1}\), by the forward semigroup integral and its dual.  More
explicitly, \cref{eq:outer-block-graph-equations} is equivalent to
\begin{align}
 H&=A_N^{-1}\{-c-(S-A_N)H+H(a+bH)\},\notag\\
 L&=\{aL+b-LcL-L(S-A_N)\}A_N^{-1}.                  \label{eq:outer-HL-contractions}
\end{align}
On the balls \(\|H\|,\|L\|\le4C_J|r|/a_*\),
\cref{eq:outer-block-smallness} makes both right-hand sides self-maps with
Lipschitz constant at most one half after one further decrease of \(r_0\).
Hence \(H,L=O_{\rm unif}(r)\), they have uniform finite derivatives, and
\(1-LH\) stays uniformly away from zero.

Set
\begin{equation}
 e_{c,N}(r)=\frac{\mathbf1+H(r)}{1-L(r)H(r)},
 \qquad
 \ell_{c,N}(r)(u\mathbf1+h)=u-L(r)h.                \label{eq:outer-eigen-from-graphs}
\end{equation}
The graph equations show that these are a right and left eigenobject for the
same real scalar eigenvalue \(\lambda_{c,N}=a+bH\), and the two
normalizations in \cref{eq:outer-eigen-normalization} are automatic.  The
stable graph is \(\Ker\ell_{c,N}\), and its generator in the \(h\)
coordinate is \(S+cL\).  In fact,
\begin{equation}
 \|S+cL-A_N\|\le C|r|.                               \label{eq:outer-stable-generator-perturbation}
\end{equation}
The graph coordinate \(h\mapsto Lh\,\mathbf1+h\), its inverse on the stable
graph, and the projection \(P_N^s\) all have norm at most one fixed constant,
because \(L,H=O_{\rm unif}(r)\) and \(1-LH\) is uniformly separated from zero.
Bounded-perturbation variation of constants applied to
\cref{eq:outer-stable-generator-perturbation} therefore gives
\cref{eq:outer-stable-layer-semigroup} with a uniform prefactor.

The first-order parts of \cref{eq:outer-block-graph-equations} give
\begin{equation}
 H(r)=2r g_N+O_{\rm unif}(r^2),
 \qquad
 L(r)=-2r\,\pi_N^T\diag(c_N)A_N^{-1}P_{\perp,N}
          +O_{\rm unif}(r^2).                        \label{eq:outer-HL-expansion}
\end{equation}
Together with \cref{eq:outer-eigen-from-graphs}, these identities prove the
right and left expansions in \cref{eq:outer-eigen-expansion}.

Differentiate the exact critical equation in \(r\).  Since the derivative
with respect to voltage is \(\mathcal J^{\rm fv}_{N}(r)\), one obtains
\begin{equation}
 \mathcal J^{\rm fv}_{N}(r)V_{0,N}'(r)
 =w_{0,N}'(r)\mathbf1.                               \label{eq:outer-critical-differentiated}
\end{equation}
Left multiplication by \(\ell_{c,N}(r)\) gives the exact identity in
\cref{eq:outer-eigen-identity}.  Moreover,
\begin{equation*}
 \Theta_N(r)=1-L(r)z_{0,N}'(r)=1+O_{\rm unif}(r^2).
\end{equation*}
The expansion of \(w_{0,N}'/\Theta_N\) yields the eigenvalue expansion.
It also shows, after decreasing \(r_0\), that the collective eigenvalue is
negative for \(r>0\) and positive for \(r<0\).  The stable complement has
no eigenvalue with nonnegative real part by
\cref{eq:outer-stable-layer-semigroup}, proving the index statements.
\end{proof}

\begin{proof}[Proof of \cref{cor:outer-reduced-actions}]
For sensed
recovery, substitution of
\(x_{0,N}(r)=r\mathbf1+g_Nr^2+O_{\rm unif}(r^3)\) gives
\begin{equation*}
 \sigma_N^{\rm rec}(r)
 =r+\pi_N^T(\varpi_N\circ g_N+d_N^{\rm rec})r^2+O_{\rm unif}(r^3),
\end{equation*}
which is \cref{eq:outer-recovery-expansion}.  In slow time, the recovery
equation restricted formally to the critical curve reads
\begin{equation*}
 w_{0,N}'(r)r_T=\sigma_N^\diamond(r)-\delta^2\nu,
\end{equation*}
and hence gives \cref{eq:outer-reduced-speed}.

Uniformly for the two laws,
\begin{equation}
 w_{0,N}'(r)=-2\alpha r+O_{\rm unif}(r^2),\qquad
 \sigma_N^\diamond(r)=r+O_{\rm unif}(r^2).         \label{eq:outer-speed-inputs}
\end{equation}
First choose \(r_{\rm out}\) so small that, on
\(0<|r|\le r_{\rm out}\),
\begin{equation*}
 \alpha|r|\le|w_{0,N}'(r)|\le3\alpha|r|,
 \qquad
 |\sigma_N^\diamond(r)-r|\le|r|/4.
\end{equation*}
Then decrease \(\delta_0\) so that
\(\delta^2\nu_*\le\rho_\delta/4\) and
\(\rho_\delta\le r_{\rm out}/2\).  The numerator in
\cref{eq:outer-reduced-speed} has the sign of \(r\) and modulus between
\(|r|/2\) and \(3|r|/2\), while \(w_{0,N}'\) has the opposite sign.
This proves \cref{eq:outer-speed-sign} and the displayed expansion.

Combining the exact identities
\cref{eq:outer-eigen-identity,eq:outer-reduced-speed} gives
\begin{equation}
 \frac{\lambda_{c,N}(r)}{q^\diamond_{N,\delta,\nu}(r)}
 =\frac{w_{0,N}'(r)^2}
  {\Theta_N(r)\{\sigma_N^\diamond(r)-\delta^2\nu\}}.          \label{eq:outer-action-identity}
\end{equation}
It has the sign of \(r\), and the preceding bounds imply
\begin{equation}
 \frac{4\alpha^2}{9}|r|
 \le\left|\frac{\lambda_{c,N}(r)}
                 {q^\diamond_{N,\delta,\nu}(r)}\right|
 \le36\alpha^2|r|.                                  \label{eq:outer-action-integrand-bounds}
\end{equation}
Integration proves positivity and
\cref{eq:outer-action-lower-bound}.  Finally,
\cref{eq:outer-speed-inputs,eq:outer-action-identity} gives
\begin{equation*}
 \frac{\lambda_{c,N}(r)}{q^\diamond_{N,\delta,\nu}(r)}
 =4\alpha^2r+O_{\rm unif}(r^2+\delta^2).
\end{equation*}
Integrating on either branch proves
\cref{eq:outer-action-expansion}.  Substituting
\(r=\delta X\), \(|X|=S_\delta/(2\alpha)\), into
\cref{eq:outer-critical-expansion} gives
\cref{eq:outer-critical-fold-offset}.
\end{proof}

\subsection{Positive-\texorpdfstring{\(\delta\)}{delta} frozen-voltage histories}

\begin{proof}[Proof of \cref{prop:frozen-voltage-history-splitting}]
We give a direct trajectory-space proof.  In particular, no determinant,
normality of \(P_N\), or backward RFDE semiflow is used.  The common layer
bound and the structural-ball radius give
\begin{equation}
 b_*:=|K|\sup_{N,\,\|\eta\|_{\mathfrak T_N}<\eta_0}
       \sum_{k=0}^m\|L_{k,N}(\eta)\|<\infty.         \label{eq:frozen-layer-common-bound}
\end{equation}

We first collect the current-state estimates.  Decrease \(r_{\rm out}\) so
that, for one \(c_0>0\),
\begin{equation}
 |\lambda_{c,N}(r)|\ge c_0|r|,
 \qquad 0<|r|\le r_{\rm out}.                       \label{eq:frozen-collective-gap}
\end{equation}
For \(r>0\), the rank-one eigenprojection and
\cref{eq:outer-stable-layer-semigroup} give
\begin{equation}
 \|e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}\|
 \le M_0e^{-\mathfrak a_N(r)s/\delta},\qquad s\ge0. \label{eq:frozen-current-attracting}
\end{equation}
For \(r<0\), put
\begin{equation}
 P_N^u(r)=e_{c,N}(r)\ell_{c,N}(r),\qquad
 P_N^s(r)=\Id-P_N^u(r).                              \label{eq:frozen-current-projections}
\end{equation}
The same estimates imply
\begin{align}
 \|e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}P_N^s(r)\|
 &\le M_0e^{-\mathfrak a_N(r)s/\delta},&&s\ge0,\notag\\
 \|e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}P_N^u(r)\|
 &\le M_0e^{\mathfrak a_N(r)s/\delta},&&s\le0.     \label{eq:frozen-current-repelling}
\end{align}
The harmonic definition in \cref{eq:frozen-history-a} is only a smooth
common lower rate for the two current blocks.  Uniform boundedness of the
projections follows from \cref{eq:outer-eigen-from-graphs}.

Write \(\kappa=\kappa_{N,\delta}(r)\) for the rest of the proof.  From
\cref{eq:frozen-history-rate},
\begin{equation}
 2\kappa<\frac{\mathfrak a_N(r)}{2\delta},\qquad
 e^{2\kappa\theta_m}\le\delta^{-2\vartheta}.        \label{eq:frozen-cap-bounds}
\end{equation}
If a current path \(y\) and a prescribed old history are assembled into
\(\widehat y\), set
\begin{equation}
 \mathcal R_\eta[\widehat y_s]
 =\delta K\sum_{k=0}^mL_{k,N}(\eta)
       \{\widehat y(s)-\widehat y(s-\theta_k)\}.     \label{eq:frozen-delay-operator}
\end{equation}
On any exponentially weighted trajectory space, translation across one
delay gives at most the factor \(e^{2\kappa\theta_m}\).  The current Green
integrals and \cref{eq:frozen-cap-bounds} therefore give the common
perturbation number
\begin{equation}
 q_{N,\delta}(r)
 \le C\frac{\delta^2
       \{1+e^{2\kappa\theta_m}\}}{\mathfrak a_N(r)}.
                                                               \label{eq:frozen-LP-number}
\end{equation}
Since \(\mathfrak a_N(r)\ge c|r|\) on the retained interval and
\(|r|\ge\rho_\delta=\delta S_\delta/(2\alpha)\),
\begin{equation}
 q_{N,\delta}(r)
 \le C\frac{\delta^{1-2\vartheta}}{S_\delta}=o(1).  \label{eq:frozen-LP-small}
\end{equation}
This proves \cref{eq:frozen-history-smallness} and permits one common choice
of \(\delta_0\) for which every contraction below has constant at most
one half.

Suppose first that \(r>0\).  For \(\phi\in\mathscr X_N\), prescribe
\(\widehat y(\theta)=\phi(\theta)\) for \(-\theta_m\le\theta\le0\) and use
variation of constants for \(s\ge0\):
\begin{equation}
 y(s)=e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}\phi(0)
 +\int_0^s e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)(s-u)}
       \mathcal R_\eta[\widehat y_u]\,\dd u.        \label{eq:frozen-attracting-VoC}
\end{equation}
For a future path put
\(\|y\|_+=\sup_{s\ge0}e^{\kappa s}\|y(s)\|_N\).
If \(u-\theta_k\ge0\), then
\begin{equation*}
 e^{\kappa u}\|y(u-\theta_k)\|_N
 \le e^{\kappa\theta_m}\|y\|_+.
\end{equation*}
If \(u-\theta_k<0\), the factor two in
\cref{eq:frozen-history-norm} gives instead
\begin{equation}
 e^{\kappa u}\|\phi(u-\theta_k)\|_N
 \le e^{-\kappa u+2\kappa\theta_k}
       \|\phi\|_{N,r,\delta}^{\sharp}
 \le e^{2\kappa\theta_m}
       \|\phi\|_{N,r,\delta}^{\sharp}.             \label{eq:frozen-old-history-bound}
\end{equation}
Using \cref{eq:frozen-current-attracting} in
\cref{eq:frozen-attracting-VoC}, the Green integral is bounded by
\begin{equation*}
 \int_0^s e^{-\mathfrak a_N(r)(s-u)/\delta}
              e^{-\kappa u}\,\dd u
 \le\frac{e^{-\kappa s}}
          {\mathfrak a_N(r)/\delta-\kappa}.
\end{equation*}
Thus \cref{eq:frozen-LP-number} and a Neumann argument give
\begin{equation}
 \|y(s)\|_N
 \le C e^{-\kappa s}\|\phi\|_{N,r,\delta}^{\sharp}.
                                                               \label{eq:frozen-attracting-current-decay}
\end{equation}
To lift this estimate to a history segment, take
\(-\theta_m\le\theta\le0\).  If \(s+\theta\ge0\), then
\begin{equation*}
 e^{2\kappa\theta}\|y(s+\theta)\|_N
 \le Ce^{-\kappa s}e^{\kappa\theta}
       \|\phi\|_{N,r,\delta}^{\sharp}.
\end{equation*}
If \(s+\theta<0\), translation of the prescribed history gives
\begin{equation*}
 e^{2\kappa\theta}\|\phi(s+\theta)\|_N
 \le e^{-2\kappa s}\|\phi\|_{N,r,\delta}^{\sharp}.
\end{equation*}
Taking the supremum proves
\cref{eq:frozen-attracting-history-estimate}.  Notice that the old-history
shift is estimated directly; a spectral bound alone would not prove this
semigroup estimate.

Now let \(r<0\).  Define the unperturbed unstable history embedding
\begin{equation}
 (\jmath_N^ua)(\theta)
 =e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)\theta}a,\qquad
 a\in\Ran P_N^u(r),\quad -\theta_m\le\theta\le0,    \label{eq:frozen-base-unstable-embedding}
\end{equation}
and the fixed complementary coordinate space
\begin{equation}
 \mathscr K_N(r)=\{\psi\in\mathscr X_N:
                         P_N^u(r)\psi(0)=0\}.       \label{eq:frozen-base-stable-coordinate}
\end{equation}
The map \(\jmath_N^u\), current evaluation, and both coordinate
projections are bounded uniformly in the fading norm.  Every history has a
unique representation \(\psi+\jmath_N^ua\).

For a given \(\psi\in\mathscr K_N(r)\), seek
\(a\in\Ran P_N^u(r)\) and a future path \(y\) with \(\|y\|_+<\infty\).
Assemble the old history as
\(\widehat y(\theta)=\psi(\theta)+(\jmath_N^ua)(\theta)\).  The
Lyapunov--Perron equations are
\begin{align}
 y(s)={}&e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}
              P_N^s(r)\psi(0)\notag\\
 &+\int_0^s e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)(s-u)}P_N^s(r)
       \mathcal R_\eta[\widehat y_u]\,\dd u\notag\\
 &-\int_s^\infty e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)(s-u)}P_N^u(r)
       \mathcal R_\eta[\widehat y_u]\,\dd u,        \label{eq:frozen-stable-LP}\\
 a={}&-\int_0^\infty
       e^{-\delta^{-1}\mathcal J_N^{\rm fv}(r)u}P_N^u(r)
       \mathcal R_\eta[\widehat y_u]\,\dd u.        \label{eq:frozen-stable-boundary}
\end{align}
The last identity is simply the unstable projection of the first one at
\(s=0\), so endpoint compatibility is exact.  The stable forward integral
has denominator
\(\mathfrak a_N(r)/\delta-\kappa\), and the unstable future integral has
denominator \(\mathfrak a_N(r)/\delta+\kappa\).  Together with
\cref{eq:frozen-old-history-bound}, this makes
\cref{eq:frozen-stable-LP,eq:frozen-stable-boundary} a contraction with
constant bounded by \cref{eq:frozen-LP-number}.  Its unique solution obeys
\begin{equation}
 \|y\|_++\|a\|_N
 \le C\|\psi\|_{N,r,\delta}^{\sharp}.               \label{eq:frozen-stable-LP-bound}
\end{equation}
Linearity defines a bounded graph map
\(F^s_{N,\delta,r,\eta}:\mathscr K_N(r)\to\Ran P_N^u(r)\), with
\begin{equation}
 \|F^s_{N,\delta,r,\eta}\|
 \le Cq_{N,\delta}(r),                               \label{eq:frozen-stable-graph-small}
\end{equation}
and
\begin{equation}
 \mathscr W^s_{N,\delta,r,\eta}
 =\{\psi+\jmath_N^uF^s_{N,\delta,r,\eta}\psi:
                 \psi\in\mathscr K_N(r)\}.         \label{eq:frozen-stable-history-graph}
\end{equation}
Conversely, variation of constants for any solution with finite
\(\|y\|_+\), followed by unstable projection and passage of the terminal
time to infinity, recovers
\cref{eq:frozen-stable-LP,eq:frozen-stable-boundary}.  The graph therefore
characterizes exactly the histories whose future decays at this rate.  It is
invariant by uniqueness.  The same two-case history-segment estimate used
above proves \cref{eq:frozen-repelling-stable-estimate}.

For the complementary line, solve directly on \((-\infty,0]\).  Given
\(a\in\Ran P_N^u(r)\), seek \(y\) satisfying
\begin{align}
 y(s)={}&e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)s}a\notag\\
 &+\int_0^s e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)(s-u)}P_N^u(r)
       \mathcal R_\eta[y_u]\,\dd u\notag\\
 &+\int_{-\infty}^s
       e^{\delta^{-1}\mathcal J_N^{\rm fv}(r)(s-u)}P_N^s(r)
       \mathcal R_\eta[y_u]\,\dd u.                 \label{eq:frozen-unstable-LP}
\end{align}
On the space with norm
\(\sup_{s\le0}e^{-\kappa s}\|y(s)\|_N\), a delayed value lies farther in
the decaying direction:
\begin{equation*}
 e^{-\kappa s}\|y(s-\theta_k)\|_N
 \le e^{-\kappa\theta_k}
       \sup_{u\le0}e^{-\kappa u}\|y(u)\|_N.
\end{equation*}
The two Green integrals hence give a contraction no larger than
\cref{eq:frozen-LP-number}.  This constructs a one-dimensional space of
complete backward histories, which is invariant by autonomy and uniqueness.
For its history at zero, the same contraction estimate gives explicitly
\begin{equation}
 \|(y_a)_0-\jmath_N^ua\|_{N,r,\delta}^{\sharp}
 \le Cq_{N,\delta}(r)\|a\|_N.                       \label{eq:frozen-unstable-graph-small}
\end{equation}
The stable and unstable graphs are \(o(1)\) perturbations of complementary
base graphs, so they remain complementary and their projections are bounded
by one common constant.  This proves
\cref{eq:frozen-repelling-history-splitting}.

For completeness, invariance of the one-dimensional line gives a continuous
one-dimensional group.  Hence there is a real \(\lambda^u\) and a normalized
current vector \(e^u\), with \(\ell_{c,N}(r)e^u=1\), for which its histories
have the form \cref{eq:frozen-unstable-history}.  Projecting
\cref{eq:frozen-unstable-LP} onto the current stable graph gives
\begin{equation}
 \|e^u-e_{c,N}(r)\|_N\le C\delta^2.                 \label{eq:frozen-unstable-vector-proof-bound}
\end{equation}
Indeed, the stable current Green denominator is
\(\lambda^u+D\gamma/(2\delta)\), while the delayed coefficient is
\(O(\delta)\); hence the stable resolvent is \(O(\delta)\), without first
using the root expansion.  Substitution of
\(e^{\lambda^us}e^u\) into \cref{eq:frozen-voltage-rfde}, followed by
\(\ell_{c,N}(r)\), yields
\begin{equation}
 \lambda^u-\delta^{-1}\lambda_{c,N}(r)
 =\delta K\sum_{k=0}^m\ell_{c,N}(r)L_{k,N}(\eta)
       \{1-e^{-\lambda^u\theta_k}\}e^u.            \label{eq:frozen-unstable-scalar-equation}
\end{equation}
The right-hand side is \(O(\delta)\), because \(\lambda^u>0\) for small
\(\delta\).  This proves \cref{eq:frozen-unstable-root-estimate}.  The group
on the line now gives
\cref{eq:frozen-unstable-inverse-estimate}.

It remains to justify the structural derivatives.  The delay locations are
fixed, the map \(\eta\mapsto(L_{0,N}(\eta),\ldots,L_{m,N}(\eta))\) is
affine in the full operator-sum norm, and the fading weight is independent of
\(\eta\).  Thus every Lyapunov--Perron map above is twice Fr\'{e}chet
differentiable on one fixed Banach space, and its first two derivatives have
common bounds.  Differentiating the contraction equation once and twice and
using \((1-q_{N,\delta}(r))^{-1}\le2\) gives common bounds for the graph
maps.  The usual graph-projection formulas give the same bounds for the
projections.  Differentiating
\cref{eq:frozen-unstable-scalar-equation} and the stable current equation
gives the asserted bounds for \(\lambda^u,e^u\), and
\cref{eq:frozen-unstable-history}.  More explicitly, solve the stable
current equation for \(e^u=e_{c,N}+e^s(\lambda,\eta)\); its stable resolvent
and its first two \((\lambda,\eta)\)-derivatives are uniformly bounded at the
displayed scales.  After moving the right side of
\cref{eq:frozen-unstable-scalar-equation} to the left, the derivative of the
resulting scalar residual with respect to \(\lambda\) is
\(1+O(\delta)\), because the atoms \(\theta_k\) are fixed.  It is at least
one half for small \(\delta\), so the ordinary implicit-function theorem
supplies the claimed \(C^2_\eta\) root and vector.  No moving evaluation atom
is differentiated.

Finally, complexify the phase space and all graph maps.  Every characteristic
solution is then a phase-space orbit and splits according to
\cref{eq:frozen-repelling-history-splitting}.  The stable estimate excludes a
characteristic exponent with real part greater than \(-\kappa\), except for
the displayed unstable line; the attracting estimate gives the analogous
exclusion there.  A generalized vector at \(\lambda^u\) would produce a
second linearly independent, nondecaying generalized history, contradicting
the one-dimensional unstable summand: indeed,
\cref{eq:frozen-unstable-root-estimate,eq:frozen-history-rate} give
\(\lambda^u>\kappa\), so every polynomial multiple of
\(e^{\lambda^us}\) lies in the backward weighted class used in
\cref{eq:frozen-unstable-LP}.  Thus the displayed real root is
algebraically simple.  This completes the proof.
\end{proof}

\begin{proof}[Proof of \cref{prop:exact-outer-history-equation}]
If \(T=\delta^2t\), then
\[
 \frac{\dd}{\dd t}V(r(T))
 =\delta^2q(r)V'(r),\qquad
 \frac{\dd}{\dd t}w(r(T))
 =\delta^2q(r)w'(r).
\]
The physical delay \(\theta_k/\delta\) in original time is the slow-time
delay \(\delta\theta_k\).  Therefore
\[
 r\!\left(T-\delta\theta_k\right)
 =\Phi_q^{-\delta\theta_k}(r(T))=r_k(r(T)),
\]
and the delayed voltage in the raw RFDE is \(V(r_k(r(T)))\).  Substitute
\(V=\mathbf1+r\mathbf1+z\), use
\(\mathcal A_N\mathbf1=0\), and note that
\(\mathcal A_Nz=A_Nz\).  The voltage equation is then exactly
\cref{eq:exact-outer-voltage-invariance}.  Since
\[
 \pi_N^TV(r)-a
 =1+r+\pi_N^Tz(r)-\{1+\delta^2\nu\}
 =r-\delta^2\nu,
\]
the resource equation is exactly
\cref{eq:exact-outer-resource-invariance}.  Since every retained \(r\) may
serve as the initial value \(r_0\), evaluating the generated equation at
\(t=0\) proves necessity at every retained point.  The same substitution in
reverse proves sufficiency on every orbit segment for which all displayed
points remain in \(J\).

It remains to justify the regularity used below.  From
\cref{eq:exact-outer-resource-invariance},
\[
 w'(r)=\frac{r-\delta^2\nu}{q(r)}
\]
is \(C^1\), so \(w\) is \(C^2\) in the interior.  The scalar flow and each
backtrack \(r_k\) are \(C^1\).  Consequently the right-hand side of
\cref{eq:exact-outer-voltage-invariance} is \(C^1\); division by the nonzero
\(q\) shows that \(z'\), hence \(V'\), is \(C^1\).  Thus \(V\) is \(C^2\)
there as claimed.

For fold time \(s=\delta t\), an offset \(\theta\) corresponds to the
slow-time offset \(\delta\theta\), which proves
\cref{eq:exact-outer-history-embedding}.  Current evaluation and
\(\pi_N^TV'(r)=1+\pi_N^Tz'(r)=1\) give
\cref{eq:exact-outer-history-phase}.  Finally,
\(T=\delta s\).  Defining \(r_*(s)=r(\delta s)\) gives
\(r_*'(s)=\delta q(r_*(s))\), and differentiation of the entire history
embedding gives
\[
 \frac{\dd}{\dd s}\mathcal H_{V,w,q}(r_*(s))
 =\delta q(r_*(s))\partial_r\mathcal H_{V,w,q}(r_*(s)).
\]
Division by its stationary current coordinate proves
\cref{eq:exact-outer-history-tangent}.
\end{proof}

\begin{proof}[Proof of \cref{prop:physical-backtrack-calculus}]
The flow definition is equivalent to
\begin{equation}
 \int_r^{r_j(r)}\frac{\dd u}{q(u)}=-\delta\theta_j.
                                                               \label{eq:backtrack-integral-identity}
\end{equation}
The sign and lower bound in \cref{eq:backtrack-q-class} give the first
estimate in \cref{eq:backtrack-basic-identities}.  Differentiation with
respect to \(r\) gives
\[
 -\frac1{q(r)}+\frac{r_j'(r)}{q(r_j(r))}=0,
\]
which is its second identity.  Differentiation of
\cref{eq:backtrack-integral-identity} in a direction \(h\) gives
\[
 \frac{D_qr_j[h]}{q(r_j)}
 -\int_r^{r_j}\frac{h(u)}{q(u)^2}\,\dd u=0,
\]
and hence \cref{eq:backtrack-first-variation}.

We next prove the differentiability without losing an unnecessary third
spatial derivative of \(q\).  For fixed
\(0\le\tau\le\delta\theta_m\), set
\[
 R_q(\tau,r)=\Phi_q^{-\tau}(r)
\]
and consider the fixed-domain equation
\begin{equation}
 \mathscr H_\tau(q,R)(r)
 =\int_r^{R(r)}\frac{\dd u}{q(u)}+\tau=0.           \label{eq:backtrack-implicit-map}
\end{equation}
The strict inclusion \(J_0\Subset J\) persists under sufficiently small
\(C^1\) perturbations of \(q\) and \(R\).  Hence this equation is defined on
an open product neighborhood.  More explicitly, writing
\(C^a_\rho(K)\) for \(C^a(K)\) with
\cref{eq:backtrack-weighted-norm}, it is a map
\[
 \mathscr H_\tau:
 C^{k+2}_\rho(J)\times\mathcal U_k(I,J_0)longrightarrow C^k_\rho(I),
\]
where \(\mathcal U_k(I,J_0)\) is the open set of \(C^k\) maps whose range
lies in \(\operatorname{int}J_0\).

On either sign component set \(y=\log|r|\).  For each fixed \(k\), the norm
\cref{eq:backtrack-weighted-norm} is uniformly equivalent to
\begin{equation}
 \max_{0\le a\le k}\sup_y
   \|\partial_y^a f(\operatorname{sgn}(J)e^y)\|,    \label{eq:backtrack-log-norm}
\end{equation}
because \((r\partial_r)^a\) is a triangular combination of
\(r^b\partial_r^b\), \(b\le a\).  In these equivalent weighted spaces the
map \(\mathscr H_\tau\) is \(C^3\): three derivatives with respect to its
upper endpoint differentiate \(q^{-1}\) only twice, and therefore
\(q,h\in C^{k+2}\) suffice after \(k\) spatial derivatives.  Moreover,
\[
 D_R\mathscr H_\tau(q,R)v=\frac{v}{q(R)}.
\]
Its inverse is multiplication by \(q(R)\), with norm bounded solely by the
common constants in \cref{eq:backtrack-q-class}.  The Banach implicit-function
theorem therefore makes \(q\mapsto R_q(\tau,\cdot)\) three times
Fr\'{e}chet differentiable in the stated spaces.

Differentiating \cref{eq:backtrack-implicit-map} with respect to \(\tau\)
gives \(\partial_\tau R_q=-q(R_q)\).  Since \(R_q(0,r)=r\), integration in
\(\tau\) proves the first line below.  For the parameter derivatives,
differentiate \cref{eq:backtrack-implicit-map} successively.  Every pure
\(q\)-derivative is an integral over \([r,R_q]\), hence has a factor
\(C\tau\); every endpoint term contains a lower variation of \(R_q\).
Induction on the parameter order, followed by multiplication by the uniformly
bounded inverse of \(D_R\mathscr H_\tau\), gives, for \(1\le b\le3\),
\begin{align}
 \max_{a\le k}\sup_y
 \left|\partial_y^a\{R_q(\tau,r)-r\}\right|
 &\le C\tau,\notag\\
 \max_{a\le k}\sup_y
 \left|\partial_y^aD_q^bR_q(\tau,r)
          [h_1,\ldots,h_b]\right|
 &\le C\tau\prod_{\ell=1}^b
       \|h_\ell\|_{k+2,\rho;J}.                    \label{eq:backtrack-fixed-time-jets}
\end{align}
Here the integral and endpoint Nemytskii estimates are uniform because
\(|R_q/r|\) stays between two fixed positive constants when
\(\epsilon_\rho\) is small.  This also proves directly that the constants do
not depend on \(\rho,\delta\), or the interval locations.  Taking
\(\tau=\delta\theta_j\) and using
\cref{eq:backtrack-log-norm} proves
\cref{eq:backtrack-map-bound,eq:backtrack-frechet-bounds}.

The exact second variation, useful for checking the signs, is
\begin{align}
 D_q^2r_j[h_1,h_2]
={}&\{h_2(r_j)+q'(r_j)R_2\}
       \int_r^{r_j}\frac{h_1}{q^2}
 +\frac{h_1(r_j)}{q(r_j)}R_2\notag\\
 &-2q(r_j)\int_r^{r_j}\frac{h_1h_2}{q^3},
 \qquad R_2=D_qr_j[h_2].                            \label{eq:backtrack-second-variation}
\end{align}
It is the derivative of \cref{eq:backtrack-first-variation}; one further
derivative is covered by \cref{eq:backtrack-fixed-time-jets}.

The collar geometry and the first displacement bound give
\begin{equation}
 \frac12\le\frac{|r_j(r)|}{|r|}\le\frac32,\qquad
 \left|\log\frac{|r_j(r)|}{|r|}\right|
 \le C\epsilon_\rho,                               \label{eq:backtrack-log-shift}
\end{equation}
after decreasing \(\epsilon_0\).  In fact, if
\(Y_q(\tau,y)=\log|R_q(\tau,\operatorname{sgn}(J)e^y)|\), then
\cref{eq:backtrack-fixed-time-jets} and smoothness of the logarithm on the
displayed ratio interval give the stronger bound
\begin{equation}
 \|Y_q(\tau,\cdot)-\Id\|_{C^k_y}
 +\sum_{b=1}^3\|D_q^bY_q(\tau,\cdot)\|_{
      \mathcal L^b(C^{k+2}_\rho(J);C^k_y)}
 \le C\frac{\tau}{\rho},                           \label{eq:backtrack-log-jet-shift}
\end{equation}
Fa\`a di Bruno's formula in the \(y\)-coordinate now proves
\cref{eq:backtrack-composition-bound}.  Apply the fundamental theorem of
calculus to the full \(C^k_y\) displacement in
\cref{eq:backtrack-log-jet-shift} to obtain
\cref{eq:backtrack-composition-difference}.  For every positive parameter
order, each Fa\`a di Bruno term contains at least one variation from
\cref{eq:backtrack-fixed-time-jets}.  A term with \(b\) such factors is
bounded first by \(C\epsilon_\rho^b\); since
\(\epsilon_\rho\le\epsilon_0\), it is at most \(C\epsilon_\rho\).
This proves all three lines of
\cref{eq:backtrack-composition-frechet}, including the third derivative.
The same calculation applied to the joint map \((f,q)\mapsto f\circ r_j(q)\)
has either no variation of \(f\), or exactly one such variation.  Applying it
instead to
\[
 f(\operatorname{sgn}(J)e^{Y_q(y)})
   -f(\operatorname{sgn}(J)e^y)
\]
leaves a factor from \cref{eq:backtrack-log-jet-shift} in every term and
proves \cref{eq:backtrack-joint-difference}; dropping that factor gives the
asserted bound for \(\mathcal C_j\).  The rectangular parameter statement is
now the third-order joint Banach-space chain rule.  Substitution of
\(\rho_\delta=\delta S_\delta/(2\alpha)\) proves
\cref{eq:backtrack-logarithmic-loss}.
\end{proof}

\begin{proof}[Proof of \cref{prop:principal-tracker-endpoint-green}]
The high regularity in
\cref{eq:outer-critical-high-regularity}, the polynomial formula for
\(D\mathcal N_N\), and the critical expansions give
\cref{eq:principal-green-coefficients}.  The sign and zeroth-order speed
bounds are \cref{eq:outer-speed-sign,eq:outer-speed-inputs}.  Repeated
differentiation of
\[
 q_0(r)=\frac{r-\varepsilon\nu}{w_{0,N}'(r)}
\]
and \(|r|\ge\rho_\delta\) gives
\cref{eq:principal-green-q-bounds}; every apparently singular term contains
at most a power of \(\varepsilon/|r|\).  In particular, differentiating
\(q_0\) uses the first, but not the second, of the two ratios
\begin{equation}
 \frac{\varepsilon}{\rho_\delta}
 =\frac{2\alpha\delta}{S_\delta},
 \qquad
 \frac{\varepsilon}{\rho_\delta^2}
 =\frac{4\alpha^2}{S_\delta^2}.                    \label{eq:principal-green-small-ratios}
\end{equation}
After decreasing \(\delta_0\), both ratios are uniformly bounded and tend
to zero.  The second ratio enters later, when the scalar Green equation is
normalized by \(|w_{0,N}'|\asymp|r|\); it is not produced by
differentiating \(q_0\).

Write a dot for differentiation along \(\dot r=q_0(r)\).  The coordinate
\cref{eq:principal-green-p-coordinate} gives the exact system
\begin{align}
 \varepsilon\dot Z
 &=\mathcal C_N(r)Z-z_{0,N}'(r)\mathfrak p+f,\notag\\
 \varepsilon\dot{\mathfrak p}
 &=w_{0,N}'(r)\mathfrak p
   -w_{0,N}'(r)\ell_N(r)Z-\varepsilon g,\notag\\
 \mathcal C_N(r)&=B_N(r)+z_{0,N}'(r)\ell_N(r).     \label{eq:principal-green-transformed-system}
\end{align}
Indeed, the second row of
\cref{eq:principal-tracker-operator} reads
\[
 g=-\frac{\dd}{\dd T}\{\varepsilon Q+\ell_NZ\}+w_{0,N}'Q,
\]
which proves the second equation; the first follows from
\(\varepsilon Q=\mathfrak p-\ell_NZ\).  This is the cancellation that the
untransformed derivative coupling conceals.

The transverse Markov estimate gives
\(\|e^{A_NT}\|_{E_N\to E_N}\le e^{-D\gamma T}\) for \(T\ge0\).
Since \(\|\mathcal C_N(r)-A_N\|\le Cr_{\rm out}\), bounded-perturbation variation of
constants yields the evolution estimate
\begin{equation}
 \|U_C(T,S)\|
 \le C e^{-D\gamma(T-S)/(2\varepsilon)},
 \qquad T\ge S.                                    \label{eq:principal-green-C-evolution}
\end{equation}
The scalar evolution is
\begin{equation}
 u_q(T,S)=\exp\left\{\varepsilon^{-1}
      \int_S^T w_{0,N}'(r(\sigma))\,\dd\sigma\right\}.       \label{eq:principal-green-scalar-evolution}
\end{equation}
It is forward stable on \(I_\delta^a\); on \(I_\delta^r\), its restriction
to the scalar line has a backward-stable inverse.  No ambient backward RFDE
evolution is being used.

We construct one slow particular solution.  Pass to
\(x=\log(|r|/\rho_\delta)\) and first apply a fixed-width linear jet
extension to the right-hand sides and coefficients at each endpoint.  At
the far side of that collar, freeze the coefficients, preserve the stable
matrix bound and the sign of \(w_{0,N}'\), and attach an artificial
\(T\)-halfline that is no longer required to have the form
\(r=\pm\rho_\delta e^x\).  Extend the data with the same linear operator
and freeze the endpoint weight.  Thus the repelling \(Z\)-row has a genuine
past halfline and its scalar row has a genuine future halfline; on the
attracting branch both rows have genuine past halflines.  The extensions
have the same weighted \(C^s\) bounds, uniformly in the length of the
physical logarithmic interval.

Writing tildes for the extended objects, the three diagonal Green kernels
are explicitly
\begin{align}
 (\mathscr G_ZF)(T)
 &=\varepsilon^{-1}\int_{-\infty}^T
       \widetilde U_C(T,S)\widetilde F(S)\,\dd S,\notag\\
 (\mathscr G_{\mathfrak p}^aH)(T)
 &=\varepsilon^{-1}\int_{-\infty}^T
       \widetilde u_q(T,S)\widetilde H(S)\,\dd S,\notag\\
 (\mathscr G_{\mathfrak p}^rH)(T)
 &=-\varepsilon^{-1}\int_T^\infty
       \widetilde u_q(T,S)\widetilde H(S)\,\dd S.  \label{eq:principal-green-diagonal-kernels}
\end{align}
Use \(\mathscr G_Z\) for the transverse row on both branches,
\(\mathscr G_{\mathfrak p}^a\) on the attracting branch, and
\(\mathscr G_{\mathfrak p}^r\) on the repelling branch.  Their block fixed
point has inputs
\(F=f-z_{0,N}'\mathfrak p\) and
\(H=-w_{0,N}'\ell_NZ-\varepsilon g\).
Restriction to the retained interval solves the original equation.  The
halfline is only a device for selecting a bounded right inverse; it is not a
history of the RFDE and is not a physical preparation.

For clarity, the logarithmic norm equivalences behind the estimate are
\begin{align}
 \|Z\|_{\mathsf Z_\varepsilon^s}
 &\asymp\varepsilon^{-1}\|Z\|_{C_x^s},&
 \|\mathfrak p\|_{\mathsf P_\varepsilon^s}
 &\asymp\varepsilon^{-1}\|\mathfrak p/r\|_{C_x^s},&
 \|Q\|_{\mathsf Q_\varepsilon^s}
 &\asymp\varepsilon^{-1}\|rQ\|_{C_x^s}.          \label{eq:principal-green-log-norms}
\end{align}
In these norms the direct scalar source obeys
\begin{equation}
 \|\mathscr G_{\mathfrak p}^{a/r}(-\varepsilon g)\|_{
       \mathsf P_\varepsilon^s}
 \le C_s\frac{\varepsilon}{\rho_\delta^2}
       \|(0,g)\|_{\mathsf Y_\varepsilon^s},       \label{eq:principal-green-g-source}
\end{equation}
because the scalar denominator is \(|w_{0,N}'|\asymp|r|\).  The two
off-diagonal Green maps satisfy
\begin{equation}
 \|\mathcal K_{Z\mathfrak p}\|\le Cr_{\rm out}^2,
 \qquad
 \|\mathcal K_{\mathfrak p Z}\|\le C,
 \qquad
 \|\mathcal K_{Z\mathfrak p}\mathcal K_{\mathfrak p Z}\|
 \le Cr_{\rm out}^2.                               \label{eq:principal-green-loop}
\end{equation}
The first estimate uses \(z_{0,N}'=O(r)\),
\(\mathfrak p=O(\varepsilon|r|)\), and the fixed transverse Green
denominator.  The second uses
\(w_{0,N}'\ell_N=O(r^2)\) and the scalar Green denominator
\(|w_{0,N}'|\asymp|r|\).  Hence the block system is invertible after one
fixed decrease of \(r_{\rm out}\).

To make the derivative induction explicit, put
\(\mathcal D=r\partial_r=(r/q_0)\partial_T\).  Commuting
\(\mathcal D^j\) through the equations produces three kinds of lower-order
terms.  Derivatives of
\(\mathcal C_N-A_N,z_{0,N}',\ell_N,w_{0,N}'\) retain their respective structural
sizes and hence the same diagonal perturbation and off-diagonal bounds as in
\cref{eq:principal-green-loop}.  Ratios such as
\(\mathcal Dw_{0,N}'/w_{0,N}'\) are merely \(O(1)\); they multiply
lower-order unknowns already controlled by induction and need no smallness.
Only commutators with the transport coefficient
\(\varepsilon q_0/r\) carry \(C_s\varepsilon/|r|\).  After the scalar row
is normalized by both \(|w_{0,N}'|\asymp|r|\) and the
\(\mathfrak p/r\) coordinate, its singular transport ratio is
\(C_s\varepsilon/|r|^2\).  The triangular equivalence between
\(\mathcal D^j\) and \(r^j\partial_r^j\), together with
\cref{eq:principal-green-small-ratios}, lets these terms be absorbed at
each derivative order.  The direct estimates
\cref{eq:principal-green-g-source,eq:principal-green-loop} then close the
same two-by-two block induction successively through order \(s\).  This proves the
\(\mathsf Z_\varepsilon^s\times\mathsf P_\varepsilon^s\) part of
\cref{eq:principal-green-slow-bound}.

To recover the original speed component, do not select another solution of
a scalar equation.  The transformed second row gives the exact identity
\begin{equation}
 Q=\frac{\mathfrak p-\ell_NZ}{\varepsilon}
  =\frac{g+q_0\mathfrak p'}{w_{0,N}'}.             \label{eq:principal-green-Q-recovery}
\end{equation}
If \(\mathfrak p\in\mathsf P_\varepsilon^s\), then the numerator obeys
\begin{equation}
 \left|\partial_r^j\{g+q_0\mathfrak p'\}\right|
 \le C_s\varepsilon|r|^{-j},\qquad0\le j\le s-1. \label{eq:principal-green-Q-forcing}
\end{equation}
The coefficient bounds for \(1/w_{0,N}'\), whose \(j\)-th derivative is
\(O(|r|^{-j-1})\), prove the
\(\mathsf Q_\varepsilon^{s-1}\) estimate.  This is the algebraic component
of the same transformed solution, so it solves both rows of the original
operator with no uniqueness bridge left implicit.  With one additional
input and coefficient derivative, the transformed construction gives
\(\mathfrak p\in\mathsf P_\varepsilon^{s+1}\), and the same identity proves
\cref{eq:principal-green-extra-q-derivative}.

It remains to impose finite endpoint traces.  For the homogeneous transformed
system, the attracting lift is the fixed point of
\begin{align}
 Z(T)&=U_C(T,0)b_z-\varepsilon^{-1}
   \int_0^TU_C(T,S)z_{0,N}'(S)\mathfrak p(S)\,\dd S,\notag\\
 \mathfrak p(T)&=u_q(T,0)b_p-\varepsilon^{-1}
   \int_0^Tu_q(T,S)w_{0,N}'(S)\ell_N(S)Z(S)\,\dd S. \label{eq:principal-green-attracting-LP}
\end{align}
On the repelling branch the mixed lift is the fixed point of
\begin{align}
 Z(T)&=U_C(T,0)b_s-\varepsilon^{-1}
   \int_0^TU_C(T,S)z_{0,N}'(S)\mathfrak p(S)\,\dd S,\notag\\
 \mathfrak p(T)&=u_q(T,T_r)b_u+\varepsilon^{-1}
   \int_T^{T_r}u_q(T,S)w_{0,N}'(S)\ell_N(S)Z(S)\,\dd S.
                                                               \label{eq:principal-green-repelling-LP}
\end{align}
The loop estimate \cref{eq:principal-green-loop} gives existence and
uniqueness.  It also shows directly that the terminal
\(\mathfrak p\)-row is uniformly transverse to the scalar unstable mode.

By linearity, split the repelling datum as
\((b_s,b_u)=(b_s,0)+(0,b_u)\).  Close the first fixed point in the norm
weighted by \(e^{c\Phi_s}\), and the second in the norm weighted by
\(e^{c\Phi_u}\).  The fast kernel preserves both weights after decreasing
\(r_{\rm out}\), because
\(|w_{0,N}'|\le C r_{\rm out}\) while its decay rate is bounded below by
\(D\gamma/(2\varepsilon)\); the same
\(O(r_{\rm out}^2)\) loop closes both fixed points.  Thus the stable datum
does not acquire a growing terminal scalar tail, and the terminal datum does
not acquire a growing initial fast tail.  The attracting integral equations
close similarly in the \(e^{c\Phi_a}\)-weighted norm.  Consequently the two
integral equations give
\cref{eq:principal-green-attracting-layer,eq:principal-green-repelling-layer}.
Successive time differentiation is triangular after multiplying the
\(j\)-th derivative by \(\varepsilon^j\).  Finally,
\(\dd T=\dd r/q_0(r)\), \(|q_0|\asymp1\), and
\(|w_{0,N}'(r)|\asymp|r|\) give
\cref{eq:principal-green-action-comparison}.

The trace identities follow at \(T=0\) and \(T=T_r\) in the two integral
systems.  Subtracting the slow particular solution from any solution with a
prescribed trace leaves the corresponding homogeneous lift.  This proves
\cref{eq:principal-green-lifts,eq:principal-green-augmented-inverse} and
uniqueness.
\end{proof}

\begin{proof}[Proof of
\cref{prop:nonlinear-resource-defect-normal-form}]
Projecting \cref{eq:exact-outer-voltage-invariance} onto \(E_N\), and using
the critical-curve equation, gives
\begin{equation}
 \varepsilon qZ'
 =B_NZ-\varepsilon Qz_{0,N}'-R_{z,N}
       -h_{\perp,N}+\varepsilon\Delta\mathcal D_{\perp,N}.
                                                               \label{eq:nonlinear-tracker-pre-Z}
\end{equation}
The collective projection gives the exact reconstruction
\[
 w=\frac23-F_N(r,z)+\varepsilon\mathcal D_{c,N}
       -\varepsilon q.
\]
Since
\[
 F_N(r,z_{0,N}+Z)-F_N(r,z_{0,N})
 =\ell_NZ+h_{c,N},
\]
subtracting this formula from \(w_{\rm seed,N}\) proves
\cref{eq:nonlinear-resource-defect}.  Substitution of
\cref{eq:nonlinear-tracker-Q-reconstruction} into
\cref{eq:nonlinear-tracker-pre-Z} then proves
\cref{eq:nonlinear-tracker-Z-normal-form}, including all five signs.

For the scalar equation, write
\[
 w=w_{\rm seed,N}-\mathfrak p_{\rm nl},\qquad
 w_{\rm seed,N}'
 =w_{0,N}'+\varepsilon(\mathcal D_{0,c,N}-q_0)'.
\]
The exact resource identity and \(q_0w_{0,N}'=r-\varepsilon\nu\) imply
\[
 -q\mathfrak p_{\rm nl}'+w_{0,N}'Q
 =-\varepsilon q(\mathcal D_{0,c,N}-q_0)'.
\]
Multiply by \(\varepsilon\) and use
\cref{eq:nonlinear-tracker-Q-reconstruction}; this is exactly
\cref{eq:nonlinear-tracker-p-normal-form}.  Reversing the calculation proves
equivalence with the two projected tracker identities.

At \(Z=Q=\mathfrak p_{\rm nl}=0\), the two quadratic remainders and the
value of \(\Delta\mathcal D_N\) vanish.  Its differential need not vanish.
The last term in the scalar equation is \(\varepsilon R_{q,N}\), proving
the affine-value statement.  Differentiating the three identities gives
\cref{eq:nonlinear-tracker-full-differential}; omitting precisely its
displayed delay/backtrack corrections leaves the principal retained block.

It remains to verify the local coordinate assertion.  Differentiating the
integral backtrack identity in a speed direction \(H\) gives
\[
 D_qr_k^q[H]
 =q(r_k^q)\int_r^{r_k^q}\frac{H(u)}{q(u)^2}\,\dd u.
\]
The common sign bounds and the interval length

\begin{proof}[Proof of \cref{prop:compatible-endpoint-jet-collar}]
Choose one-sided scalar Hermite functions \(\chi_0,\ldots,\chi_h\) on the
half-line with the displayed endpoint jets and with all jets zero at the
left edge of their support.
For \(0\le i\le h\),
\[
 \partial_\theta^i
 \{\delta^ja_j\chi_j(\theta/\delta)\}
 =\delta^{j-i}a_j\chi_j^{(i)}(\theta/\delta).
\]
Multiplication by \(\delta^i\), followed by the supremum, proves
\cref{eq:collar-Hermite-bound}.  This argument uses only scalar
multiplication and is therefore dimension uniform.

We describe the compatibility recursion.  If \(v_s=\phi\) and \(w\) are a
classical solution of
\[
 v'=\mathfrak F_{N,\delta,\nu,\eta}(v_s,w),
 \qquad w'=\mathfrak G_{N,\delta,\nu}(v_s),
\]
then repeated time differentiation at zero expresses \(v^{(j)}(0)\) as a
smooth function \(\mathfrak V_{j,N}\) of the endpoint voltage jets through
order \(j-1\), the voltage jets at the delayed points through order \(j-1\),
and the current resource.  Every resource derivative in that expression is
eliminated recursively with \(\mathfrak G\).  Retardedness is what prevents
the unknown endpoint jet of order \(j\) from appearing on the right.

Write \(\phi=\widehat\gamma+\mathcal E_{\delta,h}a\).  The support condition
and \(4c_H\delta<\theta_+\) imply that the Hermite term and all its derivatives
vanish at every strictly delayed point.  A zero-delay summand is identically
zero in \(\mathfrak F\).  Set
\[
 a_0=V_{0,N}(e)+Z_e,\qquad
 \omega_e=w_{\rm seed,N}(e)-p_e,
\]
and define successively
\[
 a_j=\mathfrak V_{j,N}
 \bigl(\widehat\gamma+\mathcal E_{\delta,h}
        (a_0,\ldots,a_{j-1},0,\ldots,0),
       \omega_e;\delta,\nu,\eta\bigr),
 \qquad1\le j\le h.
                                                               \label{eq:collar-jet-recursion}
\]
The endpoint-jet property of \(\mathcal E_{\delta,h}\) proves
\cref{eq:collar-current-data,eq:collar-compatibility}.  Conversely, those
conditions force the same recursion, so the chart is unique.  There are
exactly \(hN\) unknown endpoint derivative components and \(hN\)
compatibility identities; neither the current value nor the resource defect
adds a Green boundary row.

For completeness, we record why the singular factors are tame in
\cref{eq:strong-collar-norm}.  In the \(j\)-th compatibility identity,
multiplication by \(\delta^j\) cancels the factor \(\delta^{-1}\) in the
current field against the already controlled order-\((j-1)\) jet.  A
strictly delayed order-\((j-1)\) jet enters with the factor
\(\delta^{j+1}\).  Its strong norm costs at most
\(\delta^{-(j-1)}e^{2\kappa_e\theta_m}\), so the net coefficient is bounded
by
\begin{equation}
 C\delta^2e^{2\kappa_e\theta_m}
 \le C\delta^{2-2\vartheta}.
                                                               \label{eq:collar-delay-jet-factor}
\end{equation}
The polynomial current terms and the resource recursion obey the same
triangular estimate.  Induction in \(j\), the uniform product and operator
bounds, and \cref{eq:collar-Hermite-bound} prove
\cref{eq:collar-chart-tame-bound}.  Differentiating the finite triangular
recursion up to three times proves the asserted fixed-\(\theta\) Fr\'{e}chet
and rectangular parameter bounds; uniform third parameter derivatives of
\(w_{\rm seed,N}(e)\) follow from the physical backtrack calculus and the
high-order critical coefficients already proved.  The same factor
\cref{eq:collar-delay-jet-factor} controls every old-history evaluation.

Under the displayed monotonicity, \(r_{\phi_e}\) is a \(C^h\)
diffeomorphism onto its image.  The chain rule gives
\cref{eq:collar-derived-old-speed} and \(q_{\rm old}\in C^{h-1}\), including

\cref{eq:collar-backtrack-bridge}.  Taking the collective projection of the

using \(\omega_e=w_{\rm seed,N}(e)-p_e\) gives
\cref{eq:collar-speed-defect-bridge}.  The scalar trace identity follows
from the same resource definition.  Thus the pullback old speed and its

boundary conditions.

For clarity about parameter regularity, differentiating at fixed Eulerian

 D_\lambda q_{\rm old}\circ r_{\phi_e}
 =\delta^{-1}\left\{
   r_{\theta\lambda}

Iteration proves the stated \(C^{h-1-b}\) scale.  This is why the theorem
asserts full parameter jets for the fixed-\(\theta\) collar and its pulled

\end{proof}

\begin{proof}[Proof of \cref{prop:fold-time-tracker-normal-form}]
We prove the RFDE equivalence directly in fold time; no orbit condition is
used on the old history.  For the reconstruction in
\cref{eq:fold-time-tracker-reconstruction}, the collective projection of
the raw voltage field is
\begin{align}
 \pi_N^T\mathfrak F(v_s^{\rm hist},w)
 ={}&\delta^{-1}
   \{p-\ell_NZ-h_{c,N}
          -\varepsilon\mathcal D_{0,c,N}
          +\varepsilon q_0\}
       +\delta\mathcal D_{c,N}^{\rm ft}\notag\\
 ={}&\delta\left[
 q_0+\varepsilon^{-1}\{p-\ell_NZ-h_{c,N}\}
       +\Delta\mathcal D_{c,N}^{\rm ft}\right]
 =\delta q.                                       \label{eq:fold-time-direct-collective}
\end{align}
Here we used the critical-curve identity, the definition of
\(w_{\rm seed,N}\), and
\[
 F_N(r,z_{0,N}+Z)-F_N(r,z_{0,N})
 =\ell_NZ+h_{c,N}.
\]
Since \(\pi_N^T\partial_sv=r_s\), the collective raw equation is therefore
equivalent to \cref{eq:fold-time-tracker-r} and the algebraic speed
reconstruction.

For the transverse projection, the critical-curve identity and the Taylor
remainder give
\begin{equation}
 P_{\perp,N}\mathfrak F(v_s^{\rm hist},w)
 =\delta^{-1}\{B_NZ-h_{\perp,N}\}
       +\delta\mathcal D_{\perp,N}^{\rm ft}.       \label{eq:fold-time-direct-transverse}
\end{equation}
On the other hand,
\[
 P_{\perp,N}\partial_sv=r_sz_{0,N}'(r)+Z_s
               =\delta qz_{0,N}'+Z_s.
\]
Multiplication by \(\delta\), followed by
\cref{eq:fold-time-speed-reconstruction}, yields
\begin{align*}
 \delta Z_s
 &=B_NZ-h_{\perp,N}
       +\varepsilon\mathcal D_{\perp,N}^{\rm ft}
       -\varepsilon qz_{0,N}'\\
 &=\mathcal C_NZ-z_{0,N}'p-R_{z,N}
       +z_{0,N}'h_{c,N}-h_{\perp,N}
       +\varepsilon\{\Delta\mathcal D_{\perp,N}^{\rm ft}
                 -z_{0,N}'\Delta\mathcal D_{c,N}^{\rm ft}\},
\end{align*}
which is exactly \cref{eq:fold-time-tracker-Z}.  Every step is reversible.

Finally, \(p=w_{\rm seed,N}(r)-w\), \(r_s=\delta q\), and
\(q_0w_{0,N}'=r-\varepsilon\nu\) show that the raw resource equation is
equivalent to
\begin{align*}
 p_s
 &=\delta q\{w_{0,N}'
          +\varepsilon(\mathcal D_{0,c,N}-q_0)'\}
       -\delta(r-\varepsilon\nu)\\
 &=\delta w_{0,N}'Q
       +\delta\varepsilon q(\mathcal D_{0,c,N}-q_0)'.
\end{align*}
Multiplying by \(\delta\), and using the algebraic reconstruction of
\(\varepsilon Q\), gives precisely
\cref{eq:fold-time-tracker-p}.  This calculation is reversible as well,
and proves the equivalence of the full raw system with the three
fixed-delay identities.

It remains only to prove the conditional reparameterization statement.
Under \cref{eq:fold-time-old-history-speed}, \(q_{\rm ext}\) is
nonvanishing and \(C^1\) on the image of \(r\).  Therefore
\[
 \int_{r(s-\theta_k)}^{r(s)}
       \frac{\dd u}{q_{\rm ext}(u)}
 =\delta\int_{s-\theta_k}^{s}\dd\sigma
 =\delta\theta_k.
\]
This is \cref{eq:fold-time-backtrack-identification}.  On the current
interval \(q_{\rm ext}=q\), so the usual chain rule gives
\(\delta Z_s=\varepsilon q\,\partial_rZ\) and
\(\delta p_s=\varepsilon q\,\partial_rp\).  Consequently the fixed-delay
system becomes exactly
\cref{eq:nonlinear-tracker-Z-normal-form,eq:nonlinear-tracker-p-normal-form}
in the \(r\)-coordinate on the current image, and the delayed voltage
history is the extended-curve evaluation in
\cref{eq:exact-outer-history-embedding}.  This calculation uses no resource
equation on the prescribed old history and hence makes no past-orbit claim.
The same argument applies after the first-delay window without imposing an
orbit condition on the discarded initial history.

For a chart value \((\phi_e,\omega_e)=\Gamma_e^h(\cdots)\),
\cref{eq:fold-time-collar-interface} is simply the collective/transverse
decomposition of \(\phi_e\), together with
\cref{eq:collar-nonlinear-trace}.  Raw compatibility follows from
\cref{prop:compatible-endpoint-jet-collar}; no additional resource history
or boundary row is introduced.
\end{proof}

\begin{proof}[Proof of
\cref{prop:fixed-phase-green-collar-buffer}]
We first extract the order-zero oriented inverse already present in the proof
of \cref{prop:principal-tracker-endpoint-green}.  After the change
\(T=\delta s\), its transformed system is
\begin{equation}
 \delta Z_s=\mathcal C_NZ-z_{0,N}'p+F,\qquad
 \delta p_s=w_{0,N}'p-w_{0,N}'\ell_NZ+H.           \label{eq:fixed-phase-principal-s-system}
\end{equation}
The kernels in
\cref{eq:principal-green-diagonal-kernels}, with the block loop in
\cref{eq:principal-green-loop}, give zero-trace Green operators
\(\mathbb G_a^0,\mathbb G_r^0\) and homogeneous finite-trace lifts
\(\mathbb B_a^0,\mathbb B_r^0\).  At order zero their estimates read
\begin{align}
 \|\mathbb G_\sigma^0(F,H)\|_{\mathsf X_{\sigma,0}}
 &\le C\|(F,H)\|_{\mathsf Y_{\sigma,0}},\notag\\
 \|\mathbb B_a^0(b_z,b_p)\|_{\mathsf X_{a,0}}
 &\le C\left(\|b_z\|_N+\frac{|b_p|}{r_{\rm out}}\right),\notag\\
 \|\mathbb B_r^0(b_s,b_u)\|_{\mathsf X_{r,0}}
 &\le C\left(\|b_s\|_N+\frac{|b_u|}{r_{\rm out}}\right).    \label{eq:fixed-phase-principal-C0-bound}
\end{align}
Although \cref{prop:principal-tracker-endpoint-green} records the slow
spaces from order one upward, these finite-interval oriented integral
formulas define bounded operators directly for continuous sources; the same
operators also follow by \(C^0\)-completion of smooth sources.
For the scalar row this follows directly by dividing its Green integral by
\(|\bar r_\sigma(s)|\): the denominator
\(|w_{0,N}'|\asymp|r|\), while an \(H\)-source is measured by
\(|H|/|r|^2\).  Equivalently, the action comparison and
\(|q_0|\asymp1\) give the oriented estimate
\begin{equation}
 \sup_{s\in I_\sigma}\frac1{|\bar r_\sigma(s)|}
 \frac1\delta\int_{\mathcal O_\sigma(s)}
 |u_q(s,v)|\,|\bar r_\sigma(v)|^2\,\dd v\le C,    \label{eq:fixed-phase-moving-scalar-weight}
\end{equation}
where \(\mathcal O_a(s)=[0,s]\) and
\(\mathcal O_r(s)=[s,S_r]\), with the corresponding stable or future
scalar kernel.  The coefficient motion along
\(\partial_s\bar r_\sigma=\delta q_0\) is already included in these exact
nonautonomous kernels; no endpoint freezing is used.  On the repelling
branch, \(\mathbb G_r^0\) uses the forward transverse kernel and only the
one-dimensional future scalar kernel.

We next estimate the fixed-delay perturbation.  Write
\(w_{0,N}'(r)=r\,m_N(r)\), where \(m_N\) and its required derivatives are
uniformly bounded and \(|m_N|\ge c\).  The quotient formula
\[
 q_0(r)=\frac{r-\varepsilon\nu}{w_{0,N}'(r)}
\]
therefore gives \(q_0'=O(1+\varepsilon/r^2)=O(1)\) on
\(|r|\ge\rho_\delta\).  The exact backtrack identity
\((r_k^{q_0})'=q_0(r_k^{q_0})/q_0(r)\) then yields
\(1-(r_k^{q_0})'=O(\delta)\).  Differentiating the definition of
\(\mathcal D_{0,c,N}\), using \(z_{0,N}'=O(r)\),
\(z_{0,N}''=O(1)\), and the common layer bound, gives
\(\mathcal D_{0,c,N}'=O(\delta)\).  Hence
\(\mathfrak d_N=(\mathcal D_{0,c,N}-q_0)'=O(1)\).  Together with the critical
coefficient bounds, this proves, uniformly on both paths,
\begin{equation}
 \|z_{0,N}'\|_N+\|\ell_N\|\le C|r|,\qquad
 |w_{0,N}'|\asymp|r|,\qquad |\mathfrak d_N|\le C.  \label{eq:fixed-phase-coefficient-bounds}
\end{equation}
Split \(\mathscr D_N^{\rm lin}[\gamma,Z]\) according to whether the delayed argument
is current or lies in the prescribed old history.  The common layer bound
gives
\begin{align}
 \|\mathscr D_N^{{\rm lin},{\rm cur}}Z\|_{C^0}
 &\le C\|Z\|_{C^0},\notag\\
 \|\mathscr D_N^{{\rm lin},{\rm old}}\gamma\|_{C^0}
 &\le Ce^{2\kappa_e\theta_m}
       \|\gamma\|_{N,e,\delta}^{\sharp}.          \label{eq:fixed-phase-delay-split-bound}
\end{align}
The old part is supported in the first delay window.  Along that fixed
window \(|\bar r_\sigma(s)|\asymp|e|\); this remains dimension uniform
because \(|\partial_s\bar r_\sigma|\le C\delta\).

Eliminate \(Q\) in \cref{eq:fixed-phase-normal-block}.  The perturbing
source added to \cref{eq:fixed-phase-principal-s-system} is
\begin{align}
 \mathbb V_Z
 &=\varepsilon\{\mathscr D_{\perp,N}^{\rm lin}
                    -z_{0,N}'\mathscr D_{c,N}^{\rm lin}\},\notag\\
 \mathbb V_p
 &=\varepsilon w_{0,N}'\mathscr D_{c,N}^{\rm lin}
   +\varepsilon\mathfrak d_N(p-\ell_NZ)
   +\varepsilon^2\mathfrak d_N\mathscr D_{c,N}^{\rm lin}.    \label{eq:fixed-phase-delay-perturbation}
\end{align}
If only the current part is retained, then
\cref{eq:fixed-phase-C0-spaces,eq:fixed-phase-coefficient-bounds} imply
\begin{equation}
 \|\mathbb G_\sigma^0\mathbb V_{\rm cur}(Z,p)\|_{
       \mathsf X_{\sigma,0}}
 \le C\left\{\frac{\varepsilon}{\rho_\delta}
                 +\frac{\varepsilon^2}{\rho_\delta^2}\right\}
       \|(Z,p)\|_{\mathsf X_{\sigma,0}}.          \label{eq:fixed-phase-current-feedback-proof}
\end{equation}
Indeed, \(p-\ell_NZ=O(|r|\|(Z,p)\|_{\mathsf X})\); the two scalar
delay sources are therefore \(O(\varepsilon|r|)\) and
\(O(\varepsilon^2)\), respectively.  Division by the
\(|r|^2\)-weight in the source norm gives the two displayed ratios.  The
transverse delay source is smaller and is absorbed by the same bound.
Likewise,
\begin{equation}
 \|\mathbb G_\sigma^0\mathbb V_{\rm old}\gamma\|_{
       \mathsf X_{\sigma,0}}
 \le Ce^{2\kappa_e\theta_m}
       \left\{\frac{\varepsilon}{\rho_\delta}
                 +\frac{\varepsilon^2}{\rho_\delta^2}\right\}
       \|\gamma\|_{N,e,\delta}^{\sharp}.         \label{eq:fixed-phase-old-feedback-proof}
\end{equation}
The cap estimate \cref{eq:frozen-cap-bounds},
\(\rho_\delta=\delta S_\delta/(2\alpha)\), and
\(0<\vartheta<1/4\) turn these bounds into
\begin{equation}
 C\left\{\frac{\delta}{S_\delta}
           +\frac{\delta^2}{S_\delta^2}\right\}
 \quad\hbox{and}\quad
 C\left\{\frac{\delta^{1-2\vartheta}}{S_\delta}
           +\frac{\delta^{2-2\vartheta}}{S_\delta^2}\right\},              \label{eq:fixed-phase-feedback-smallness}
\end{equation}
respectively.  Both tend to zero, proving
\cref{eq:fixed-phase-C0-Neumann-number}.

For a fixed old history, the solution equation is
\begin{equation}
 x=\mathbb B_\sigma^0\mathbf b_\sigma+\mathbb G_\sigma^0(F,H)
   +\mathbb G_\sigma^0\mathbb V_{\rm old}\gamma
   +\mathbb G_\sigma^0\mathbb V_{\rm cur}x,\qquad x=(Z,p).  \label{eq:fixed-phase-Neumann-equation}
\end{equation}
Here \(\mathbf b_\sigma=(\gamma(0),b_\sigma)\in E_N\times\R\).
The current feedback has norm below one half for small \(\delta\), uniformly
in \(N\).  Neumann inversion, followed by
\cref{eq:fixed-phase-principal-C0-bound,%
eq:fixed-phase-old-feedback-proof}, proves existence, uniqueness, and
\cref{eq:fixed-phase-C0-inverse}.  On the attracting branch, standard
method of steps identifies the fixed point with the unique mild initial-value
solution.  On the repelling branch, method of steps handles only the causal
delayed \(Z\)-evaluations in both rows; existence and uniqueness of the mixed
terminal problem come from the oriented Green--Neumann equation, not from a
backward RFDE flow.  In both cases the solution is classical away from the
initial seam.

It remains to handle the repelling collar feedback.  Current evaluation is
bounded in \(\mathsf X_{r,0}\), and
\cref{eq:fixed-phase-collar-feedback} gives
\begin{equation*}
 \|\mathfrak C_ep(0)\|_{N,e,\delta}^{\sharp}
 \le C_\Gamma\|(Z,p)\|_{\mathsf X_{r,0}}.
\end{equation*}
Because \((\mathfrak C_ec)(0)=0\) for every \(c\), this term changes no finite transverse
trace.  Its contribution to \cref{eq:fixed-phase-Neumann-equation} is the
additional operator
\begin{equation*}
 \mathbb G_r^0\mathbb V_{\rm old}\mathfrak C_e
       \operatorname{ev}_{p,0},
\end{equation*}
whose norm is at most \(C_\Gamma\beta_{\rm col}=o(1)\) by
\cref{eq:fixed-phase-old-feedback-proof}.  A second Neumann inversion proves
the feedback assertion with the same two boundary rows.

Finally, fix \(Z_e\) and the core gauge in the differential of
\(\Gamma_e^h\).  Its current voltage is independent of \(p_e\), whereas
\cref{eq:collar-chart-tame-bound} gives exactly the last bound in
\cref{eq:fixed-phase-collar-feedback}.  Thus the abstract feedback is the
one generated by the raw-compatible chart.  Nothing in this argument solves
the deleted phase equation, controls the boundary-layer value of \(Q\), or
produces high-order slow jets, which proves the stated scope boundary.
\end{proof}

most \(Cr_{\rm out}^2D\).  The remaining term costs at most
\(C\varepsilon e^{2\kappa_{e_\sigma}\theta_m}D\), because
\(r_{\rm out}\le1\) and the branch length in the \(r\)-coordinate is
bounded.  The frozen-cap estimate gives
\(e^{2\kappa_{e_\sigma}\theta_m}\le\delta^{-2\vartheta}\).
This proves \cref{eq:normal-phase-H0-coupling}.

Finally apply the inverse estimate
\cref{eq:phase-event-inverse-bound} at \(j=0\) to \((Q,0,0)\).  The result
is \cref{eq:normal-phase-response-bound}.  The proof used no derivative of
\(Q\), so the endpoint boundary layer in
\cref{prop:endpoint-scale-not-slow-speed} causes no loss.  The scope
statements follow because no phase-induced normal or moving-event column
has entered the calculation.
\end{proof}

\begin{proof}[Proof of \cref{lem:reduced-flow-phase-defect}]
Put
\(
 x_{0,N}(r)=r\mathbf1+z_{0,N}(r)
\), so that \(x_{0,N}'=e_{0,N}\).  If
\(r_k(r)=\Phi_{q_0}^{-\delta\theta_k}(r)\), then
\cref{eq:backtrack-basic-identities} gives
\begin{equation}
 r_k'(r)=\frac{q_0(r_k(r))}{q_0(r)}.              \label{eq:phase-defect-backtrack-tangent}
\end{equation}
Differentiating the exact reduced delay row
\(
 \mathcal D_{0,N}(r)
 =K\sum_kL_{k,N}\{x_{0,N}(r)-x_{0,N}(r_k(r))\}
\)
and multiplying by \(q_0(r)\) therefore yields
\begin{align}
 q_0(r)\mathcal D_{0,N}'(r)
 =K\sum_{k=0}^mL_{k,N}\{&q_0(r)e_{0,N}(r)\notag\\
 &-q_0(r_k(r))e_{0,N}(r_k(r))\}.                 \label{eq:phase-defect-D0-tangent}
\end{align}

Along the backward-extended reduced path one has
\(
 \bar r_\sigma(s-\theta_k)=r_k(\bar r_\sigma(s))
\).
The collective-history differential of the first term in
\cref{eq:fold-time-tracker-delay}, in a direction \(\xi\), is
\begin{align*}
 K\sum_{k=0}^mL_{k,N}\{&e_{0,N}(\bar r_\sigma(s))\xi(s)\\
 &-e_{0,N}(\bar r_\sigma(s-\theta_k))\xi(s-\theta_k)\}.
\end{align*}
The subtracted term contributes
\(-\mathcal D_{0,N}'(\bar r_\sigma(s))\xi(s)\).
Substitute
\(
 \xi(s)=q_0(\bar r_\sigma(s))\widehat a(s)
\)
and subtract \cref{eq:phase-defect-D0-tangent}, multiplied by
\(\widehat a(s)\).  Every current evaluation cancels, leaving exactly
\cref{eq:reduced-flow-phase-delay-defect}.  This proves
\cref{eq:exact-phase-delay-cancellation}.  A constant phase makes each
remaining difference zero, including the intact zero-delay summand, and
proves \cref{eq:constant-phase-delay-annihilation}.

It remains to estimate the phase differences.  If
\(s-\theta_k\ge0\), the two corresponding \(r\)-values lie on the same
backtrack segment.  The common-collar geometry and
\cref{eq:backtrack-basic-identities} give
\begin{align*}
 |a(\bar r_\sigma(s))-a(\bar r_\sigma(s-\theta_k))|
 &\le \int_{\bar r_\sigma(s-\theta_k)}^{\bar r_\sigma(s)}
                 |a'(u)|\,|\dd u|\\
 &\le C\frac{\delta\theta_k}{|\bar r_\sigma(s)|}
              \|a\|_{\mathsf A_\phi^1(I_\sigma)}.
\end{align*}
If \(s-\theta_k<0\), the constant extension in
\cref{eq:formal-phase-collar-extension} replaces the delayed value by the
entry value.  Since then \(0\le s<\theta_k\), the same integral estimate
from \(e_{\sigma,0}\) to \(\bar r_\sigma(s)\) gives the identical bound.
The reduced speed, \(e_{0,N}\), and the layer sums are uniformly bounded,
and every delayed \(r\)-value is comparable in modulus with the current
one.  Summing proves \cref{eq:phase-delay-pointwise-bound}.  Projection
does not enlarge the norm.  Thus
\begin{align*}
 \int_{I_\sigma}|\mathscr P_{c,\sigma,N}a(r)|\,\dd r
 &\le C\delta\log\frac{r_{\rm out}}{\rho_\delta}
              \|a\|_{\mathsf A_\phi^1},\\
 \sup_{r\in I_\sigma}|r|\,
       |\mathscr P_{c,\sigma,N}a(r)|
 &\le C\delta\|a\|_{\mathsf A_\phi^1}.
\end{align*}
These are precisely the two terms of the
\(\mathsf H_\phi^0\)-norm and prove
\cref{eq:phase-delay-action-bound}.

We finally identify the full column.  At \(Z=p=0\), the derivatives of
the quadratic terms \(h_{c,N}\) and \(h_{\perp,N}\) vanish.  Terms obtained
by differentiating coefficients multiplying \(Z\), \(p\), or
\(\Delta\mathcal D_N^{\rm ft}=0\) vanish as well.  The algebraic speed
row therefore gives
\[
 \widehat Q_{\rm full}
 =\varepsilon^{-1}(p-\ell_NZ)
   +\mathscr D_{c,N}^{\rm lin}[\gamma,Z]
   +\mathscr P_{c,\sigma,N}a
 =Q_{\rm n}+\mathscr P_{c,\sigma,N}a.
\]
Substitution in \cref{eq:phase-event-a-equation} proves
\cref{eq:full-delayed-phase-row}.  Differentiating the affine \(Z\)-row
\(-R_{z,N}(r)\) in the direction
\(\xi=q_0a\), and adding the phase differential of the delay row, gives
the first line of \cref{eq:exact-phase-normal-source-column}.

For the \(p\)-row, the affine value is
\(
 \varepsilon R_{q,N}=\varepsilon^2q_0\mathfrak d_N
\).
Its derivative along the reduced path is
\(\varepsilon q_0R_{q,N}'a\).  The part of the speed differential not
already present in the fixed-phase block is
\(\mathscr P_{c,\sigma,N}a\); it contributes
\(\varepsilon^2\mathfrak d_N\mathscr P_{c,\sigma,N}a\), while the delay
row itself contributes
\(\varepsilon w_{0,N}'\mathscr P_{c,\sigma,N}a\).  This is the second
line of \cref{eq:exact-phase-normal-source-column}.  Setting \(a=c\) and
using \cref{eq:constant-phase-delay-annihilation} proves
\cref{eq:constant-phase-residual-column}.  The calculation used only the
formal reduced-flow extension, so it makes none of the collar or tracker
identifications excluded in the statement.
\end{proof}

\begin{proof}[Proof of \cref{prop:delayed-phase-event-core}]
Let

 \mathcal P_\sigma(a,\widehat\tau)
   =(\mathscr P_{c,\sigma,N}a,0,0).

Then
\(
 \mathscr B_{\phi,\sigma}^{\rm del}
 =\mathscr B_{\phi,\sigma}-\mathcal P_\sigma
\).
By \cref{eq:phase-delay-action-bound,eq:phase-event-inverse-bound},

 \|\mathscr B_{\phi,\sigma}^{-1}\mathcal P_\sigma\|

\end{equation*}
Because \(\rho_\delta\asymp\delta S_\delta\), the right-hand side tends

 (I-\mathscr B_{\phi,\sigma}^{-1}\mathcal P_\sigma)^{-1}

Its geometric-series estimate is

\begin{proof}[Proof of \cref{prop:event-aligned-normal-traces}]
The chain rule at a moving terminal time gives
\[
 \left.\frac{\dd}{\dd\lambda}\right|_0
 p_\lambda(S_\lambda)
 =\dot p(S)+p_*'(S)\dot S.
\]
Applying the same rule pointwise to the history trace gives
\[
 \left.\frac{\dd}{\dd\lambda}\right|_0
 Z_\lambda(S_\lambda-L+\theta)
 =\dot Z(S-L+\theta)+Z_*'(S-L+\theta)\dot S.
\]
Substitution of
\(\dot S=\delta^{-1}\widehat\tau\) proves
\cref{eq:raw-moving-normal-traces}.  Differentiating the event in
\cref{eq:event-aligned-section-row} gives
\[
 \xi(S)+r_*'(S)\dot S
 =\xi(S)+q_e\widehat\tau=b_{\rm raw}.
\]
Now write
\[
 \dot p(S)
 =n_p(S)+\frac{p_*'(S)}{\delta q_e}\xi(S)
\]
and use the analogous identity for \(\dot Z(S-L+\theta)\).  The two
terms containing \(\xi(S)\) and \(\widehat\tau\) combine into
\[
 \frac{x_*'}{\delta q_e}
       \{\xi(S)+q_e\widehat\tau\}
 =\frac{x_*'}{\delta q_e}b_{\rm raw},
\]
for \(x_*=p_*\) or the translated \(Z_*\)-history.  This proves
\cref{eq:event-aligned-triangular-rows}.  If \(b_{\rm raw}=0\), no
\(\widehat\tau\)-column remains.  At \(Z_*=p_*=0\), both normal time
derivatives in \cref{eq:raw-moving-normal-traces} vanish, proving the
reduced-base statement.

For a fixed regular handoff section, the implicit-function theorem and
\(r_*'(s_h)\ne0\) give
\[
 \dot s_h=-\frac{\xi(s_h)}{r_*'(s_h)}.
\]
The chain rule for
\(Z_\lambda(s_h(\lambda)+\theta)\) and
\(p_\lambda(s_h(\lambda))\) now gives
\cref{eq:fixed-r-handoff-traces}.  The formula contains neither
\(S_\lambda\) nor \(\widehat\tau\).  On the reduced branch, autonomy and
the terminal condition \(r^\#(0)=e_1\) uniquely give
\cref{eq:terminal-relative-reduced-path}; its fixed-\(r\) crossing at
\(e_h=\Phi_{q_0}^{-\delta L_b}(e_1)\) is
\(\varsigma=-L_b\).  Simultaneous translation of the component equations
carries their Green kernels to translated kernels with the same time
differences and identical bounds.  Therefore the fixed reduced-base
component operator has the same norm estimates in these coordinates,
without differentiating a moving trace.  For a general
moving nonlinear family, \(s_h(\lambda)-S_\lambda\) need not remain fixed;
the proof makes no flushing claim for that family.

Finally, direct differentiation of
\cref{eq:moving-trace-strong-norm} gives
\[
 \|\delta^{-1}x'\|_{d,\delta}
 =\max_{0\le j\le d}\delta^{j-1}
       \|\partial_s^{j+1}x\|_\infty
 \le\delta^{-2}\|x\|_{d+1,\delta}.
\]
The loss is sharp.  For a fixed nonzero smooth compactly supported \(f\),
put
\[
 x_\delta(s)=\delta^2f((s-S)/\delta).
\]
Then
\(\|x_\delta\|_{d+1,\delta}=O(\delta^2)\), whereas
\(\|\delta^{-1}x_\delta'\|_{d,\delta}\asymp1\).
Moreover, the derivative of the translation action is its unbounded
generator \(x\mapsto x'\).  A continuous piecewise
\(W^{d,\infty}\) function with a jump in its \(d\)-th derivative supplies
the standard failure of same-space differentiability.  This proves
\cref{eq:moving-trace-one-order-loss} and the asserted regularity boundary.
All remaining exclusions are immediate because only traces, not a solution
map or a tracker, were constructed.
\end{proof}

\begin{proof}[Proof of
\cref{lem:current-absorbed-delay-splitting}]
For \(\theta_k=0\), the corresponding summand in
\cref{eq:fixed-phase-collar-delay} is the intact identity
\[
 L_{k,N}\{\widehat u_{\gamma,Z}(s)
                -\widehat u_{\gamma,Z}(s)\}=0.
\]
Delete it before splitting the delay.  For \(k\in\mathcal I_+\), write its
current part as \(L_{k,N}Z(s)\) and its shifted part as
\(-L_{k,N}\widehat Z(s-\theta_k)\).  Substitution in
\cref{eq:fixed-phase-delay-perturbation} puts every current term in
\(\mathbb A_{\rm cur}\) and every shifted term in \(\mathbb B_k\), with the
signs displayed in \cref{eq:current-absorbed-delay-blocks}.  This proves the
exact identity \cref{eq:current-absorbed-delay-equation}.

It remains to prove the attracting evolution estimate.  Let
\(\mathbb A_0\) denote the principal two-by-two matrix in
\cref{eq:fixed-phase-principal-s-system}.  In the
\(\|\cdot\|_{a,\rm out}\)-norm, the coefficient bounds and
\cref{eq:attracting-outer-buffer-range} give
\begin{align}
 \|\mathbb A_{\rm cur}-\mathbb A_0\|&\le C\varepsilon,\notag\\
 \max_{1\le i\le d}\delta^{-i}
 \|\partial_s^i\mathbb A_{\rm cur}(s)\|&\le C_d.   \label{eq:current-absorbed-coefficient-motion}
\end{align}
To justify the higher derivatives, apply
\cref{prop:physical-backtrack-calculus} to \(q_0\) and to each composition
of \(z_{0,N}\) with its exact backtrack.  The common layer bound and the
principal coefficient estimates through order \(k_d+2\) give, on the fixed
outer annulus,
\begin{equation}
 \max_{0\le j\le d+1}
 \bigl\{|\partial_r^j\mathcal D_{0,c,N}(r)|
              +|\partial_r^jq_0(r)|\bigr\}\le C_d.            \label{eq:current-absorbed-D0-jets}
\end{equation}
Consequently
\(\max_{j\le d}|\partial_r^j\mathfrak d_N|\le C_d\), and the
same calculation applied to
\(\mathscr R_{k,N}^{\rm cur},c_{k,N},b_{k,N}\) gives
\begin{equation}
 \max_{0\le j\le d}\sup_{r_{\rm out}/2\le r\le r_{\rm out}}
 \left\{\|\partial_r^j\mathbb A_{\rm cur}(r)\|
   +\varepsilon^{-1}\sum_{k\in\mathcal I_+}
        \|\partial_r^j\mathbb B_k(r)\|\right\}\le C_d.       \label{eq:current-absorbed-block-jets}
\end{equation}
Since \(\partial_s\bar r_a=\delta q_0(\bar r_a)\), the chain rule now
proves the second line of
\cref{eq:current-absorbed-coefficient-motion}.  The first line includes
\(\varepsilon\mathfrak d_N(p-\ell_NZ)\); the structural layer bound controls
the finite sum without a factor depending on \(N\).  The second line follows
from the displayed block-jet estimate.

The diagonal transverse evolution is forward stable at a fixed rate, and
the scalar coefficient satisfies \(w_{0,N}'\le-c r_{\rm out}\) on the
buffer.  The two off-diagonal maps have the same small block loop as
\cref{eq:principal-green-loop}.  The Volterra block inversion used in
the proof of \cref{prop:principal-tracker-endpoint-green} therefore gives
an evolution \(\mathbb U_{a,0}\) of the principal block with
\(\|\mathbb U_{a,0}(s,u)\|\le Ce^{-c(s-u)/\delta}\).  Duhamel's identity is
\begin{equation*}
 \mathbb U_{a,\rm cur}(s,u)=\mathbb U_{a,0}(s,u)
 +\delta^{-1}\int_u^s\mathbb U_{a,0}(s,v)
       \{\mathbb A_{\rm cur}(v)-\mathbb A_0(v)\}
       \mathbb U_{a,\rm cur}(v,u)\,\dd v.
\end{equation*}
In the exponentially weighted kernel norm the integral operator has norm
at most \(C\varepsilon\), because the \(\delta^{-1}\) prefactor is paired
with an exponential convolution of width \(O(\delta)\).  Neumann inversion
therefore gives
\[
 \|\mathbb U_{a,\rm cur}(s,u)\|
 \le Ce^{-c(s-u)/\delta}.
\]
Differentiate the evolution equations in \(s\) and \(u\).  Each derivative
of an endpoint contributes one factor \(\delta^{-1}\), while
\cref{eq:current-absorbed-coefficient-motion} controls every coefficient
commutator.  More precisely,
\(\partial_s\mathbb U=\delta^{-1}\mathbb A_{\rm cur}(s)\mathbb U\) and
\(\partial_u\mathbb U=-\delta^{-1}\mathbb U\mathbb A_{\rm cur}(u)\);
Leibniz' rule and induction on \(i+j\) prove
\cref{eq:current-absorbed-attracting-evolution}.
\end{proof}

\begin{proof}[Proof of
\cref{prop:attracting-finite-generation-flushing}]
Since \(|q_0|\le q_+\),
\[
 |\bar r_a(s)-r_{\rm out}|\le q_+\delta L_b=o(1)
\]
uniformly for the choice \cref{eq:attracting-flush-buffer-length}.  The same
speed bound gives \(S_a\ge c/\delta\), whereas
\(L_b=O(S_\delta)\).  After decreasing \(\delta_0\), this proves
\(L_b<S_a\) and \cref{eq:attracting-outer-buffer-range}.

We now work in fold time.  Let \(\Theta_+=\{\theta_k:k\in\mathcal I_+\}\)
and fix the finite seam set
\begin{equation*}
 \Sigma_b=\left\{\theta_{i_1}+\cdots+\theta_{i_n}\in[0,L_b]:
        n\ge0,\ \theta_{i_j}\in\Theta_+\right\}.
\end{equation*}
It is finite because
\(n\min\Theta_+\le L_b\).  Let
\(PC_{\Theta_+}^d([0,L_b];E_N\times\R)\) be the Banach space of continuous
functions whose restrictions to the closures of the components of
\([0,L_b]\setminus\Sigma_b\) are \(W^{d,\infty}\), with one-sided traces at
the seams.  On this space use the strong norm
\begin{equation}
 \|x\|_{\mathfrak S_{a,d}}
 =\max_{0\le i\le d}\delta^i
   \operatorname*{ess\,sup}_{0\le s\le L_b}
   \left\{\|\partial_s^iZ(s)\|_N
             +\frac{|\partial_s^ip(s)|}{r_{\rm out}}\right\}.   \label{eq:attracting-piecewise-strong-norm}
\end{equation}
The displayed expression is a complete norm because the partition is
finite.  Translation by any \(\theta_k\in\Theta_+\) maps its seam set into
\(\Sigma_b\), and convolution with the smooth evolution family maps each
cell into the same piecewise Sobolev class.  Thus the operators below act on
this fixed Banach space.  Define
\begin{align}
 (\mathbb K_ax)(s)
 &=
 \delta^{-1}\sum_{k\in\mathcal I_+}
 \int_{\theta_k}^{s}
   \mathbb U_{a,\rm cur}(s,u)\mathbb B_k(u)
                 Z(u-\theta_k)\,\dd u,            \label{eq:attracting-generation-operator}\\
 x_{\rm col}(s)
 &=
 \mathbb U_{a,\rm cur}(s,0)\binom{\gamma(0)}{b_p}\notag\\
 &\quad+
 \delta^{-1}\sum_{k\in\mathcal I_+}
 \int_0^{\min\{s,\theta_k\}}
   \mathbb U_{a,\rm cur}(s,u)\mathbb B_k(u)
                 \gamma(u-\theta_k)\,\dd u.       \label{eq:attracting-direct-collar-term}
\end{align}
An integral whose lower endpoint exceeds its upper endpoint is zero.
Variation of constants gives the exact equation
\begin{equation}
 x=x_{\rm col}+\mathbb K_ax.                      \label{eq:attracting-generation-fixed-point}
\end{equation}

The fixed-support layer-sum bound gives the dimension-uniform estimate
\begin{equation*}
 \sum_{k\in\mathcal I_+}
 \|\mathbb B_k(s)\|_{\mathcal L(\R^N,
          (E_N\times\R,\|\cdot\|_{a,\rm out}))}
 \le C\varepsilon.
\end{equation*}
Differentiating
\cref{eq:attracting-generation-operator} on each cell, and using
\cref{eq:current-absorbed-attracting-evolution}, gives
\begin{equation}
 \|\mathbb K_a\|_{\mathcal L(\mathfrak S_{a,d})}
 \le C_d\varepsilon
 \le C_d\delta^{2-2\vartheta}=:\beta_{a,\delta}.   \label{eq:attracting-generation-number}
\end{equation}
For the old-history term, evaluation behind the seam costs at most
\(e^{2\kappa_{r_{\rm out}}\theta_m}\le\delta^{-2\vartheta}\), but it is
multiplied by \(\varepsilon\).  The endpoint term has no such loss.  Hence
\begin{equation}
 \|x_{\rm col}\|_{\mathfrak S_{a,d}}
 \le C_d\left\{\|\gamma\|_{\mathrm{cap},h,r_{\rm out}}
                 +\frac{|b_p|}{r_{\rm out}}\right\}.           \label{eq:attracting-direct-collar-bound}
\end{equation}
For completeness, the derivative estimate follows cell by cell from
Leibniz' rule.  At differentiation order \(j+1\), an endpoint term contains
\(\delta^{-1}\partial_s^j\{\mathbb B_k(s)Z(s-\theta_k)\}\), hence only
derivatives of \(Z\) through order \(j\).  The factor
\(O(\varepsilon/\delta)\) is absorbed by the next \(\delta\)-weight, while
derivatives of the kernel are controlled by
\cref{eq:current-absorbed-attracting-evolution}.  Induction through order
\(d\) proves the same bound on every component of the fixed partition, with
no dependence on the number of seams.  Neumann inversion of
\cref{eq:attracting-generation-fixed-point} therefore gives the same bound
for \(x\).

Expand the inverse only \(M\) generations:
\begin{equation}
 x=\sum_{n=0}^{M-1}\mathbb K_a^nx_{\rm col}
                  +\mathbb K_a^Mx.                \label{eq:attracting-finite-generation-expansion}
\end{equation}
Consider \(s\in[L_b-\theta_m,L_b]\).  A word with \(n\le M-1\) shifted
edges contains current evolution intervals whose total length is at least
\(s-n\theta_m\) for the endpoint term and at least
\(s-(n+1)\theta_m\) for an old-history insertion.  By
\cref{eq:attracting-flush-buffer-length}, the latter, smaller quantity is
at least \(L_0+1\).  Multiplying the evolution estimates along the word
therefore gives \(e^{-cL_0/\delta}\).  More explicitly, an \(n\)-edge term
is an integral over an ordered simplex in variables \(u_1,\ldots,u_n\);
the shifted constraints are
\(u_{j+1}\le u_j-\theta_{i_j}\).  Summing the lengths of the intervening
current-evolution factors telescopes exactly to
\(s-v-\sum_{j=1}^n\theta_{i_j}\), where \(v=0\) for an endpoint word and
\(0\le v\le\theta_m\) for an old-history insertion.  This gives the two
lower bounds just displayed.
Differentiating the simplex integral creates either kernel derivatives or
boundary faces.  The former are canceled by the strong \(\delta\)-weights,
and a boundary face merely removes one convolution without changing the
telescoping lower bound.  Since \(n\le M-1\) and \(d\) are fixed, the
remaining convolution polynomials are absorbed by decreasing \(c\).
Consequently,
\begin{align}
 \left\|\sum_{n=0}^{M-1}\mathbb K_a^nx_{\rm col}
       \right\|_{\mathfrak S_{a,d};
             [L_b-\theta_m,L_b]}
 &\le C_de^{-cL_0/\delta}\mathcal D_a,\notag\\
 \|\mathbb K_a^Mx\|_{\mathfrak S_{a,d}}
 &\le C_d\delta^{M(2-2\vartheta)}\mathcal D_a,     \label{eq:attracting-word-tail-bounds}\\
 \mathcal D_a
 &=\|\gamma\|_{\mathrm{cap},h,r_{\rm out}}
                 +\frac{|b_p|}{r_{\rm out}}.\notag
\end{align}
Condition \cref{eq:attracting-flush-generation-count} implies \(M>d\).
Here the required smoothing can be seen without invoking an unstated
theorem.  Continuity of \(\gamma\) and \(Z(0)=\gamma(0)\) first gives a
classical \(C^1\) solution for \(s>0\).  Inductively, if the solution is
\(C^j\) after \((j-1)\theta_m\), differentiating the retarded equation on
\([j\theta_m,L_b]\) reads only values from the preceding \(C^j\) region and
gives \(C^{j+1}\) regularity there.  Since
\(L_b-\theta_m>M\theta_m>d\theta_m\), the whole exit interval is genuinely
\(C^d\), even when the initial seam was not compatible through order \(d\).

On the exit interval, \(\partial_s\bar r_a=\delta q_0(\bar r_a)\),
\(|q_0|\asymp1\), and \(|r|\asymp r_{\rm out}\).  Repeated chain rule gives
\begin{equation}
 \max_{0\le j\le d}\sup_{J_b}|r|^j|\partial_r^jf|
 \le C_d\delta^{-2d}
   \max_{0\le i\le d}\delta^i
       \sup_{L_b-\theta_m\le s\le L_b}|\partial_s^if|.         \label{eq:attracting-strong-to-slow}
\end{equation}
The corresponding estimate for \(p\) has the harmless outer factor
\(|r|^{-1}\) already present in the
\(\mathsf P_\varepsilon^d\)-norm.

It remains to recover \(Q\) without losing another derivative.  With
\(H=0\), the scalar row and the algebraic row in
\cref{eq:fixed-phase-normal-block} combine exactly to
\[
 \delta p_s=\varepsilon
       (w_{0,N}'+\varepsilon\mathfrak d_N)Q.
\]
After reparameterization,
\begin{equation}
 Q_b(r)=\frac{q_0(r)p_b'(r)}
              {w_{0,N}'(r)+\varepsilon\mathfrak d_N(r)}.       \label{eq:attracting-Q-from-p}
\end{equation}
The jet bound above and \(|w_{0,N}'|\asymp r_{\rm out}\) give explicitly
\begin{equation*}
 \max_{0\le j\le d-1}
 \sup_{J_b}r_{\rm out}^{j+1}
 \left|\partial_r^j
   \{w_{0,N}'+\varepsilon\mathfrak d_N\}^{-1}\right|\le C_d.
\end{equation*}
Hence the
\(\mathsf P_\varepsilon^d\)-bound for \(p_b\) controls
\(Q_b\) in \(\mathsf Q_\varepsilon^{d-1}\).
Applying \cref{eq:attracting-strong-to-slow} to
\cref{eq:attracting-word-tail-bounds}, and inserting the common
\(\varepsilon^{-1}\) normalization, proves
\cref{eq:attracting-finite-generation-flush}.

Finally, \(S_\delta/\delta\to\infty\), so
\(\delta^{-2d}e^{-cA_dS_\delta/\delta}\) is smaller than every algebraic
power.  Moreover,
\[
 M(2-2\vartheta)-2d
 >2M(1-2\vartheta)-2d>0,
\]
which proves the last assertion.
\end{proof}

\begin{proof}[Proof of
\cref{lem:repelling-component-generations}]
The exact algebra comes first.  From
\cref{eq:current-absorbed-delay-rows},
\[
 b_{k,N}=\varepsilon
    (w_{0,N}'+\varepsilon\mathfrak d_N)c_{k,N}
 =\varepsilon a_pc_{k,N}.
\]
Therefore the coefficient of \(Z\) in the second row of
\(\mathbb A_{\rm cur}\) is
\[
 -w_{0,N}'\ell_N-\varepsilon\mathfrak d_N\ell_N
       +\sum_{k\in\mathcal I_+}b_{k,N}
 =-a_p\left(\ell_N-\varepsilon
                 \sum_{k\in\mathcal I_+}c_{k,N}\right).
\]
The first row is split in the same way.  This proves
\cref{eq:repelling-component-equations}, including all four signs.

We next supply the weighted jets on the entire repelling branch; the
fixed-annulus estimate \cref{eq:current-absorbed-block-jets} alone is not
enough near \(-\rho_\delta\).  Write
\[
 m_N(r)=\frac{w_{0,N}'(r)}r,
\]
using the removable value at \(r=0\).  The critical-curve jets give
\[
 0<c_m\le |m_N(r)|\le C_m,\qquad
 \max_{j\le d+2}\bigl\{|\partial_r^jm_N(r)|
             +|\partial_r^j(m_N^{-1})(r)|\bigr\}\le C_d.
\]
For the shared-resource reduced speed,
\begin{equation}
 q_0(r)=\frac{r-\varepsilon\nu}{w_{0,N}'(r)}
       =\left(1-\frac{\varepsilon\nu}{r}\right)m_N(r)^{-1}.
                                                               \label{eq:repelling-q0-quotient}
\end{equation}
Consequently, for \(1\le n\le d+1\),
\begin{equation}
 |r|^{n-1}|\partial_r^nq_0(r)|
 \le C_d\left(1+\frac{\varepsilon}{|r|^2}\right)
 \le C_d,                                            \label{eq:repelling-q0-improved-jets}
\end{equation}
because
\(\varepsilon/\rho_\delta^2=4\alpha^2/S_\delta^2\).
The gain of one power of \(|r|\) here is essential; the ordinary weighted
\(q_0\)-bound would not control the jets of \(\mathfrak d_N\).

Put
\[
 f_N(r)=r\mathbf1+z_{0,N}(r),\qquad
 r_k(r)=\Phi_{q_0}^{-\delta\theta_k}(r).
\]
At each retained \(r\), choose a same-sign local annulus on which
\(|u|\asymp |r|\) and which contains the complete \(q_0\)-backtrack tube.
On this annulus
\(\delta\theta_m/|r|\le C/S_\delta\) and
\(\|f_N\|_{d+2,|r|}\le C_d|r|\).  The composition-difference part of
\cref{prop:physical-backtrack-calculus}, applied on that annulus, therefore
gives
\begin{equation}
 |r|^n\left\|
 \partial_r^n\{f_N(r)-f_N(r_k(r))\}\right\|_N
 \le C_d\delta,\qquad 0\le n\le d+1.              \label{eq:repelling-backtrack-difference-jets}
\end{equation}
The common layer-sum bound and the definition of
\(\mathcal D_{0,c,N}\) imply
\begin{equation}
 |r|^n|\partial_r^n\mathcal D_{0,c,N}(r)|
 \le C_d\delta,\qquad 0\le n\le d+1.              \label{eq:repelling-D0-weighted-jets}
\end{equation}
Since \(\mathfrak d_N=(\mathcal D_{0,c,N}-q_0)'\),
\cref{eq:repelling-q0-improved-jets,eq:repelling-D0-weighted-jets}
give, for \(0\le j\le d\),
\[
 |r|^j|\partial_r^j\mathfrak d_N|
 \le C_d\left\{\frac{\delta}{|r|}
          +1+\frac{\varepsilon}{|r|^2}\right\}\le C_d.
\]
This is \cref{eq:repelling-d-weighted-jets}.

The principal coefficient estimates give
\[
 |r|^j\|\partial_r^j(\mathcal C_N-A_N)\|\le C_d|r|,
 \qquad
 |r|^j\bigl\{\|\partial_r^jz_{0,N}'\|
       +\|\partial_r^j\ell_N\|
       +|\partial_r^jw_{0,N}'|\bigr\}\le C_d|r|.
\]
Together with \(\varepsilon\le C|r|\), the coefficient definitions and
Leibniz' rule prove \cref{eq:repelling-weighted-block-jets}.  The sign of
\(w_{0,N}'\) on the negative branch and
\(\varepsilon/|r|\le C\delta/S_\delta=o(1)\) prove both sides of
\cref{eq:repelling-component-sizes}.  Weighted differentiation of
\(a_p^{-1}\), followed by multiplication with the weighted \(q_0\)-jets,
proves \cref{eq:repelling-inverse-coefficient-jets}.

Along \(\bar r_r\), repeated differentiation of
\(\partial_s\bar r_r=\delta q_0(\bar r_r)\) now yields, for
\(0\le m\le d\),
\begin{align}
 &\|\partial_s^m(\mathbb A_Z-A_N)\|
 +|\bar r_r|^{-1}\bigl\{|\partial_s^ma_p|
                  +\|\partial_s^mC_{Zp}\|\bigr\}\notag\\
 &\quad
 +|\bar r_r|^{-2}\|\partial_s^mC_{pZ}\|
 +\varepsilon^{-1}\sum_k\|\partial_s^mB_k^Z\|
 +(\varepsilon|\bar r_r|)^{-1}
       \sum_k\|\partial_s^mB_k^p\|\le C_d.          \label{eq:repelling-fold-time-block-jets}
\end{align}

The transverse diagonal block is a bounded \(O(\varepsilon)\) perturbation
of the uniformly forward-stable \(\mathcal C_N\)-evolution.  Duhamel's
identity, exactly as in the attracting proof, gives the first estimate in
\cref{eq:repelling-component-kernel-bounds}.  The scalar formula in
\cref{eq:repelling-component-kernels} is its exact future-stable kernel.
Differentiating the two endpoint equations for \(U_Z\), and differentiating
the exponential formula for \(u_p\), proves all mixed endpoint derivative
bounds.  The weighted coefficient control in
\cref{eq:repelling-fold-time-block-jets} absorbs the commutators after the
factor \(\delta^{i+j}\) is inserted.  Notice that this construction reverses
only one scalar ODE; it never evolves an RFDE backward.

Variation of constants now gives
\cref{eq:repelling-component-LP}.  The sign in its second row is worth
checking: if \(\delta p_s=a_pp+f_p\) and \(p(S_r)=0\), then
\(p=-\mathcal G_p^+f_p\).  Since the current coupling there is
\(-C_{pZ}Z\), it contributes \(+\mathcal K_{pZ}Z\).

We next close the current loop before introducing any delay generation.
In the norm \cref{eq:repelling-component-strong-space}, the source
\(C_{Zp}p\) is \(O(|\bar r_r|^2)\) times the normalized state.  The fast
transverse Green denominator is bounded below independently of \(r\), so
\begin{equation}
 \|\mathcal K_{Zp}\|\le Cr_{\rm out}^2.             \label{eq:repelling-KZp-proof}
\end{equation}
For the reverse edge, put \(R(s)=|\bar r_r(s)|\).  Since
\(R_s=-\delta q_0(\bar r_r)\), \(a_p\asymp R\), and
\(\partial_u\Psi_p(s,u)=a_p(u)/\delta\), one has
\begin{equation}
 \frac{\dd R(u)^2}{\dd\Psi_p(s,u)}
 =-\frac{2\delta^2q_0(\bar r_r(u))R(u)}{a_p(u)}
 \le C\delta^2.                                     \label{eq:repelling-radius-action-motion}
\end{equation}
Thus, with \(\xi=\Psi_p(s,u)\),
\[
 R(u)\le\{R(s)^2+C\delta^2\xi\}^{1/2}.
\]
Changing variables from \(u\) to \(\xi\), and using
\(\dd u=\delta a_p(u)^{-1}\dd\xi\), gives the explicit bound
\begin{equation}
 \frac1{R(s)}\delta^{-1}
 \int_s^{S_r}e^{-c\Psi_p(s,u)}
       R(u)^2\,\dd u
 \le\frac C{R(s)}
       \int_0^\infty e^{-c\xi}
          \{R(s)^2+C\delta^2\xi\}^{1/2}\,\dd\xi
 \le C\left(1+\frac{\delta}{R(s)}\right)\le C.      \label{eq:repelling-KpZ-integral}
\end{equation}
Here \(\delta/R(s)\le C/S_\delta\).  Endpoint differentiation produces
either another integral with a polynomial factor in \(\xi\), controlled by
the same calculation and
\cref{eq:repelling-fold-time-block-jets}, or a lower-limit boundary term.
Each such boundary term contains
\(C_{pZ}(s)/a_p(s)=O(R(s))\), exactly the factor needed by the scalar
normalization.  Induction through order \(d\) therefore proves
\(\|\mathcal K_{pZ}\|\le C_d\), without an
\(r_{\rm out}/R(s)\) loss.  The fixed decrease of
\(r_{\rm out}\) made for \cref{eq:principal-green-loop} can therefore be
retained so that the product norm is below \(1/4\).  Indeed, for component
data \((d_Z,d_p)\), the current problem is
\[
 Z=d_Z-\mathcal K_{Zp}p,\qquad
 p=d_p+\mathcal K_{pZ}Z,
\]
and hence
\begin{equation}
 Z=\sum_{\ell=0}^\infty
   (-\mathcal K_{Zp}\mathcal K_{pZ})^\ell
       (d_Z-\mathcal K_{Zp}d_p),\qquad
 p=d_p+\mathcal K_{pZ}Z.                           \label{eq:repelling-current-loop-series}
\end{equation}
This also proves uniqueness.

The oriented edge weights are proved before any full mixed kernel is formed.
Choose \(\kappa\) below both constants in
\cref{eq:repelling-component-kernel-bounds}.  Then
\[
 W_Z(s,u)\|U_Z(s,u)\|\le C_de^{-c(s-u)/\delta},
 \qquad
 W_p(s,u)|u_p(s,u)|\le C_de^{-c\Psi_p(s,u)}
\]
after decreasing \(c\).  If \([s,u]\) is contained in an outer interval
where \(a_p\ge c_{\rm out}>0\), then
\(\Psi_p(s,u)\ge c_{\rm out}(u-s)/\delta\), which proves the
\(W_{p,{\rm out}}\) version.  These are two-point kernel estimates; the last
one is deliberately not extended across the inner part of the branch.

Expand \cref{eq:repelling-current-loop-series} and use Fubini on each
finite word.  Multiplication of adjacent edge kernels multiplies their
weights, while the coupling at the common vertex has the sizes in
\cref{eq:repelling-component-sizes}.  Use half of each available kernel
exponent as the bookkeeping weight.  The remaining half gives the same
Green denominators, so the weighted loop is still bounded by
\(Cr_{\rm out}^2\).  The fixed choice of \(r_{\rm out}\) is made small
enough that this is one number \(q_{\rm loop}<1\).  Thus a word with
forward edges \((u_j,s_j)\), future scalar edges \((s'_j,u'_j)\), and
\(\ell\) current loops is bounded, relative to its unweighted coefficient
product, by
\begin{equation}
 C_dq_{\rm loop}^{\ell}
 \exp\left\{-c\sum_j\frac{s_j-u_j}{\delta}
             -c\sum_j\Psi_p(s'_j,u'_j)\right\}.     \label{eq:repelling-decorated-word-action}
\end{equation}
Endpoint differentiation either differentiates one kernel or exposes a
boundary face.  The kernel jets and
\cref{eq:repelling-fold-time-block-jets} give the same estimate for every
such word through order \(d\).  Summing
\(\sum_{\ell\ge0}q_{\rm loop}^{\ell}\) proves precisely the local,
oriented assertion following
\cref{eq:repelling-current-loop-bound}; no global weighted operator claim is
used.

We make the function space explicit.  If
\(\mathcal I_+=\varnothing\), the strict generation operator is zero, the
additive seam set is \(\{0\}\), and every conclusion below follows directly
from the current-loop solution.  Hence assume
\(\mathcal I_+\ne\varnothing\).  The positive delays then have a positive
minimum, so only finitely many of their finite sums lie in \([0,S_r]\).
The corresponding continuous piecewise \(W^{d,\infty}\) space is Banach.
A delay shift moves a seam to another member of the same additive set;
\(\mathcal G_Z\) and \(\mathcal G_p^+\) preserve the partition and create
no reflected seam.  Thus every operator in
\cref{eq:repelling-component-generation-recursion} acts on one fixed
space.  Its bounds do not depend on the number of partition cells.

It remains to estimate a strict generation.  For a current path \(y\), the
strict \(Z\)-source has size \(O(\varepsilon)y\).  The strict scalar source
has the exact factor
\[
 B_k^py=-\varepsilon a_pc_{k,N}y.
\]
The future scalar Green division cancels \(a_p\), leaving
\(O(\varepsilon)y\) in \(p\), or
\(O(\varepsilon/|\bar r_r|)y\) in the normalized state.  Delay translation
changes \(|\bar r_r|\) by a uniformly comparable factor because
\(\delta\theta_m/\rho_\delta=O(S_\delta^{-1})\).  After the current loop is
summed,
\begin{equation}
 \|\mathscr T_ry\|_{\mathfrak S_{r,d}}
 \le C_d\frac{\varepsilon}{\rho_\delta}
          \|y\|_{\mathfrak S_{r,d}},               \label{eq:repelling-strict-generation-proof}
\end{equation}
where \(\mathscr T_r\) denotes one complete recursion step in
\cref{eq:repelling-component-generation-recursion}.  Cellwise
differentiation is triangular, as in the attracting proof, and uses no
derivative beyond order \(d\).

The direct old-history source is read only in the first delay window.  Its
cap loss is at most \(\delta^{-2\vartheta}\), but every such read is
multiplied by \(\varepsilon\); hence it is absorbed in the displayed bound
for \(x^{[0]}\).  The endpoint and terminal lifts have the stated size by
the two diagonal kernels and the current-loop estimate.  This proves the
first line of \cref{eq:repelling-generation-number}.

For the edge count, split linearly
\[
 x^{[n]}=x_E^{[n]}+x_U^{[n]}
       +\sum_{k_0\in\mathcal I_+}x_{H,k_0}^{[n]}.
\]
The endpoint root \(E\) has insertion label \(0\), and the terminal root
\(U\) has insertion label \(S_r\).  A history root \(H,k_0\), at integration
label \(v_0\in[0,\theta_{k_0})\), reads
\(\gamma(\sigma_0)\) at
\(\sigma_0=v_0-\theta_{k_0}\).  Thus it contains the direct collar edge
\(\sigma_0\to v_0=\sigma_0+\theta_{k_0}\).  Applying the recursion \(n\)
times adds exactly \(n\) further \(\mathscr S\)-edges in all three cases.
This proves the edge convention stated in the lemma.

Since
\[
 \frac{\varepsilon}{\rho_\delta}
 =\frac{2\alpha\delta}{S_\delta}
 \le C\frac{\delta^{1-2\vartheta}}{S_\delta},
\]
the second line follows.  The recursion therefore converges absolutely.
Its sum satisfies the exact component equations, and the uniqueness in
\cref{prop:fixed-phase-green-collar-buffer} identifies it with the stated
mixed-boundary solution.  By construction, only a strict shifted
\(Z(s-\theta_k)\) of the current path raises the recursion index; the direct
old-history edge, future scalar propagation, and all current loops do not.
\end{proof}

\begin{proof}[Proof of
\cref{prop:repelling-component-flushing}]
The repelling travel time satisfies \(S_r\ge c/\delta\), whereas
\(L_b=O(S_\delta)\).  Thus
\(2L_b+\theta_m<S_r\) for small \(\delta\).  On the inner buffer,
\begin{equation}
 \rho_\delta\le|\bar r_r(s)|
 \le C_{A_d}\rho_\delta,\qquad 0\le s\le L_b,     \label{eq:repelling-inner-buffer-geometry}
\end{equation}
because \(|\partial_s\bar r_r|\asymp\delta\) and
\(\delta L_b=O(\delta S_\delta)=O(\rho_\delta)\).  On
\([S_r-L_b-\theta_m,S_r]\), one has instead
\begin{equation}
 \tfrac12r_{\rm out}\le|\bar r_r(s)|\le r_{\rm out}.          \label{eq:repelling-terminal-buffer-geometry}
\end{equation}
Linearity gives the two asserted components.

We now expose the orientation hidden by the full mixed Green operator.
Expand every same-generation inverse by
\cref{eq:repelling-current-loop-series}.  A resulting decorated component
word contains forward transverse edges of lengths \(F_j\ge0\), future
scalar edges of lengths \(P_j\ge0\), and exactly \(n\) recursive strict
delay edges.  For endpoint and terminal words set
\(t_{\rm in}=0,S_r\), respectively.  For an old-history word use its current
insertion label \(t_{\rm in}=v_0\in[0,\theta_m]\), thereby absorbing the
direct collar edge \(\sigma_0\to v_0\) into the root.  The time labels then
give the exact telescoping identity
\begin{equation}
 t_{\rm out}-t_{\rm in}
 =\sum_{j=1}^n\theta_{k_j}
       +\sum_jF_j-\sum_jP_j.                       \label{eq:repelling-oriented-word-identity}
\end{equation}
Equivalently, if an old-history word is rooted at its true history label
\(\sigma_0\), the right-hand side has the additional direct delay
\(\theta_{k_0}\), and
\[
 t_{\rm out}-\sigma_0
 =\theta_{k_0}+\sum_{j=1}^n\theta_{k_j}
       +\sum_jF_j-\sum_jP_j.
\]
Current loops add one edge of each orientation but no strict delay.  Their
arbitrary multiplicities sum geometrically by
\cref{eq:repelling-current-loop-bound}.  This is the component-word action
lemma; no causal support is attributed to the full mixed Green operator.

Consider first the inner component on
\([L_b-\theta_m,L_b]\).  Its insertions occur at the initial trace or in
the first delay window.  If \(n\le M-1\), then
\begin{equation}
 t_{\rm out}-t_{\rm in}
       -\sum_{j=1}^n\theta_{k_j}\ge L_0.           \label{eq:repelling-inner-word-distance}
\end{equation}
Indeed, the endpoint-rooted lower bound is
\[
 L_b-\theta_m-(M-1)\theta_m
   =L_b-M\theta_m=L_0+\theta_m+1,
\]
whereas the old-history-rooted lower bound, using
\(v_0\le\theta_m\), is
\[
 L_b-\theta_m-\theta_m-(M-1)\theta_m
   =L_b-(M+1)\theta_m=L_0+1.
\]
This is why the direct collar edge is not a generation but is still paid
for in the buffer length.
Hence the total forward transverse length is at least \(L_0\) plus the
total future scalar length.  The component kernel product is therefore
bounded by \(e^{-cL_0/\delta}\).  Keeping the coarser alternative in which
a far-future scalar subword is estimated directly gives the harmless
envelope
\begin{equation}
 e^{-cL_0/\delta}+e^{-cS_\delta L_0},              \label{eq:repelling-inner-low-word-tail}
\end{equation}
because \(a_p/\delta\ge cS_\delta\) throughout the inner buffer.  Boundary
faces created by differentiation are decorated words with fewer
convolutions and the same oriented distance.  The strong \(\delta\)-weights
cancel all kernel derivatives.

For the terms with at least \(M\) strict edges, sum the convergent remainder
using \cref{eq:repelling-generation-number}.  The genuine generation number
is even \(C\delta/S_\delta\); using the stated, larger uniform budget gives
\begin{equation}
 \left\|\sum_{n\ge M}x_{\rm in}^{[n]}\right\|_{\mathfrak S_{r,d}}
 \le C_d\left(\frac{\delta^{1-2\vartheta}}{S_\delta}\right)^M
       \|\gamma\|_{\mathrm{cap},h,-\rho_\delta}.  \label{eq:repelling-inner-generation-tail}
\end{equation}

Turn next to the terminal component.  Put
\[
 s_*=S_r-L_b,\qquad s_o=S_r-L_b-\theta_m,\qquad
 W_T=[s_o,s_*],\qquad O_T=[s_o,S_r].
\]
Thus \(W_T\) is the complete-history handoff window and \(O_T\) is contained
in the outer buffer of
\cref{eq:repelling-terminal-buffer-geometry}.  A decorated terminal word is
rooted at \(t_{\rm in}=S_r\).  If it has \(n<M\) recursive strict edges,
traverse it from the root toward its output in \(W_T\).  If it stays in
\(O_T\), the oriented identity gives
\[
 \sum_jP_j-\sum_jF_j
 =S_r-t_{\rm out}+\sum_{j=1}^n\theta_{k_j}\ge L_b,
\]
so its accumulated outer scalar action is at least \(cL_b/\delta\).  If it
leaves \(O_T\), the first-exit argument below gives the still stronger
bound.  Hence all \(n<M\) words contribute
\[
 C_de^{-cL_0/\delta}
       \frac{|b_u|}{r_{\rm out}}.
\]
Here and below, each complete generation uses the already resummed current
inverse.  Since \(M\) is fixed, all loop factors in a finite root-prefix are
absorbed into \(C_d\); the global generation number likewise already
contains its loop sum.

Now let \(n\ge M\).  In root-to-output order select the first \(M\)
recursive strict edges nearest \(S_r\), together with their complete
generation blocks.  Let \(\mathcal V_M\) contain every time label in this
prefix: both endpoints of every strict edge, all component integration
vertices, and the full intervening kernel intervals of every same-generation
current loop.  This last clause is essential; checking only the strict-edge
endpoints would not localize the loop.  If
\(\mathcal V_M\subset O_T\), each of these \(M\) complete blocks is measured
in the outer normalized state and has the one-edge bound
\begin{equation}
 \beta_{r,\rm out}\le C_d\varepsilon
       \le C_d\delta^{2-2\vartheta}.               \label{eq:repelling-terminal-generation-number}
\end{equation}
Indeed, on \(O_T\),
\(B_k^Z=O(\varepsilon)\) and
\(B_k^p=-\varepsilon a_pc_{k,N}\), while
\[
 \delta^{-1}\int_s^{S_r}u_p(s,u)a_p(u)\,\dd u\le1.
\]
The scalar normalization costs only the fixed factor
\(r_{\rm out}^{-1}\), the layer sum is dimension-uniform, and the restricted
current loops contribute the fixed factor
\((1-q_{\rm loop})^{-1}\).  The weighted coefficient jets give the same
bound through order \(d\).
All remaining recursion blocks are estimated by the global generation
number \(\beta_{r,\delta}\).  Summing delay labels by the layer-sum bound and
loop multiplicities by the current-loop series gives the explicit stay
estimate
\begin{equation}
 \sum_{\mathscr W:\,{\rm stay}}\|\mathscr W\|
 \le C_d\beta_{r,\rm out}^M
       \left(\sum_{j\ge0}\beta_{r,\delta}^j\right)
       \frac{|b_u|}{r_{\rm out}}.                  \label{eq:repelling-terminal-stay-sum}
\end{equation}

Suppose instead that \(\mathcal V_M\not\subset O_T\).  Follow the selected
prefix from \(S_r\) and stop at its first hit or crossing of the left
boundary \(s_o\).  Forward \(Z\)-edges and strict delay edges move time
labels to the right; only a future scalar edge can make this first leftward
crossing.  Split that scalar kernel at \(s_o\), using its semigroup identity.
All component labels before the split lie in \(O_T\).  Applied to this
root-prefix, \cref{eq:repelling-oriented-word-identity} gives the
crossing-number identity
\begin{equation}
 \sum_{\rm pre}P_j-\sum_{\rm pre}F_j
 =S_r-s_o+\sum_{\rm pre}\theta_{k_j}
 \ge L_b+\theta_m.                                 \label{eq:repelling-terminal-first-exit-action}
\end{equation}
Since every pre-exit scalar segment lies in \(O_T\),
\begin{equation*}
 \prod_{\rm pre}|u_p|
 \le C_d\exp\left\{-\frac{c_{\rm out}}{\delta}
                   \sum_{\rm pre}P_j\right\}
 \le C_de^{-c(L_b+\theta_m)/\delta}
 \le C_de^{-cL_0/\delta}.
\end{equation*}
The suffix after the split may leave or re-enter
\(O_T\); it cannot remove an action factor already extracted.  Estimate all
remaining recursion blocks globally.  There are only \(M\) possible
first-exit generation-block classes.  If the marked crossing lies inside a
current loop, its possible location is summed by
\(\sum_{\ell\ge0}(\ell+1)q_{\rm loop}^{\ell}<\infty\) and is absorbed into
\(C_d\).  Therefore
\begin{equation}
 \sum_{\mathscr W:\,{\rm exit}}\|\mathscr W\|
 \le C_de^{-cL_0/\delta}
       \left(\sum_{j\ge0}\beta_{r,\delta}^j\right)
       \frac{|b_u|}{r_{\rm out}}.                  \label{eq:repelling-terminal-exit-sum}
\end{equation}
The remaining geometric series is dimension-uniform and bounded after
decreasing \(\delta_0\).  Combining the short, stay, and first-exit word
classes gives
\begin{equation}
 \left\|x_{\rm u}\right\|_{\mathfrak S_{r,d};
       [S_r-L_b-\theta_m,S_r-L_b]}
 \le C_d\left\{e^{-cL_0/\delta}
            +\delta^{M(2-2\vartheta)}\right\}
       \frac{|b_u|}{r_{\rm out}}.                 \label{eq:repelling-terminal-strong-tail}
\end{equation}
Cellwise differentiation of an iterated integral produces interior kernel
derivatives and boundary faces.  Each face is again a decorated word with
the same root and output labels; if a convolution collapses, its recursive
edge count can only decrease.  Reclassifying the face as short, stay, or
first-exit and using
\cref{eq:repelling-weighted-kernel-bounds} proves
\cref{eq:repelling-terminal-stay-sum,eq:repelling-terminal-exit-sum,%
eq:repelling-terminal-strong-tail} through order \(d\), including every
one-sided seam trace.

The method-of-steps seam argument used on the attracting branch applies to
the sum solving the original retarded equation.  Since \(M>d\) and both
output windows lie after more than \(d\theta_m\) of forward evolution, their
restrictions are genuine \(C^d\) histories, not merely piecewise ones.

It remains to convert the strong estimates to slow norms.  On the inner
window, \cref{eq:repelling-inner-buffer-geometry} and
\(\partial_s\bar r_r=\delta q_0(\bar r_r)\) give
\begin{equation}
 \max_{0\le j\le d}\sup |r|^j|\partial_r^jf|
 \le C_{d,A_d}\left(\frac{S_\delta}{\delta}\right)^d
       \max_{0\le i\le d}\delta^i\sup|\partial_s^if|.        \label{eq:repelling-inner-strong-to-slow}
\end{equation}
On the terminal window the corresponding factor is
\(C_d\delta^{-2d}\).  The factor \(|r|^{-1}\) required for the
\(\mathsf P_\varepsilon^d\)-norm is already present in
\cref{eq:repelling-component-strong-space}.  Applying these conversions to
\cref{eq:repelling-inner-low-word-tail,%
eq:repelling-inner-generation-tail,%
eq:repelling-terminal-strong-tail} gives the \(Z\)- and \(p\)-parts of
\cref{eq:repelling-inner-flush,eq:repelling-terminal-flush}.

Finally, because \(H=0\), the exact recovery identity remains
\begin{equation}
 Q(r)=\frac{q_0(r)p'(r)}
              {w_{0,N}'(r)+\varepsilon\mathfrak d_N(r)}.      \label{eq:repelling-Q-from-p}
\end{equation}
By \cref{eq:repelling-inverse-coefficient-jets}, for \(0\le i\le d-1\),
\[
 |r|^{i+1}|\partial_r^iQ(r)|
 \le C_d\max_{1\le j\le i+1}
          |r|^{j-1}|\partial_r^jp(r)|.
\]
Thus the \(\mathsf P_\varepsilon^d\)-bound gives the
\(\mathsf Q_\varepsilon^{d-1}\)-bound with no additional power of
\(\delta\).
Inserting the common \(\varepsilon^{-1}\) normalization proves the two
displayed estimates.

The first and third terms in \(\mathcal E_{r,\rm in}\) tend to zero by
superalgebraic decay and
\(M(1-2\vartheta)>d\).  Since
\[
 \left(\frac{S_\delta}{\delta}\right)^d
 e^{-cS_\delta L_0}
 =S_\delta^d\delta^{-d}
       e^{-cA_dS_\delta^2},
\]
one fixed sufficiently large \(A_d\) also makes the middle term a positive
power of \(\delta\).  The terminal exponents are positive by the same
inequality used in the attracting proof.  This completes the proof and the
scope statements follow from the construction.
\end{proof}

\begin{proof}[Proof of
\cref{prop:compatible-repelling-collar-closure}]
The strong cap bound in
\cref{eq:strong-compatible-feedback-bound} implies the fading \(C^0\)
bound in \cref{eq:fixed-phase-collar-feedback}.  Therefore
\cref{prop:fixed-phase-green-collar-buffer} gives a unique closed-loop
solution of \cref{eq:strong-compatible-feedback-problem}.  Since the
problem and the feedback are linear, its inverse splits into the two
columns in \cref{eq:compatible-repelling-column-split}.  The two actual
histories constructed below belong to \(C^h\).  Hence the corresponding
open high-order component solutions supplied by
\cref{lem:repelling-component-generations} coincide with these mild
closed-loop columns by the \(C^0\) uniqueness just invoked.

Apply the \(C^0\) inverse separately to those columns.  Current evaluation
in \(\mathsf X_{r,0}\), followed by
\cref{eq:strong-compatible-feedback-bound}, gives
\begin{align*}
 \frac{|p_{\rm c}(0)|}{\rho_\delta}
 &\le C\|\gamma_0\|_{N,-\rho_\delta,\delta}^{\sharp},&
 \frac{|p_{\rm u}(0)|}{\rho_\delta}
 &\le C\frac{|b_u|}{r_{\rm out}}.
\end{align*}
The zeroth term of the strong cap norm dominates the fading norm.  Hence
\begin{align*}
 \|\gamma_{\rm c}\|_{\mathrm{cap},h,-\rho_\delta}
 &\le\|\gamma_0\|_{\mathrm{cap},h,-\rho_\delta}
       +C_\Gamma\frac{|p_{\rm c}(0)|}{\rho_\delta},\\
 \|\gamma_{\rm u}\|_{\mathrm{cap},h,-\rho_\delta}
 &\le C_\Gamma\frac{|p_{\rm u}(0)|}{\rho_\delta},
\end{align*}
which proves \cref{eq:compatible-repelling-history-columns}.  Notice that
\((\mathfrak C_ec)(0)=0\) leaves the transverse boundary row unchanged.

The first closed-loop column is the fixed-history solution with total old
history \(\gamma_{\rm c}\) and zero terminal datum.  Substitution of the
first bound in \cref{eq:compatible-repelling-history-columns} into
\cref{eq:repelling-inner-flush} proves
\cref{eq:compatible-repelling-inner-column}.

For the second column, split once more into the fixed-history response to
\(\gamma_{\rm u}\), with zero terminal datum, and the response to
\((0,b_u)\).  The latter is bounded on \(J_{r,\rm u}\) by
\cref{eq:repelling-terminal-flush}.  We record the cross estimate for the
former.  Use the decorated-word expansion in the proof of
\cref{prop:repelling-component-flushing}.  For an endpoint- or
old-history-rooted word with fewer than \(M\) recursive strict edges and
output in \([S_r-L_b-\theta_m,S_r-L_b]\),
\cref{eq:repelling-oriented-word-identity} gives
\begin{equation*}
 t_{\rm out}-t_{\rm in}
      -\sum_{j=1}^n\theta_{k_j}\ge L_0
\end{equation*}
after decreasing \(\delta_0\): indeed \(t_{\rm in}\le\theta_m\), whereas
\(S_r\ge c/\delta\), \(L_b=O(S_\delta)\), and, more explicitly,
\begin{equation*}
 S_r-L_b-(M+1)\theta_m
   >L_b-M\theta_m=L_0+\theta_m+1
\end{equation*}
for small \(\delta\).  Thus the total forward
transverse length exceeds the total future scalar length by at least
\(L_0\), and the weighted component kernels give
\(C_de^{-cL_0/\delta}\).

All words with at least \(M\) strict edges are bounded globally by
\cref{eq:repelling-generation-number}; after summing the geometric
remainder their strong norm is at most
\begin{equation*}
 C_d\left(\frac{\delta^{1-2\vartheta}}{S_\delta}\right)^M
       \|\gamma_{\rm u}\|_{\mathrm{cap},h,-\rho_\delta}.
\end{equation*}
Kernel differentiation and boundary faces preserve the same root, output,
and oriented-distance alternatives, exactly as in
\cref{eq:repelling-inner-generation-tail}.  The output window is in the
outer annulus, so the strong-to-slow conversion costs
\(C_d\delta^{-2d}\).  Recovering \(Q\) from
\cref{eq:repelling-Q-from-p} costs no further power.  Consequently the
fixed-history cross response is bounded by
\begin{equation*}
 C_{d,A_d}\mathcal E_{r,\rm cross}(\delta)
       \varepsilon^{-1}
       \|\gamma_{\rm u}\|_{\mathrm{cap},h,-\rho_\delta}.
\end{equation*}
Together with \cref{eq:compatible-repelling-history-columns} and the pure
terminal estimate this proves
\cref{eq:compatible-repelling-terminal-column}.  Condition
\cref{eq:compatible-repelling-generation-count} makes the algebraic term in
\(\mathcal E_{r,\rm cross}\) tend to zero; the exponential term is
superalgebraically small.

Finally, hold \(Z_e,\widehat\gamma,\nu,\eta\) fixed in
\cref{prop:compatible-endpoint-jet-collar}.  Its current voltage value is
independent of \(p_e\), so the resulting voltage-history derivative
vanishes at \(\theta=0\).  The tame chart estimate
\cref{eq:collar-chart-tame-bound} gives exactly
\cref{eq:strong-compatible-feedback-bound}.  This identifies the abstract
feedback with the raw-compatible collar column and completes the proof.
\end{proof}

\begin{proof}[Proof of \cref{prop:physical-history-phase-quotient}]
Because the rescaled fold-time RFDE is autonomous, differentiation of the
base solution in fold time gives a solution of the variational equation.  At the
history level this says
\begin{equation}
 \mathcal U_*(s,u)
 \left(\theta\mapsto v_*'(u+\theta),w_*'(u)\right)
 =\left(\theta\mapsto v_*'(s+\theta),w_*'(s)\right).
                                                               \label{eq:phase-quotient-time-tangent}
\end{equation}
Moreover,
\[
 \varrho_N
 \left(\theta\mapsto v_*'(s+\theta),w_*'(s)\right)
 =\pi_N^Tv_*'(s)=r_*'(s).
\]
Division by the nonzero endpoint speed proves both the normalization in
\cref{eq:physical-normalized-tangent} and
\cref{eq:physical-tangent-transport}.  It also shows uniqueness among scalar
multiples of the time tangent.

Since \(\varrho_N\tau_*(s)=1\), the operator \(Q_*(s)\) is a projection with
\begin{equation}
 \operatorname{Ran}Q_*(s)=\ker\varrho_N,
 \qquad
 \ker Q_*(s)=\operatorname{span}\{\tau_*(s)\}.      \label{eq:phase-quotient-range-kernel}
\end{equation}
For \(x\in\ker\varrho_N\), use the variational cocycle identity and
\cref{eq:physical-tangent-transport} to compute
\begin{align*}
 &Q_*(s)\mathcal U_*(s,t)Q_*(t)\mathcal U_*(t,u)x\\
 &\quad=Q_*(s)\mathcal U_*(s,u)x
 -Q_*(s)\mathcal U_*(s,t)\tau_*(t)
       \varrho_N\mathcal U_*(t,u)x\\
 &\quad=Q_*(s)\mathcal U_*(s,u)x,
\end{align*}
because the second term is a scalar multiple of \(Q_*(s)\tau_*(s)=0\).
This is \cref{eq:physical-quotient-cocycle}.  The range--kernel identities
identify the process with the quotient process, and the triangle inequality
gives \cref{eq:physical-quotient-bound}.  Finally, normalizing any nonzero
multiple of the time tangent by \(\varrho_N\tau=1\) recovers exactly
\cref{eq:physical-normalized-tangent}; hence the construction contains no
choice of tangent scale or frozen eigendirection.
\end{proof}

\begin{proof}[Proof of \cref{prop:exact-resource-gauge-quotient}]
The history derivative in
\cref{eq:exact-outer-history-tangent} gives
\(E_s^c=\partial_r\mathcal H_{V,w,q}(r_s)\).  Since the autonomous
time tangent is \(\delta q(r_s)E_s^c\),
\cref{eq:phase-quotient-time-tangent} yields
\[
 \mathcal U_*(s,\sigma)\{\delta q(r_\sigma)E_\sigma^c\}
 =\delta q(r_s)E_s^c,
\]
which proves \cref{eq:resource-gauge-center-cocycle}.

For reference, the full fold-time variational equation along the tracker is
\begin{align}
 \xi'(s)={}&\delta^{-1}\mathcal J_*(s)\xi(s)
 -\delta^{-1}\omega(s)\mathbf1\notag\\
 &+\delta K\sum_{k=0}^mL_{k,N}(\eta)
       \{\xi(s)-\xi(s-\theta_k)\},\notag\\
 \omega'(s)={}&\delta\pi_N^T\xi(s).                 \label{eq:resource-gauge-full-variation}
\end{align}
At the current endpoint,
\[
 \xi(s)=u(s)+V'(r_s)a(s),\qquad
 \omega(s)=w'(r_s)a(s).
\]
Differentiate \(a=\omega/w'(r_s)\), use \(r_s'=\delta q(r_s)\), and
differentiate the exact resource identity
\[
 q(r)w'(r)=r-\delta^2\nu
 \quad\Longrightarrow\quad
 q'(r)w'(r)+q(r)w''(r)=1=\pi_N^TV'(r).
\]
The full variation equation then gives
\begin{align*}
 a'
 &=\frac{\delta\pi_N^T\{u+V'a\}}{w'}
   -a\frac{\delta q w''}{w'}\\
 &=\frac{\delta}{w'}\pi_N^Tu+\delta q'a,
\end{align*}
which is \cref{eq:resource-gauge-a}.

For \(-\theta_m\le\theta<0\), histories translate, so
\[
 (\partial_s-\partial_\theta)\xi(s+\theta)=0.
\]
The definition of \(e_s\), or direct differentiation of
\(\delta q(r_s)e_s(\theta)=v_*'(s+\theta)\), gives
\begin{equation}
 (\partial_s-\partial_\theta)e_s(\theta)
 =-\delta q'(r_s)e_s(\theta).                       \label{eq:resource-gauge-e-transport}
\end{equation}
Apply \(\partial_s-\partial_\theta\) to
\(\psi_s=\xi_s-e_sa\) and use
\cref{eq:resource-gauge-a,eq:resource-gauge-e-transport}; the two
\(\delta q'a\) terms cancel and give
\cref{eq:resource-gauge-transport}.

The normalized center history is not itself a solution but obeys the
transport law \cref{eq:resource-gauge-center-cocycle}.  Equivalently, the
full variational operator applied to \(E_s^c\) equals
\(\partial_sE_s^c+\delta q'(r_s)E_s^c\).  Substitute
\[
 (\xi_s,\omega)
 =(\psi_s,0)+aE_s^c
\]
in \cref{eq:resource-gauge-full-variation}, subtract the derivative of
\(aE_s^c\), and use \cref{eq:resource-gauge-a}.  All terms proportional to
\(a\), including the delayed tangent terms, cancel.  The voltage endpoint
of the remaining rank-one term is
\[
 -V'(r_s)\frac{\delta}{w'(r_s)}\pi_N^Tu(s)
 =-\delta H_s\pi_N^Tu(s).
\]
This proves \cref{eq:resource-gauge-boundary}.  It also proves directly that
\(\psi_s\) is unchanged by adding a multiple of \(E_s^c\).

We next solve the nonstandard transport equation along characteristics.
For fixed \(s,\theta\), set
\[
 F(\sigma)=\psi_{s+\sigma}(\theta-\sigma),
 \qquad \theta\le\sigma\le0.
\]
Then \(F(\theta)=u(s+\theta)\), \(F(0)=\psi_s(\theta)\), and
\cref{eq:resource-gauge-transport} gives
\[
 F'(\sigma)
 =-\delta h_{s+\sigma}(\theta-\sigma)
       \pi_N^Tu(s+\sigma).
\]
Integration proves
\[
 \psi_s(\theta)=u_s(\theta)
 -\delta\int_\theta^0h_{s+\sigma}(\theta-\sigma)
        \pi_N^Tu_s(\sigma)\,\dd\sigma,
\]
which is \cref{eq:resource-gauge-history-conjugacy}.  In particular,
\[
 \psi_s(-\theta_k)
 =u(s-\theta_k)-(\mathcal K_su_s)(-\theta_k).
\]
Substitution in \cref{eq:resource-gauge-boundary} proves
\cref{eq:resource-gauge-standard-rfde}; the Volterra term therefore has the
displayed positive sign inside the delay bracket.  This derivation is along
characteristics on the delay-enlarged time interval, so no derivative of a
formally time-dependent conjugacy has been discarded.

It remains to prove the uniform bounds.  Under
\cref{eq:resource-gauge-collar-bounds}, the ratios of \(q\), the vector
\(V'\), and hence \(e_s\) are bounded by common constants, while
\[
 \|h_s(\theta)\|_N\le\frac{C}{\rho_\delta}.
\]
If \(\theta\le\sigma\le0\), then
\(e^{2\kappa_*(\theta-\sigma)}\le1\).  Therefore
\begin{align*}
 e^{2\kappa_*\theta}\|(\mathcal K_s\phi)(\theta)\|_N
 &\le\frac{C\delta}{\rho_\delta}
   \int_\theta^0
      e^{2\kappa_*(\theta-\sigma)}
      e^{2\kappa_*\sigma}\|\phi(\sigma)\|_N\,\dd\sigma\\
 &\le C\frac{\delta\theta_m}{\rho_\delta}
       \|\phi\|^\sharp_{\kappa_*}.
\end{align*}
The definition of \(\rho_\delta\) proves
\cref{eq:resource-gauge-volterra-bound}.  For small \(\delta\), the bound is
at most \(1/2\), and the Neumann series proves
\cref{eq:resource-gauge-volterra-inverse}.

Conversely, let \(u\) solve
\cref{eq:resource-gauge-standard-rfde} and define
\(\psi_s=(\Id-\mathcal K_s)u_s\).  Reversing the characteristic calculation
shows that \(\psi\) satisfies
\cref{eq:resource-gauge-transport}; reversing the endpoint substitution
shows that it satisfies \cref{eq:resource-gauge-boundary}.  Solve the scalar
linear equation \cref{eq:resource-gauge-a} from any prescribed value of
\(a\), and reconstruct
\[
 (\xi_s,\omega(s))=(\psi_s,0)+a(s)E_s^c.
\]
The cancellations already proved above, now read in reverse, show that this
pair solves the full variational equation.  Thus
\cref{eq:resource-gauge-history-conjugacy,eq:resource-gauge-history-inverse}
give the asserted two-sided quotient coordinate equivalence.

Finally, \(|\omega|/|w'(r_s)|=|a|\) and
\(\xi_s=\psi_s+e_sa\), with \(\|e_s\|^\sharp_{\kappa_*}\le C\).
The two triangle inequalities give the claimed uniform equivalence between
\cref{eq:resource-gauge-anisotropic-norm} and
\(\|\psi_s\|^\sharp_{\kappa_*}+|a|\).
\end{proof}

\subsection{Why unnormalized backward convergence does not select a history}

The next elementary RFDE example makes precise why an outer-history theorem
needs a normalized boundary coordinate.  It rules out a logical inference;
it remains compatible with a separately imposed global anchor for the
particular network field of the main article.

\begin{proposition}[Unnormalized backward convergence and loss of jet control]
\label[proposition]{prop:backward-asymptotic-nonselection}
Fix \(0<\delta_*<1\), let
\(\tau:(0,\delta_*)\to(0,\infty)\) be arbitrary, and write
\(\tau_\delta=\tau(\delta)\).  On \(\R^2\) use
\(|(y,z)|_1=|y|+|z|\), and put
\begin{equation*}
 X_\delta=C([-\tau_\delta,0];\R^2),\qquad
 X^1_\delta=C^1([-\tau_\delta,0];\R^2),
\end{equation*}
with
\begin{equation*}
 \|\phi\|_{X_\delta}=\sup_\vartheta|\phi(\vartheta)|_1,
 \qquad
 \|\phi\|_{X^1_\delta}=\|\phi\|_{X_\delta}
       +\sup_\vartheta|\phi'(\vartheta)|_1.
\end{equation*}
Consider the RFDE
\begin{equation}
 \dot x(t)=\mathcal F_\delta(x_t),
 \qquad \mathcal F_\delta(\phi)=\delta^{-2}\phi(0),           \label{eq:nonselection-rfde}
\end{equation}
whose compatible \(C^1\) histories satisfy
\begin{equation}
 \mathcal M^1_\delta
 =\{\phi\in X^1_\delta:\phi'(0)=\delta^{-2}\phi(0)\}.        \label{eq:nonselection-manifold}
\end{equation}

Even after fixing the common phase row
\begin{equation}
 \phi_y(0)=1,                                        \label{eq:nonselection-phase-row}
\end{equation}
bounded complete backward extension and convergence to zero in the
unnormalized history norm do not select a history.  Indeed, for every
\(A\in\R\),
\begin{equation}
 x_{\delta,A}(t)=e^{t/\delta^2}(1,A),\qquad t\le0,   \label{eq:nonselection-family}
\end{equation}
solves \cref{eq:nonselection-rfde} and satisfies
\begin{equation}
 \sup_{t\le0}\|x_{\delta,A,t}\|_{X_\delta}<\infty,
 \qquad
 \|x_{\delta,A,t}\|_{X_\delta}\longrightarrow0
       \quad(t\to-\infty).                          \label{eq:nonselection-asymptotics}
\end{equation}

Even superalgebraic closeness in \(X^1_\delta\), together with smooth
parameter dependence for each fixed \(\delta\), does not imply a tame
\(C^1_\nu C^2_\eta\) history jet.  Fix \(0<a<b\), put
\(\omega_\delta=e^{b/\delta^2}\), and, on
\(K=[-1/2,1/2]^2\), define
\begin{align}
 A_\delta(\nu,\eta)
 &=e^{-a/\delta^2}(1+\nu)
   \{\sin(\omega_\delta\eta)+1-\cos(\omega_\delta\eta)\},\notag\\
 \Phi_\delta(\nu,\eta)(\vartheta)
 &=e^{\vartheta/\delta^2}
       (1,A_\delta(\nu,\eta)),
 \qquad -\tau_\delta\le\vartheta\le0.             \label{eq:nonselection-jet-family}
\end{align}
The formula extends to a \(C^\infty\) map from \(\R^2\) to the closed
linear space \(\mathcal M^1_\delta\); restrict it to \(K\).  Every member
obeys \cref{eq:nonselection-phase-row}, extends to a solution satisfying
\cref{eq:nonselection-asymptotics}, and, relative to
\(\Phi_\delta^{\rm ref}(\vartheta)=e^{\vartheta/\delta^2}(1,0)\),
\begin{equation}
 \sup_K\|\Phi_\delta-\Phi_\delta^{\rm ref}\|_{X^1_\delta}
 \le\frac92(1+\delta^{-2})e^{-a/\delta^2}
 =O(\delta^q)\quad\text{for every }q>0.              \label{eq:nonselection-flat-size}
\end{equation}
At \((\nu,\eta)=(0,0)\), however,
\begin{align}
 \|\partial_\eta\Phi_\delta\|_{X_\delta}
 &=e^{(b-a)/\delta^2},&
 \|\partial_{\eta\eta}\Phi_\delta\|_{X_\delta}
 &=e^{(2b-a)/\delta^2},\notag\\
 \|\partial_\nu\partial_\eta\Phi_\delta\|_{X_\delta}
 &=e^{(b-a)/\delta^2},&
 \|\partial_\nu\partial_{\eta\eta}\Phi_\delta\|_{X_\delta}
 &=e^{(2b-a)/\delta^2}.                              \label{eq:nonselection-jet-blowup}
\end{align}
In particular, for arbitrary fixed \(C>0\), \(M,m,c\ge0\), and \(p>0\),
the ratio of each of the four displayed derivative norms to the envelope
\begin{equation}
 C\delta^{-M}\langle S_\delta\rangle^m e^{cS_\delta},
 \qquad S_\delta=\sqrt{2p\log(1/\delta)},            \label{eq:nonselection-tame-envelope}
\end{equation}
tends to \(+\infty\) as \(\delta\downarrow0\).  Since
\(\|\cdot\|_{X_\delta}\le\|\cdot\|_{X^1_\delta}\), the same conclusion
holds a fortiori for the strong-history derivative norms.
\end{proposition}

\begin{proof}
The functional \(\mathcal F_\delta\) is bounded linear on \(X_\delta\),
so \cref{eq:nonselection-rfde} is an RFDE, in fact in the nondelayed
subclass.  Direct differentiation proves \cref{eq:nonselection-family}.
Its history segment is
\begin{equation*}
 x_{\delta,A,t}(\vartheta)=e^{(t+\vartheta)/\delta^2}(1,A),
\end{equation*}
and hence
\begin{equation*}
 \|x_{\delta,A,t}\|_{X_\delta}=(1+|A|)e^{t/\delta^2},
 \qquad
 \|x_{\delta,A,t}\|_{X^1_\delta}
 =(1+\delta^{-2})(1+|A|)e^{t/\delta^2}.
\end{equation*}
This proves \cref{eq:nonselection-asymptotics}; the distinguished histories
\(x_{\delta,A,0}\) all satisfy the same phase row, while present evaluation
of the second component recovers the free coefficient \(A\).  Moreover,
\(\partial_\vartheta\phi(0)=\delta^{-2}\phi(0)\), so these histories and
the family below lie in \(\mathcal M^1_\delta\), which is the kernel of a
bounded linear map on \(X^1_\delta\) and hence is closed.

For the second component of \cref{eq:nonselection-jet-family}, relative to
the reference history, use
\(|\sin u|\le1\), \(|1-\cos u|\le2\), and
\(1+\nu\le3/2\).  This gives \cref{eq:nonselection-flat-size}, since
\(e^{-a/\delta^2}\) dominates every power of \(\delta\).  At \(\eta=0\),
direct differentiation of \(A_\delta\) gives coefficients
\begin{equation*}
 e^{(b-a)/\delta^2},\quad e^{(2b-a)/\delta^2},\quad
 e^{(b-a)/\delta^2},\quad e^{(2b-a)/\delta^2}
\end{equation*}
in the four derivative slots of
\cref{eq:nonselection-jet-blowup}.  The exponential profile has
\(X_\delta\)-norm one, proving the equalities.  Finally, the logarithm of
the ratio of the first quantity to
\cref{eq:nonselection-tame-envelope} is
\begin{equation*}
 \frac{b-a}{\delta^2}-M\log(1/\delta)
 -m\log\langle S_\delta\rangle-cS_\delta-\log C,
\end{equation*}
which tends to \(+\infty\).  The two derivative norms with exponent
\((2b-a)/\delta^2\) diverge still faster, proving all four assertions.
\end{proof}

Thus the words ``complete backward'', ``bounded'', and ``convergent in an
unnormalized history norm'', even with the displayed phase row, do not by
themselves select a complete history.  Nor may tame parameter jets be
inferred from fixed-\(\delta\) smoothness and superalgebraic \(C^0\) or
\(C^1\) closeness.  A valid construction must add a model-specific selection
or normalization and prove its uniform jets.
