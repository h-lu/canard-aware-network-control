\section{Hidden-return structure and quantitative persistence}
\label{sec:abstract-response}

\subsection{The cokernel hidden-return map}

We first isolate the structural statement which determines the response
coefficient.  Its algebra is the standard one-dimensional Schur elimination;
the point of recording it is to state exactly which source, inverse, return,
and uniform bounds must be realized by the RFDE.  No independent novelty is
claimed for the block elimination itself.  It is distinct from the
implicit-function argument below.

\begin{lemma}[Projection--lift chain rule]
\label[lemma]{lem:projection-lift-chain}
Let \(\mathcal H,\mathcal X,\mathcal C,\mathcal U\) be Banach spaces,
\(\Pi\in\mathcal L(\mathcal X,\mathcal C)\), and suppose
\(F:(-\eta_0,\eta_0)\times\mathcal H\to\mathcal X\) and
\(\mathcal I:(-\eta_0,\eta_0)\times\mathcal U\to\mathcal H\) are jointly
\(C^1\) on the relevant open neighborhoods.  Write
\(F_\eta=F(\eta,\cdot)\) and
\(\mathcal I_\eta=\mathcal I(\eta,\cdot)\).  If
\begin{equation}
 \Pi F_\eta=\Pi F_0
 \quad\hbox{as maps on \(\mathcal H\) for every admitted \(\eta\)}, \label{eq:abstract-exact-blindness}
\end{equation}
then the reduced field \(Q_\eta=\Pi F_\eta\circ\mathcal I_\eta\) obeys
\begin{equation}
 \left.\partial_\eta Q_\eta(u)\right|_{\eta=0}
 =D(\Pi F_0)(\mathcal I_0(u))
      \left.\partial_\eta\mathcal I_\eta(u)\right|_{\eta=0}.     \label{eq:projection-lift-chain}
\end{equation}
Thus exact blindness removes the direct structural derivative but need not
remove the derivative transmitted through the invariant lift.
\end{lemma}

\begin{proof}
Differentiate \(\Pi F_\eta(\mathcal I_\eta(u))\).  The direct term is
\(\Pi(\partial_\eta F_\eta)|_{\eta=0}=0\) by
\cref{eq:abstract-exact-blindness}; the remaining chain-rule term is
\cref{eq:projection-lift-chain}.
\end{proof}

\subsection{An abstract two-atom source criterion}

The following elementary moment lemma isolates the part of the mechanism
that depends only on separated delay locations.  Its purpose is to give a
portable verification interface; the nonlinear RFDE content begins only
after this source has been lifted through the history equation and returned
to a collective cokernel.

\begin{proposition}[Sharp two-atom inversion of a redistributed moment]
\label[proposition]{prop:abstract-two-atom-source}
Let \(\mathcal A\) and \(E\ne\{0\}\) be real Banach spaces, let
\(\mathsf v\in\mathcal L(\mathcal A,E)\), fix real locations
\(\theta_0,\ldots,\theta_m\), and let
\(c\in\R\setminus\{0\}\).  Define
\[
 \mathcal A_{\rm red}^{m+1}
 =\left\{a=(a_0,\ldots,a_m)\in\mathcal A^{m+1}:
          \sum_{k=0}^m a_k=0\right\},
 \qquad
 \|a\|_1=\sum_{k=0}^m\|a_k\|_{\mathcal A},
\]
and
\begin{equation}
 \mathsf S_{\theta,c}a
 =c\,\mathsf v\!\left(\sum_{k=0}^m\theta_ka_k\right).
 \label{eq:abstract-two-atom-source}
\end{equation}
Write
\(\theta_-=\min_k\theta_k\),
\(\theta_+=\max_k\theta_k\), and
\(\Delta_\theta=\theta_+-\theta_-\).
If \(\Delta_\theta>0\), then
\begin{equation}
 \operatorname{Ran}\mathsf S_{\theta,c}=\operatorname{Ran}\mathsf v,
 \qquad
 \|\mathsf S_{\theta,c}\|
 =\frac{|c|\Delta_\theta}{2}\|\mathsf v\|.          \label{eq:abstract-moment-norm}
\end{equation}
Suppose in addition that
\(\mathsf j\in\mathcal L(E,\mathcal A)\) is a bounded right inverse of
\(\mathsf v\).  Choose indices \(k_-\) and \(k_+\) at the extreme
locations.  Then
\begin{equation}
 \begin{aligned}
  (\mathsf Q_{\theta,c}y)_{k_-}
   &=-\frac{\mathsf j y}{c\Delta_\theta},&
  (\mathsf Q_{\theta,c}y)_{k_+}
   &= \frac{\mathsf j y}{c\Delta_\theta},\\
  (\mathsf Q_{\theta,c}y)_k&=0&&(k\notin\{k_-,k_+\})
 \end{aligned}                                      \label{eq:abstract-two-atom-right-inverse}
\end{equation}
defines a right inverse of \(\mathsf S_{\theta,c}\), with
\begin{equation}
 \|\mathsf Q_{\theta,c}\|
 =\frac{2\|\mathsf j\|}{|c|\Delta_\theta},
 \qquad
 \|\mathsf S_{\theta,c}\|\|\mathsf Q_{\theta,c}\|
 =\|\mathsf v\|\|\mathsf j\|\ge1.                \label{eq:abstract-moment-right-inverse-norm}
\end{equation}
If \(\|\mathsf v\|=\|\mathsf j\|=1\), the two-atom inverse realizes
the pointwise minimum probe cost and the dual channel is a scaled isometry:
\begin{align}
 \inf_{\mathsf S_{\theta,c}a=y}\|a\|_1
 &=\|\mathsf Q_{\theta,c}y\|_1
   =\frac{2\|y\|_E}{|c|\Delta_\theta},             \label{eq:abstract-moment-optimal-cost}\\
 \|\ell\mathsf S_{\theta,c}\|
 &=\frac{|c|\Delta_\theta}{2}\|\ell\|_{E^*},
 \qquad
 \ell(y)=(\ell\mathsf S_{\theta,c})(\mathsf Q_{\theta,c}y) \label{eq:abstract-moment-dual-isometry}
\end{align}
for every \(y\in E\) and \(\ell\in E^*\).  If
\(\Delta_\theta=0\), then \(\mathsf S_{\theta,c}=0\) on
\(\mathcal A_{\rm red}^{m+1}\).
\end{proposition}

\begin{proof}
Put \(\bar\theta=(\theta_-+\theta_+)/2\).  Since \(\sum_ka_k=0\),
\[
 \|\mathsf S_{\theta,c}a\|_E
 =|c|\left\|\mathsf v\!\left(
       \sum_k(\theta_k-\bar\theta)a_k\right)\right\|_E
 \le\frac{|c|\Delta_\theta}{2}\|\mathsf v\|\|a\|_1.
\]
Putting \(-x/2\) at \(k_-\), \(x/2\) at \(k_+\), and zero elsewhere,
then taking the supremum over nonzero \(x\in\mathcal A\), proves the norm
equality.  The same construction without the factors \(1/2\) proves the
range identity.

The tuple in \cref{eq:abstract-two-atom-right-inverse} has zero sum and
\(\sum_k\theta_k(\mathsf Q_{\theta,c}y)_k=c^{-1}\mathsf j y\), so it is a
right inverse.  Its exact norm is the first equality in
\cref{eq:abstract-moment-right-inverse-norm}; the product formula follows
from \cref{eq:abstract-moment-norm}, and it is at least one because
\(\mathsf v\mathsf j=\Id_E\).

If both operator norms are one, then
\(\|y\|_E\le\|\mathsf j y\|_{\mathcal A}\le\|y\|_E\).  Every solution
of \(\mathsf S_{\theta,c}a=y\) is bounded below by the reciprocal source
norm, and \(\mathsf Q_{\theta,c}y\) attains this bound.  Finally,
\[
 \|\ell\|
 \le\|\ell\mathsf S_{\theta,c}\|\|\mathsf Q_{\theta,c}\|,
 \qquad
 \|\ell\mathsf S_{\theta,c}\|
 \le\|\ell\|\|\mathsf S_{\theta,c}\|,
\]
and the reciprocal norm identities force equality.  If
\(\Delta_\theta=0\), all locations equal \(\theta_*\), so
\(\sum_k\theta_ka_k=\theta_*\sum_ka_k=0\).
\end{proof}

For each \(N\), let
\(\mathcal T_N,\mathcal D_{\perp,N},\mathcal Y_{\perp,N},
\mathcal D_{\parallel,N},\mathcal Y_{\parallel,N}\) be real Banach spaces,
where \(\mathcal T_N\) consists of projection-invisible structural
directions.  Let
\[
 L_{\perp,N}:\mathcal D_{\perp,N}\longrightarrow\mathcal Y_{\perp,N},
 \qquad
 L_{\parallel,N}:\mathcal D_{\parallel,N}
       \longrightarrow\mathcal Y_{\parallel,N}
\]
be bounded linear operators, and let
\[
 \mathsf S_N:\mathcal T_N\to\mathcal Y_{\perp,N},\qquad
 \mathsf B_N:\mathcal D_{\perp,N}\to\mathcal Y_{\parallel,N}
\]
be bounded and linear.  Here \(\mathsf S_NR\) is the transverse source left
by a blind direction,
\(L_{\perp,N}^{-1}\) propagates it through the full-history range problem,
and \(\mathsf B_N\) returns that response to the collective equation.

\begin{theorem}[Cokernel Schur complement for a hidden response]
\label[theorem]{thm:hidden-return-schur}
Assume that \(L_{\perp,N}\) is a bounded isomorphism, that
\(\operatorname{Ran}L_{\parallel,N}\) is closed of codimension one, and that
\(f_{\nu,N}\in\mathcal Y_{\parallel,N}\) has nonzero class in
\(\operatorname{coker}L_{\parallel,N}\).  Define the hidden-return map
intrinsic to this fixed collective cokernel,
\begin{equation}
 \mathcal K_N(R)
 =\left[\mathsf B_NL_{\perp,N}^{-1}\mathsf S_NR\right]
 \in\operatorname{coker}L_{\parallel,N}.               \label{eq:hidden-return-cokernel}
\end{equation}
There is a unique bounded linear functional
\(\Lambda_N:\mathcal T_N\to\R\) such that
\begin{equation}
 \mathcal K_N(R)+\Lambda_N(R)[f_{\nu,N}]=0
 \quad\hbox{in }\operatorname{coker}L_{\parallel,N}.   \label{eq:hidden-return-balance}
\end{equation}
Equivalently, if
\(\mathfrak l_N\in\mathcal Y_{\parallel,N}^*\setminus\{0\}\) satisfies
\(\ker\mathfrak l_N=\operatorname{Ran}L_{\parallel,N}\), then
\begin{equation}
 \Lambda_N(R)
 =-\frac{\mathfrak l_N\mathsf B_NL_{\perp,N}^{-1}\mathsf S_NR}
          {\mathfrak l_N f_{\nu,N}}.                  \label{eq:hidden-return-schur}
\end{equation}
This scalar is unchanged by rescaling \(\mathfrak l_N\) or adding a
collective range term \(L_{\parallel,N}w_R\),
\(w_R\in\mathcal D_{\parallel,N}\), to the returned forcing.  It is
also invariant under block-preserving transverse coordinate changes: if
\(J_N,K_N\) are Banach-space isomorphisms and
\begin{equation}
 \widetilde L_{\perp,N}=K_NL_{\perp,N}J_N^{-1},\qquad
 \widetilde{\mathsf S}_N=K_N\mathsf S_N,\qquad
 \widetilde{\mathsf B}_N=\mathsf B_NJ_N^{-1},         \label{eq:hidden-return-coordinate-change}
\end{equation}
then the product defining \(\mathcal K_N\) is unchanged.  The normalization
of the physical matching parameter, however, is held fixed.

Moreover,
\begin{equation}
 \|\Lambda_N\|
 \le
 \frac{\|\mathsf B_N\|\,\|L_{\perp,N}^{-1}\|\,
       \|\mathsf S_N\|}
      {\operatorname{dist}(f_{\nu,N},
          \operatorname{Ran}L_{\parallel,N})}.       \label{eq:hidden-return-uniform-bound}
\end{equation}
Consequently, common numerator bounds and a common positive lower bound on
the displayed distance give \(\sup_N\|\Lambda_N\|<\infty\).
Whenever \(\Lambda_N\ne0\), its zero set is a closed codimension-one
hyperplane in \(\mathcal T_N\), and its complement is open and dense.
\end{theorem}

\begin{proof}
The cokernel is one dimensional and \([f_{\nu,N}]\ne0\), so this class is a
basis.  Every class \(\mathcal K_N(R)\) therefore has a unique coefficient in
that basis, which proves \cref{eq:hidden-return-balance}.  Applying any
\(\mathfrak l_N\) with the stated kernel gives \cref{eq:hidden-return-schur};
linearity and boundedness follow by composition.  The quotient norm of
\([f_{\nu,N}]\) is its distance from the closed range, which gives
\cref{eq:hidden-return-uniform-bound}.  Rescaling the
cokernel functional cancels between numerator and denominator.  A collective
range term is killed by \(\mathfrak l_N\), and
\cref{eq:hidden-return-coordinate-change} cancels inside the three-factor
product.  Finally, the kernel of a
nonzero bounded scalar functional is closed and has codimension one.
\end{proof}

The meaning of \cref{thm:hidden-return-schur} is the first-order block system
\begin{equation}
 L_{\perp,N}h_N=\mathsf S_NR,\qquad
 L_{\parallel,N}x_N+f_{\nu,N}\lambda_N+\mathsf B_Nh_N=0.       \label{eq:hidden-return-block-system}
\end{equation}
By \cref{lem:projection-lift-chain}, exact projection blindness removes the
direct structural derivative of the reduced vector field.  For
\cref{eq:hidden-return-block-system}, one must additionally verify that
direct structural derivatives of auxiliary matcher, phase, endpoint, or
preparation rows vanish in the cokernel---either because they are zero or
because they belong to \(\operatorname{Ran}L_{\parallel,N}\).  After this
model-dependent check, eliminating \(h_N\) and passing to the collective
cokernel gives
\(\lambda_N=\Lambda_N(R)\).  This is the hidden-return calculation; existence
of a nearby nonlinear root with a controlled remainder is a separate step.

The collective cokernel used in the shared-resource model is not one of the
finite-interval trace--phase--endpoint isomorphisms.  It is the following
phase-fixed whole-line operator.

\begin{lemma}[Gaussian cokernel of the fold matching operator]
\label[lemma]{lem:fold-gaussian-cokernel}
Fix \(c>0\) and \(M\ge0\), and define
\begin{align*}
 \mathcal Y_\infty^{c,M}
 &=\left\{f\in C(\R;\R^2):
   \|f\|_{\mathcal Y_\infty^{c,M}}
   :=\sup_{s\in\R}e^{-c|s|}\langle s\rangle^{-M}|f(s)|<\infty\right\},\\
 \mathcal X_\infty^{c,M+1}
 &=\left\{\xi\in C(\R;\R^2):
   \|\xi\|_{\mathcal X_\infty^{c,M+1}}
   :=\sup_{s\in\R}e^{-c|s|}\langle s\rangle^{-M-1}|\xi(s)|<\infty\right\}.
\end{align*}
For \(\xi=(U,V)\), let
\[
 L_0\xi=(U'-sU-V,V'+U)
\]
and give the phase-fixed domain
\begin{equation}
 \mathcal D_\infty^{c,M}
 =\{\xi\in C^1(\R;\R^2):
      \xi\in\mathcal X_\infty^{c,M+1},
      L_0\xi\in\mathcal Y_\infty^{c,M},\ U(0)=0\}     \label{eq:whole-line-fold-domain}
\end{equation}
its graph norm; this is a Banach space because the differential operator is
closed in the displayed weighted spaces.  Then
\(L_0:\mathcal D_\infty^{c,M}\to\mathcal Y_\infty^{c,M}\) is injective,
has closed range of codimension one, and
\begin{equation}
 \operatorname{Ran}L_0=\ker\mathfrak l_0,
 \qquad
 \mathfrak l_0f
 =\int_\R e^{-s^2/2}\{sf_1(s)+f_2(s)\}\,\dd s.        \label{eq:fold-gaussian-cokernel}
\end{equation}
Moreover, \(L_0\) has a bounded right inverse from
\(\ker\mathfrak l_0\) to \(\mathcal D_\infty^{c,M}\).
\end{lemma}

\begin{proof}
Put \(E(s)=e^{s^2/2}\), \(I(s)=\int_0^sE(t)\,\dd t\), and
\[
 \tau(s)=(-1,s)^T,\qquad
 n(s)=(I(s),E(s)-sI(s))^T.
\]
These form a fundamental pair for \(L_0\xi=0\).  Writing
\(\xi=a\tau+bn\), variation of constants gives
\begin{equation}
 a'=E^{-1}\{-(E-sI)f_1+If_2\}=:A_f,\qquad
 b'=E^{-1}(sf_1+f_2)=:B_f.                            \label{eq:whole-line-ab}
\end{equation}
For every \(\xi\in\mathcal D_\infty^{c,M}\),
\[
 b(s)=E(s)^{-1}\{sU(s)+V(s)\}\longrightarrow0
 \quad(s\to\pm\infty).
\]
Integrating the second equation in \cref{eq:whole-line-ab} therefore shows
that \(\mathfrak l_0(L_0\xi)=0\).

Conversely, if \(f\in\ker\mathfrak l_0\), set
\begin{equation}
 a(s)=\int_0^sA_f(t)\,\dd t,\qquad
 b(s)=\int_{-\infty}^sB_f(t)\,\dd t
     =-\int_s^\infty B_f(t)\,\dd t.                  \label{eq:whole-line-green}
\end{equation}
Then \(\xi=a\tau+bn\) solves \(L_0\xi=f\) and has \(U(0)=0\).
The elementary bounds
\[
 |I(s)|\le CE(s)\langle s\rangle^{-1},\qquad
 |E(s)-sI(s)|\le CE(s)
\]
together with
\begin{align*}
 \left|\int_s^\infty e^{-t^2/2}(tf_1(t)+f_2(t))\,\dd t\right|
 &\le C_{c,M}e^{-s^2/2+c|s|}\langle s\rangle^M\|f\|_{\mathcal Y_\infty^{c,M}}
 &&(s\ge0),\\
 \left|\int_{-\infty}^s e^{-t^2/2}(tf_1(t)+f_2(t))\,\dd t\right|
 &\le C_{c,M}e^{-s^2/2+c|s|}\langle s\rangle^M\|f\|_{\mathcal Y_\infty^{c,M}}
 &&(s\le0)
\end{align*}
and the elementary exponential integral bound for \(A_f\), give
\[
 \|\xi\|_{\mathcal X_\infty^{c,M+1}}
 \le C_{c,M}\|f\|_{\mathcal Y_\infty^{c,M}}.
\]
Since \(L_0\xi=f\), the same inequality plus
\(\|L_0\xi\|_{\mathcal Y_\infty^{c,M}}=\|f\|_{\mathcal Y_\infty^{c,M}}\)
is the required graph-norm estimate.  The graph domain is complete because
the maximal first-order operator is closed in the displayed weighted
spaces, and the phase condition \(U(0)=0\) is closed.  This constructs the
asserted bounded right inverse.  If \(f=0\), the
coefficient \(b\) is constant and the Gaussian mode \(n\) is excluded by
\(\mathcal X_\infty^{c,M+1}\); the phase row then removes the remaining
tangent mode \(\tau\).  Hence the operator is injective.  Finally,
\(\mathfrak l_0\) is a nonzero bounded functional on
\(\mathcal Y_\infty^{c,M}\), so its kernel is closed and has codimension one.
\end{proof}

\subsection{Quantitative persistence of the response coefficient}

The construction of a selected complete-history root has two logically
different parts.  A range argument must first construct the history graph and
the two selected traces, including their parameter derivatives.  Once that
has been done, a scalar argument turns two Melnikov pairings and their
remainders into a root response.  We isolate the latter implication here so
that the model-specific calculation below has a reusable conclusion.  In
particular, the following theorem does not construct an invariant graph or a
trace inverse: their existence and regularity are explicit hypotheses.

Let \(I\subset\R\) be a compact interval.  For each \(N\ge2\), let
\(\mathcal H_N\) be a history space, let
\(\nu_{0,N}\in I\), and assume
\begin{equation}
 \inf_{N\ge2}\operatorname{dist}(\nu_{0,N},\partial I)=d_*>0.
 \label{eq:abstract-center-buffer}
\end{equation}
The notation below permits the graph, the trace spaces, and their norms to
depend on \(N\) and \(\delta\); only the constants displayed as common are
required to be independent of \(N\).

\begin{theorem}[Uniform range-to-root and Melnikov transfer]
\label{thm:abstract-root-response}
Fix \(\bar\delta,\bar\zeta>0\).  Suppose that the following data have been
verified for every \(N\ge2\), \(0<\delta<\bar\delta\), \(\nu\in I\), and
\(|\zeta|<\bar\zeta\).

\smallskip
\noindent
\emph{Complete-history range.}
An already solved range problem supplies a Banach chart \(\mathcal U_N\),
a phase section \(\Sigma_N\subset\mathcal U_N\), an exact
complete-history lift with invariant range
\[
 \mathcal I_{N,\delta,\nu,\zeta}:\mathcal U_N\longrightarrow\mathcal H_N,
\]
and selected one-sided traces
\(z^a_{N,\delta,\nu,\zeta}\) and
\(z^r_{N,\delta,\nu,\zeta}\).  Their central values lie on the same phase
section.  Present-state evaluation \(\operatorname{ev}_{0,N}\) is a left
inverse of \(\mathcal I_{N,\delta,\nu,\zeta}\), so the lift is injective.
A scalar matcher
\(\mathfrak m_{N,\delta,\nu,\zeta}:\Sigma_N\times\Sigma_N\to\R\)
defines
\begin{equation}
 \mathscr G_N(\delta,\nu,\zeta)
 =\mathfrak m_{N,\delta,\nu,\zeta}
   \bigl(z^a_{N,\delta,\nu,\zeta}(0),
         z^r_{N,\delta,\nu,\zeta}(0)\bigr).
 \label{eq:abstract-gap-definition}
\end{equation}
The matcher has the zero-fiber property
\begin{equation}
 \mathscr G_N(\delta,\nu,\zeta)=0
 \quad\Longleftrightarrow\quad
 \mathcal I_{N,\delta,\nu,\zeta}
       \bigl(z^a_{N,\delta,\nu,\zeta}(0)\bigr)
 =\mathcal I_{N,\delta,\nu,\zeta}
       \bigl(z^r_{N,\delta,\nu,\zeta}(0)\bigr).
 \label{eq:abstract-zero-fiber}
\end{equation}
Thus a zero is equality of the selected histories, not merely equality of a
projected observable.  The lift, traces, and matcher possess the parameter
jets needed to make \(\mathscr G_N\) jointly \(C^2\) in \((\nu,\zeta)\) on
an open neighborhood of the closed parameter box.  This includes the pure
\(\nu\nu\) and mixed \(\nu\zeta\) derivatives.

\smallskip
\noindent
\emph{Melnikov data and tails.}
There are Banach spaces \(V_N\), measurable functions
\(\psi_N:\R\to V_N^*\) and
\(f_{\nu,N},f_{\zeta,N}:\R\to V_N\), and numbers
\(R_\delta\to\infty\) as \(\delta\to0\), such that the absolutely
convergent pairings
\begin{align}
 a_N&=\int_{\R}\langle\psi_N(s),f_{\nu,N}(s)\rangle\,\dd s,
 &
 b_N&=\int_{\R}\langle\psi_N(s),f_{\zeta,N}(s)\rangle\,\dd s
 \label{eq:abstract-melnikov-coefficients}
\end{align}
represent the derivatives of the same normalized matcher as in
\cref{eq:abstract-gap-definition}.  Thus any phase, endpoint, or moving-history
contribution must already be included in these profiles or proved to vanish.
Assume that, with constants independent of \(N\),
\begin{equation}
 0<a_*\le |a_N|\le a^*,\qquad |b_N|\le b^*.
 \label{eq:abstract-coefficient-bounds}
\end{equation}
Put
\begin{equation}
 a_{N,\delta}=\int_{-R_\delta}^{R_\delta}
     \langle\psi_N,f_{\nu,N}\rangle\,\dd s,
 \qquad
 b_{N,\delta}=\int_{-R_\delta}^{R_\delta}
     \langle\psi_N,f_{\zeta,N}\rangle\,\dd s.
 \label{eq:abstract-truncated-coefficients}
\end{equation}
Assume the common tail estimate
\begin{equation}
 \int_{|s|>R_\delta}
 \left(
  |\langle\psi_N(s),f_{\nu,N}(s)\rangle|
  +|\langle\psi_N(s),f_{\zeta,N}(s)\rangle|
 \right)\dd s
 \le C_{\rm tail}\delta.
 \label{eq:abstract-tail-bound}
\end{equation}

\smallskip
\noindent
\emph{Range-to-gap remainders.}
The output of the graph and trace estimates obeys, throughout the same
parameter box,
\begin{align}
 \left|\frac{\mathscr G_N(\delta,\nu,\zeta)}{\delta}
       -a_{N,\delta}(\nu-\nu_{0,N})\right|
 &\le C_{\rm rg}\delta,
 \label{eq:abstract-gap-level}\\
 |\partial_\nu\mathscr G_N-\delta a_{N,\delta}|
 &\le C_{\rm rg}\delta^2,
 \label{eq:abstract-gap-slope}\\
 |\partial_\zeta\mathscr G_N-\delta^2b_{N,\delta}|
 &\le C_{\rm rg}(\delta^3+\delta^2|\zeta|),
 \label{eq:abstract-gap-response}\\
 |\partial_{\nu\nu}\mathscr G_N|
 +|\partial_{\nu\zeta}\mathscr G_N|
 +|\partial_{\zeta\zeta}\mathscr G_N|
 &\le C_{\rm rg}\delta^2.
 \label{eq:abstract-gap-second-jets}
\end{align}

Then there are \(\delta_0\in(0,\bar\delta)\),
\(\zeta_0\in(0,\bar\zeta)\), \(c_0\in(0,d_*)\), and \(C>0\), all
independent of \(N\), such that for every \(N\ge2\),
\(0<\delta<\delta_0\), and \(|\zeta|<\zeta_0\), the equation
\(\mathscr G_N(\delta,\nu,\zeta)=0\) has exactly one solution in
\(|\nu-\nu_{0,N}|<c_0\).  This solution is a selected complete-history
root by \cref{eq:abstract-zero-fiber}, is \(C^2\) in \(\zeta\), and
satisfies
\begin{align}
 \partial_\zeta\nu_{c,N}(\delta,\zeta)
 &=-\delta\frac{b_N}{a_N}
   +O(\delta^2+\delta|\zeta|),
 \label{eq:abstract-root-derivative}\\
 |\partial_{\zeta\zeta}\nu_{c,N}(\delta,\zeta)|
 &\le C\delta,
 \label{eq:abstract-root-second-derivative}\\
 \left|\nu_{c,N}(\delta,\zeta)-\nu_{c,N}(\delta,0)
       +\delta\frac{b_N}{a_N}\zeta\right|
 &\le C(\delta^2|\zeta|+\delta\zeta^2).
 \label{eq:abstract-root-increment}
\end{align}
The constant implicit in \cref{eq:abstract-root-derivative} is common in
\(N\).  More generally, if a physical parameter is scaled as
\(\mu=\mu_f(\delta)+\delta^q\nu\), where \(q>0\) and \(\mu_f\) is
independent of \(\zeta\), then
\begin{equation}
 \left|\mu_{c,N}(\delta,\zeta)-\mu_{c,N}(\delta,0)
       +\frac{b_N}{a_N}\delta^{q+1}\zeta\right|
 \le C\{\delta^{q+2}|\zeta|+\delta^{q+1}\zeta^2\}.
 \label{eq:abstract-physical-increment}
\end{equation}
In particular, \(q=2\) gives a leading \(\delta^3\zeta\) response.
\end{theorem}

\begin{proof}
The tail bound gives
\begin{equation}
 |a_{N,\delta}-a_N|+|b_{N,\delta}-b_N|
 \le C_{\rm tail}\delta.                            \label{eq:abstract-tail-consequence}
\end{equation}
Consequently, after decreasing \(\delta_0\),
\begin{align}
 \left|\frac{\mathscr G_N}{\delta}
       -a_N(\nu-\nu_{0,N})\right|&\le C_0\delta,
 \label{eq:abstract-full-gap-level}\\
 |\partial_\nu\mathscr G_N-\delta a_N|&\le C_1\delta^2,
 \label{eq:abstract-full-gap-slope}\\
 |\partial_\zeta\mathscr G_N-\delta^2b_N|
 &\le C_2(\delta^3+\delta^2|\zeta|),
 \label{eq:abstract-full-gap-response}
\end{align}
uniformly in the parameter box.  Here the bounded diameter of \(I\) absorbs
the factor \(|\nu-\nu_{0,N}|\) when
\cref{eq:abstract-tail-consequence} is inserted into
\cref{eq:abstract-gap-level}.

Set \(\zeta_0=\bar\zeta/2\) and choose, for example,
\(c_0=d_*/2\).  At
\(\nu=\nu_{0,N}\pm c_0\),
\cref{eq:abstract-full-gap-level} has the two opposite leading values
\(\pm a_Nc_0\).  They retain opposite signs once
\(C_0\delta_0<a_*c_0/2\), so continuity gives a zero in the local
window.  If also \(C_1\delta_0<a_*/2\), then
\begin{equation}
 |\partial_\nu\mathscr G_N|\ge\frac12a_*\delta
 \label{eq:abstract-uniform-slope}
\end{equation}
and its sign is the sign of \(a_N\) throughout that window.  The zero is
therefore unique.  The ordinary implicit-function theorem, applied at each
\(\zeta\), and uniqueness patch these local branches into one \(C^2\)
function on the common \(\zeta\)-interval.

Along the root,
\begin{equation}
 \partial_\zeta\nu_{c,N}
 =-\frac{\partial_\zeta\mathscr G_N}
          {\partial_\nu\mathscr G_N}.                \label{eq:abstract-ift-ratio}
\end{equation}
Write the numerator and denominator as
\(\delta^2b_N+e_\zeta\) and \(\delta a_N+e_\nu\).  By
\cref{eq:abstract-full-gap-slope,eq:abstract-full-gap-response},
\(|e_\nu|\le C_1\delta^2\) and
\(|e_\zeta|\le C_2(\delta^3+\delta^2|\zeta|)\).  Using
\cref{eq:abstract-uniform-slope,eq:abstract-coefficient-bounds} in
\cref{eq:abstract-ift-ratio} gives
\begin{align*}
 \left|\partial_\zeta\nu_{c,N}
       +\delta\frac{b_N}{a_N}\right|
 &\le \frac{|e_\zeta|}{|\delta a_N+e_\nu|}
  +\frac{\delta^2|b_N|\,|e_\nu|}
   {|\delta a_N|\,|\delta a_N+e_\nu|} \\
 &\le C(\delta^2+\delta|\zeta|),
\end{align*}
which is \cref{eq:abstract-root-derivative}.

Differentiating
\(\mathscr G_N(\delta,\nu_{c,N}(\zeta),\zeta)=0\) once more gives
\begin{equation*}
 \partial_\nu\mathscr G_N\,\partial_{\zeta\zeta}\nu_{c,N}
 =-\left\{
   \partial_{\nu\nu}\mathscr G_N(\partial_\zeta\nu_{c,N})^2
  +2\partial_{\nu\zeta}\mathscr G_N\partial_\zeta\nu_{c,N}
  +\partial_{\zeta\zeta}\mathscr G_N\right\}.
\end{equation*}
The first derivative is \(O(\delta)\); hence
\cref{eq:abstract-gap-second-jets,eq:abstract-uniform-slope} yield
\cref{eq:abstract-root-second-derivative}.  Integrating
\cref{eq:abstract-root-derivative} from \(0\) to \(\zeta\) proves
\cref{eq:abstract-root-increment}.  Multiplication by \(\delta^q\) proves
\cref{eq:abstract-physical-increment}.
\end{proof}

\begin{corollary}[Schur-to-root persistence]
\label[corollary]{cor:schur-to-root}
For every \(N\ge2\), let \(R_N\in\mathcal T_N\), with
\(\sup_N\|R_N\|_{\mathcal T_N}<\infty\), and suppose the hypotheses of
\cref{thm:hidden-return-schur,thm:abstract-root-response} hold with common
constants for this family.  Assume that
\(\mathcal Y_{\parallel,N}\) is a Banach space of the
\(V_N\)-valued whole-line profiles used in
\cref{eq:abstract-melnikov-coefficients}, and choose its bounded cokernel
functional to be
\[
 \mathfrak l_Nf=\int_\R\langle\psi_N(s),f(s)\rangle\,\dd s,
 \qquad
 \ker\mathfrak l_N=\operatorname{Ran}L_{\parallel,N}.
\]
Assume that the profiles in \cref{eq:abstract-melnikov-coefficients}
satisfy
\begin{equation}
 a_N=\mathfrak l_Nf_{\nu,N},\qquad
 f_{\zeta,N}
 =\mathsf B_NL_{\perp,N}^{-1}\mathsf S_NR_N
   +L_{\parallel,N}w_{N,R_N}                          \label{eq:schur-to-root-profiles}
\end{equation}
for some \(w_{N,R_N}\in\mathcal D_{\parallel,N}\), with the profile and
range-term bounds required by the common tail and range-to-gap hypotheses.
Thus all direct
structural contributions from matcher, phase, endpoint, and preparation rows
have either been included in the returned forcing or verified to lie in the
collective range.  Then
\begin{equation}
 b_N(R_N)=\mathfrak l_N\mathsf B_NL_{\perp,N}^{-1}\mathsf S_NR_N,
 \qquad -\frac{b_N(R_N)}{a_N}=\Lambda_N(R_N),         \label{eq:schur-to-root-ratio}
\end{equation}
and
\begin{align}
 \partial_\zeta\nu_{c,N}(\delta,\zeta)
 &=\delta\Lambda_N(R_N)+O(\delta^2+\delta|\zeta|),
                                                               \label{eq:schur-root-derivative}\\
 \left|\mu_{c,N}(\delta,\zeta)-\mu_{c,N}(\delta,0)
       -\delta^{q+1}\zeta\Lambda_N(R_N)\right|
 &\le C\{\delta^{q+2}|\zeta|+\delta^{q+1}\zeta^2\}. \label{eq:schur-root-increment}
\end{align}
The constants are uniform in \(N\) whenever the hypotheses of the two
theorems are uniform.  On every admissible tangent space where
\(\Lambda_N\ne0\), the leading zero-response directions form the
codimension-one hyperplane \(\ker\Lambda_N\).  If \(q=2\) and several fixed
preparations have the same leading block data in
\cref{eq:schur-to-root-profiles} while satisfying the range-to-gap bounds
with preparation-dependent constants, then all their baseline-subtracted
increments define the same dimension-uniform germ
\begin{equation}
 [\mu^{\mathbf p}_{c,N}(\delta,\zeta)
      -\mu^{\mathbf p}_{c,N}(\delta,0)]_{\rm resp}
 =[\delta^3\zeta\Lambda_N(R_N)]_{\rm resp}.           \label{eq:abstract-canonical-germ}
\end{equation}
This last assertion is pairwise for fixed preparations; it does not claim a
single remainder constant over an unbounded preparation class.
\end{corollary}

\begin{proof}
Applying \(\mathfrak l_N\) to
\cref{eq:schur-to-root-profiles} kills the collective range term and proves
\cref{eq:schur-to-root-ratio}.  Substitution into
\cref{eq:abstract-root-derivative,eq:abstract-physical-increment} gives
\cref{eq:schur-root-derivative,eq:schur-root-increment}; the final statement
about the hyperplane is \cref{thm:hidden-return-schur}.  For \(q=2\), divide
the difference between \cref{eq:schur-root-increment} and its leading term by
\(\delta^3|\zeta|\) and let \((\delta,\zeta)\to(0,0)\); this proves
\cref{eq:abstract-canonical-germ} for each fixed preparation and hence
pairwise on intersections of their parameter boxes.
\end{proof}

The quotient in \cref{eq:schur-to-root-ratio} has a geometric realization
which does not require choosing a scalar representative of the collective
cokernel.  We record it before returning to the model dictionary.  The word
``connection'' below refers to the zero-fiber equality of the two histories
supplied by \cref{thm:abstract-root-response}; it does not assert that those
histories have been selected by physical outer invariant manifolds.

\begin{theorem}[Selected weighted complete-history connection jet]
\label[theorem]{thm:weighted-connection-jet}
Assume the hypotheses of \cref{cor:schur-to-root}, and let
\(\mu=\mu_f(\delta)+\delta^q\nu\) be the fixed physical-parameter
normalization in \cref{eq:abstract-physical-increment}.  For each
\(\delta>0\), put
\begin{equation}
 \nu^b_{N,\delta}=\nu_{c,N}(\delta,0),\qquad
 \xi=\delta^{-1}(\nu-\nu^b_{N,\delta})
     =\delta^{-(q+1)}
       \{\mu-\mu_{c,N}(\delta,0)\}.                 \label{eq:weighted-connection-coordinate}
\end{equation}
In the local root window, the selected complete-history connection locus
along the structural ray \(\zeta R_N\) is a \(C^2\) curve
\begin{equation}
 \xi=\Xi_{N,\delta}(\zeta),\qquad
 \Xi_{N,\delta}(\zeta)
 =\delta^{-1}\{\nu_{c,N}(\delta,\zeta)
                         -\nu_{c,N}(\delta,0)\}.     \label{eq:weighted-connection-graph}
\end{equation}
There is one constant, independent of \(N\), such that
\begin{align}
 |\Xi_{N,\delta}(\zeta)-\zeta\Lambda_N(R_N)|
 &\le C\{\delta|\zeta|+\zeta^2\},                 \label{eq:weighted-connection-level}\\
 |\partial_\zeta\Xi_{N,\delta}(\zeta)-\Lambda_N(R_N)|
 &\le C(\delta+|\zeta|),                            \label{eq:weighted-connection-slope}\\
 |\partial_{\zeta\zeta}\Xi_{N,\delta}(\zeta)|
 &\le C.                                             \label{eq:weighted-connection-curvature}
\end{align}
Consequently the conormal line at the unperturbed selected
connection,
\begin{equation}
 \mathcal N^*_{N,\delta}(R_N)
 =\operatorname{span}\!\left\{
   \dd\xi-\partial_\zeta\Xi_{N,\delta}(0)\,\dd\zeta
   \right\},                                        \label{eq:selected-connection-conormal}
\end{equation}
converges, uniformly in \(N\), to
\begin{equation}
 \mathcal N^*_{N,0}(R_N)
 =\operatorname{span}\{\dd\xi-\Lambda_N(R_N)\,\dd\zeta\}.
                                                               \label{eq:limiting-connection-conormal}
\end{equation}

The family of raywise coefficients has the bounded cokernel-valued algebraic
extension
\begin{equation}
 \mathfrak J_N:\R\times\mathcal T_N
       \longrightarrow\operatorname{coker}L_{\parallel,N},
 \qquad
 \mathfrak J_N(\xi,R)
 =\xi[f_{\nu,N}]+\mathcal K_N(R),                   \label{eq:cokernel-connection-jet}
\end{equation}
where \(\mathcal K_N\) is defined in
\cref{eq:hidden-return-cokernel}.  It satisfies
\begin{equation}
 \ker\mathfrak J_N
 =\{(\Lambda_N(R),R):R\in\mathcal T_N\}.           \label{eq:cokernel-connection-kernel}
\end{equation}
Thus \(\ker\mathfrak J_N\), rather than a chosen scalar gap row, is the
candidate full weighted connection hyperplane.  Its realization as the
tangent hyperplane of a nonlinear locus requires the joint Fr\'{e}chet
hypotheses in \cref{thm:structural-ball-connection} and is stated in
\cref{cor:schur-full-conormal}.

The conormal line is independent of the regular scalar function used to
define the same local connection curve.  At finite \(\delta\), under a
\(C^1\) change of the \((\xi,\zeta)\)-parameter chart it transforms by
cotangent pullback.  Convergence of the limiting weighted line is preserved
for a fixed nonsingular chart, or for a chart family whose derivative and
inverse derivative converge uniformly to a nonsingular front-face derivative;
singular \(\delta\)-dependent rescalings are excluded.  In particular, the
numerical slope is covariant under a fixed parameter rescaling and is not
claimed to be invariant under arbitrary mixing of the physical and structural
parameters.  If several fixed preparations satisfy the final
hypothesis of \cref{cor:schur-to-root} with the same block maps, then
\cref{eq:cokernel-connection-jet} is their common algebraic extension.  After
each locus is centered at its own baseline as in
\cref{eq:weighted-connection-coordinate}, its restriction to the verified ray
gives the same conormal \cref{eq:limiting-connection-conormal}.  A nonlinear
full-space realization additionally needs
\cref{thm:structural-ball-connection}.  The comparison is pairwise and
supplies no uniform constant over an unbounded preparation class.
\end{theorem}

\begin{proof}
The zero-fiber property and the uniform nonzero \(\nu\)-slope in
\cref{thm:abstract-root-response} make the connection locus a local
\(C^2\) graph.  Divide \cref{eq:schur-root-increment} by
\(\delta^{q+1}\), and use
\cref{eq:weighted-connection-coordinate}; this gives
\cref{eq:weighted-connection-level}.  Dividing
\cref{eq:schur-root-derivative} by \(\delta\) gives
\cref{eq:weighted-connection-slope}, while
\cref{eq:abstract-root-second-derivative} gives
\cref{eq:weighted-connection-curvature}.  The graph is defined by
\(\xi-\Xi_{N,\delta}(\zeta)=0\), so its conormal line is
\cref{eq:selected-connection-conormal};
\cref{eq:weighted-connection-slope} at \(\zeta=0\) proves
\cref{eq:limiting-connection-conormal}.

By \cref{eq:hidden-return-balance},
\(\mathcal K_N(R)+\Lambda_N(R)[f_{\nu,N}]=0\).
Because \([f_{\nu,N}]\ne0\) spans the one-dimensional cokernel, this
identity proves \cref{eq:cokernel-connection-kernel}.  Two regular scalar
defining functions of the same curve have differentials with the same
one-dimensional annihilator and hence differ there by a nonzero scalar.
The chain rule gives cotangent pullback under a parameter-chart change.
Finally, \cref{eq:abstract-canonical-germ} and
\cref{eq:schur-root-derivative} give the stated pairwise preparation
comparison after separate baseline centering.
\end{proof}

The preceding theorem realizes one structural ray.  A conormal on the full
structural parameter space requires joint, rather than merely directional,
regularity.  The following Banach-parameter version states exactly the
additional estimate.  It starts from an already constructed
complete-history gap; as in \cref{thm:abstract-root-response}, it does not
construct the history range.

\begin{theorem}[Uniform structural-ball connection graph]
\label[theorem]{thm:structural-ball-connection}
For every \(N\ge2\), let \(\mathcal T_N\) be a real Banach space.  Fix
\(\bar\delta,\eta_*>0\), independent of \(N\), and equip
\(\R\times\mathcal T_N\) with the maximum product norm.  Suppose an exact
complete-history range and a zero-fiber matcher give a scalar gap
\[
 \mathscr G_N(\delta,\nu,\eta),
 \qquad 0<\delta<\bar\delta,\quad
 \nu\in I,\quad \eta\in B_{2\eta_*}(0)\subset\mathcal T_N,
\]
which, for each \(\delta\), is jointly \(C^2\) in \((\nu,\eta)\) on an open
neighborhood of \(I\times\overline B_{\eta_*}(0)\).  Let \(a_N\in\R\) and
\(b_N\in\mathcal T_N^*\) satisfy
\begin{equation}
 0<a_*\le|a_N|\le a^*,
 \qquad \sup_N\|b_N\|_{\mathcal T_N^*}\le b^*.       \label{eq:ball-coefficient-bounds}
\end{equation}
Assume, uniformly for
\((\delta,\nu,\eta)\in(0,\bar\delta)\times I\times
\overline B_{\eta_*}(0)\) and in \(N\),
\begin{align}
 \left|\frac{\mathscr G_N}{\delta}
       -a_N(\nu-\nu_{0,N})\right|
 &\le C_{\rm ball}\delta,                           \label{eq:ball-gap-level}\\
 |\partial_\nu\mathscr G_N-\delta a_N|
 &\le C_{\rm ball}\delta^2,                        \label{eq:ball-gap-slope}\\
 \|D_\eta\mathscr G_N-\delta^2b_N\|_{\mathcal T_N^*}
 &\le C_{\rm ball}\{\delta^3+\delta^2\|\eta\|_{\mathcal T_N}\},
                                                               \label{eq:ball-gap-response}\\
 |\partial_{\nu\nu}\mathscr G_N|
 +\|D_\eta\partial_\nu\mathscr G_N\|_{\mathcal T_N^*}
 +\|D_{\eta\eta}^2\mathscr G_N\|_{\mathcal L^2(\mathcal T_N;\R)}
 &\le C_{\rm ball}\delta^2.                        \label{eq:ball-gap-second-jets}
\end{align}
Assume also the common center-buffer condition
\cref{eq:abstract-center-buffer}.  Put
\begin{equation}
 \Lambda_N=-a_N^{-1}b_N\in\mathcal T_N^*.           \label{eq:ball-response-covector}
\end{equation}

There are \(0<\delta_0<\bar\delta\), \(0<\eta_0<\eta_*\),
\(0<c_0<d_*\), and \(C>0\), independent of \(N\), such that
for \(0<\delta<\delta_0\) and
\(\eta\in B_{\eta_0}(0)\), the equation
\(\mathscr G_N(\delta,\nu,\eta)=0\) has exactly one solution in
\(|\nu-\nu_{0,N}|<c_0\).  The root map
\(\eta\mapsto\nu_{c,N}(\delta,\eta)\) is \(C^2\) Fr\'{e}chet, and
\begin{align}
 \|D_\eta\nu_{c,N}(\delta,\eta)-\delta\Lambda_N\|_{\mathcal T_N^*}
 &\le C\{\delta^2+\delta\|\eta\|_{\mathcal T_N}\},
                                                               \label{eq:ball-root-derivative}\\
 \|D_{\eta\eta}^2\nu_{c,N}(\delta,\eta)\|_{\mathcal L^2(\mathcal T_N;\R)}
 &\le C\delta,                                      \label{eq:ball-root-second-derivative}\\
 |\nu_{c,N}(\delta,\eta)-\nu_{c,N}(\delta,0)
       -\delta\Lambda_N(\eta)|
 &\le C\{\delta^2\|\eta\|_{\mathcal T_N}
          +\delta\|\eta\|_{\mathcal T_N}^2\}.    \label{eq:ball-root-increment}
\end{align}
If \(\mu=\mu_f(\delta)+\delta^q\nu\), then
\begin{equation}
 |\mu_{c,N}(\delta,\eta)-\mu_{c,N}(\delta,0)
       -\delta^{q+1}\Lambda_N(\eta)|
 \le C\{\delta^{q+2}\|\eta\|_{\mathcal T_N}
          +\delta^{q+1}\|\eta\|_{\mathcal T_N}^2\}.          \label{eq:ball-physical-increment}
\end{equation}

In the weighted coordinate
\begin{equation}
 \Xi_{N,\delta}(\eta)
 =\delta^{-1}\{\nu_{c,N}(\delta,\eta)
                      -\nu_{c,N}(\delta,0)\}
 =\delta^{-(q+1)}\{\mu_{c,N}(\delta,\eta)
                      -\mu_{c,N}(\delta,0)\},        \label{eq:ball-weighted-graph}
\end{equation}
the selected connection locus is a codimension-one \(C^2\) graph in
\(\R\times\mathcal T_N\), and
\begin{align}
 |\Xi_{N,\delta}(\eta)-\Lambda_N(\eta)|
 &\le C\{\delta\|\eta\|+\|\eta\|^2\},\notag\\
 \|D_\eta\Xi_{N,\delta}(\eta)-\Lambda_N\|_{\mathcal T_N^*}
 &\le C\{\delta+\|\eta\|\},
 \qquad
 \|D_{\eta\eta}^2\Xi_{N,\delta}(\eta)\|\le C.    \label{eq:ball-weighted-jet}
\end{align}
Hence its \(\dd\xi\)-calibrated baseline conormal representative
\begin{equation}
 \omega_{N,\delta}
 =(1,-D_\eta\Xi_{N,\delta}(0))
 \in(\R\times\mathcal T_N)^*                         \label{eq:ball-conormal}
\end{equation}
converges in dual norm, uniformly in \(N\), to
\begin{equation}
 \omega_{N,0}=(1,-\Lambda_N).                        \label{eq:ball-limiting-conormal}
\end{equation}
\end{theorem}

\begin{proof}
The endpoint-sign and strict-monotonicity argument in the proof of
\cref{thm:abstract-root-response}, now uniform over the smaller closed
structural ball, gives existence and uniqueness in one common
\(\nu\)-window.  The Banach-space implicit-function theorem gives joint
\(C^2\) dependence on \(\eta\).  Along the root,
\begin{equation}
 D_\eta\nu_{c,N}
 =-\frac{D_\eta\mathscr G_N}{\partial_\nu\mathscr G_N}.         \label{eq:ball-ift-ratio}
\end{equation}
The denominator has modulus at least \(a_*\delta/2\).  Insert
\cref{eq:ball-gap-slope,eq:ball-gap-response} into
\cref{eq:ball-ift-ratio} and use
\cref{eq:ball-coefficient-bounds,eq:ball-response-covector}; this proves
\cref{eq:ball-root-derivative} in the operator norm of
\(\mathcal T_N^*\).

For unit vectors \(h,k\in\mathcal T_N\), twice differentiating the root
identity gives
\begin{align*}
 \partial_\nu\mathscr G_N\,
       D_{\eta\eta}^2\nu_{c,N}[h,k]
 =-\{&\partial_{\nu\nu}\mathscr G_N
       D_\eta\nu_{c,N}[h]D_\eta\nu_{c,N}[k]\\
 &+D_\eta\partial_\nu\mathscr G_N[h]
       D_\eta\nu_{c,N}[k]
  +D_\eta\partial_\nu\mathscr G_N[k]
       D_\eta\nu_{c,N}[h]
  +D_{\eta\eta}^2\mathscr G_N[h,k]\}.
\end{align*}
The first derivative is \(O(\delta)\).  Divide the bounds in
\cref{eq:ball-gap-second-jets} by the lower bound for the root slope to
obtain \cref{eq:ball-root-second-derivative}.  Taylor's formula on the line
segment \(t\mapsto t\eta\), together with
\cref{eq:ball-root-derivative} at \(\eta=0\) and
\cref{eq:ball-root-second-derivative}, gives
\cref{eq:ball-root-increment}.  Multiplication by \(\delta^q\) proves
\cref{eq:ball-physical-increment}.  Dividing the last three root estimates
by \(\delta\) proves
\cref{eq:ball-weighted-graph,eq:ball-weighted-jet}.  The graph is defined by
\(\xi-\Xi_{N,\delta}(\eta)=0\), so
\cref{eq:ball-conormal,eq:ball-limiting-conormal} follow.
\end{proof}

\begin{corollary}[Full conormal from the hidden-return Schur map]
\label[corollary]{cor:schur-full-conormal}
Under \cref{thm:hidden-return-schur,thm:structural-ball-connection}, suppose
\(a_N=\mathfrak l_Nf_{\nu,N}\) and
\[
 b_N(R)=\mathfrak l_N
   \mathsf B_NL_{\perp,N}^{-1}\mathsf S_NR,
 \qquad R\in\mathcal T_N.
\]
Then \(\Lambda_N\) in \cref{eq:ball-response-covector} is the Schur
covector of \cref{eq:hidden-return-balance}, and the limiting tangent
hyperplane of
the nonlinear structural-ball connection graph is
\begin{equation}
 \ker\mathfrak J_N
 =\{(\Lambda_N(R),R):R\in\mathcal T_N\},            \label{eq:full-conormal-schur-kernel}
\end{equation}
and its calibrated limiting conormal is \((1,-\Lambda_N)\).  Thus the
limiting full Fr\'{e}chet conormal, not only its restrictions to scalar rays,
is identified by the hidden-return factorization.
\end{corollary}

\begin{proof}
The kernel identity is \cref{eq:cokernel-connection-kernel}; the dual-norm
realization is \cref{eq:ball-limiting-conormal}.
\end{proof}

\begin{proposition}[Selected-to-physical weighted-\(C^1\) criterion]
\label[proposition]{prop:physical-weighted-c1-transfer}
Let \(\mu^{\rm sel}_{c,N}\) be the selected structural-ball root in
\cref{thm:structural-ball-connection}, with
\(\mu=\mu_f(\delta)+\delta^q\nu\).  Suppose that a construction which uses no
preparation datum independently supplies a physical connection root
\[
 \mu^{\rm phys}_{c,N}(\delta,\eta),
 \qquad \eta\in B_{\eta_0}(0)\subset\mathcal T_N,
\]
which is \(C^1\) Fr\'{e}chet in \(\eta\).  Assume there is a function
\(\varepsilon_{\rm phys}(\delta)\downarrow0\) such that
\begin{equation}
 \sup_{N\ge2}\sup_{\|\eta\|<\eta_0}
 \delta^{-(q+1)}
 \left\|D_\eta\mu^{\rm phys}_{c,N}(\delta,\eta)
             -D_\eta\mu^{\rm sel}_{c,N}(\delta,\eta)
 \right\|_{\mathcal T_N^*}
 \le \varepsilon_{\rm phys}(\delta).                 \label{eq:physical-weighted-c1-hypothesis}
\end{equation}
Center the two loci at their own baselines and set
\[
 \Xi^\diamond_{N,\delta}(\eta)
 =\delta^{-(q+1)}
   \{\mu^\diamond_{c,N}(\delta,\eta)
                  -\mu^\diamond_{c,N}(\delta,0)\},
 \qquad \diamond\in\{\mathrm{sel},\mathrm{phys}\}.
\]
Then
\begin{align}
 |\Xi^{\rm phys}_{N,\delta}(\eta)
       -\Xi^{\rm sel}_{N,\delta}(\eta)|
 &\le \varepsilon_{\rm phys}(\delta)\|\eta\|,
                                                               \label{eq:physical-weighted-c0-transfer}\\
 \|D_\eta\Xi^{\rm phys}_{N,\delta}(\eta)-\Lambda_N\|
 &\le C\{\delta+\|\eta\|\}
       +\varepsilon_{\rm phys}(\delta).              \label{eq:physical-weighted-jet-transfer}
\end{align}
In particular, the \(\dd\xi\)-calibrated baseline conormal of the physical locus
converges in dual norm, uniformly in \(N\), to \((1,-\Lambda_N)\).

A gap-level sufficient condition for
\cref{eq:physical-weighted-c1-hypothesis} is the following.  Suppose
\(\mathscr G_N^{\rm sel}\) and \(\mathscr G_N^{\rm phys}\) are jointly
\(C^2\) defining gaps on one common \(\nu\)-window, their displayed roots lie
in that window, and, uniformly in \(N\),
\begin{gather}
 |\partial_\nu\mathscr G_N^\diamond|\ge c\delta,
 \qquad
 \|D_\eta\mathscr G_N^\diamond\|\le C\delta^2,
 \qquad
 |\partial_{\nu\nu}\mathscr G_N^\diamond|
  +\|D_\eta\partial_\nu\mathscr G_N^\diamond\|
     \le C\delta^2,                                   \label{eq:physical-gap-common-bounds}\\
 \delta^{-1}|\partial_\nu
       (\mathscr G_N^{\rm phys}-\mathscr G_N^{\rm sel})|
 +\delta^{-2}\|D_\eta
       (\mathscr G_N^{\rm phys}-\mathscr G_N^{\rm sel})\|
 \le \varepsilon_{\rm gap}(\delta),                  \label{eq:physical-gap-c1-comparison}\\
 \rho(\delta):=
 \sup_{N,\|\eta\|<\eta_0}
 |\nu^{\rm phys}_{c,N}(\delta,\eta)
       -\nu^{\rm sel}_{c,N}(\delta,\eta)|
 \longrightarrow0.                                   \label{eq:physical-root-c0-comparison}
\end{gather}
Here \(\diamond\in\{\mathrm{sel},\mathrm{phys}\}\) and
\(\varepsilon_{\rm gap}(\delta)\to0\).  Under these assumptions one may take
\begin{equation}
 \varepsilon_{\rm phys}(\delta)
 \le C\{\varepsilon_{\rm gap}(\delta)+\rho(\delta)\}. \label{eq:physical-transfer-modulus}
\end{equation}

If two phase sections describe the same physical complete-history connection
set and are related by a parameter-coherent \(C^1\) flow holonomy, their
parameter loci, and hence their conormal lines, agree.  Replacing a regular
defining gap by another one for the same locus does not change the conormal;
under a \(C^1\) parameter-chart change it transforms by cotangent pullback.
The limiting weighted conclusion is transported only through a fixed
nonsingular chart, or a chart family whose derivative and inverse derivative
converge uniformly to a nonsingular front-face derivative.
Consequently cutting and rejoining an already constructed physical passage
by an exact intermediate holonomy leaves the conormal response unchanged.
The proposition does not construct the physical root or verify any of the
comparison hypotheses above.
\end{proposition}

\begin{proof}
The derivative of each weighted graph is
\[
 D_\eta\Xi^\diamond_{N,\delta}
 =\delta^{-(q+1)}D_\eta\mu^\diamond_{c,N}.
\]
Thus \cref{eq:physical-weighted-c1-hypothesis} gives the derivative
comparison, and integration on the segment \(t\mapsto t\eta\) gives
\cref{eq:physical-weighted-c0-transfer}.  Combining the derivative comparison
with \cref{eq:ball-weighted-jet} proves
\cref{eq:physical-weighted-jet-transfer} and the conormal limit.

For the gap-level criterion, implicit differentiation gives
\[
 D_\eta\nu^\diamond_{c,N}
 =-\left.
   \frac{D_\eta\mathscr G_N^\diamond}
        {\partial_\nu\mathscr G_N^\diamond}
   \right|_{\nu=\nu^\diamond_{c,N}}.
\]
Evaluate both quotients first at the same root.  The lower slope bound and
\cref{eq:physical-gap-c1-comparison} bound their difference by
\(C\delta\varepsilon_{\rm gap}(\delta)\).  Moving from one root to the other
and using \cref{eq:physical-gap-common-bounds,eq:physical-root-c0-comparison}
adds at most \(C\delta\rho(\delta)\).  Since
\(\mu=\mu_f(\delta)+\delta^q\nu\), this proves
\cref{eq:physical-weighted-c1-hypothesis,eq:physical-transfer-modulus}.

Exact flow holonomy changes only the section representation of the same
complete-history connection.  Two regular defining functions of one
codimension-one zero set have proportional differentials on that set, and
the chain rule gives cotangent pullback under a chart change.  These facts
prove the final naturality statements.
\end{proof}

\paragraph{How to apply the range-to-root theorem.}
For a new RFDE or other infinite-dimensional matching problem, one must
verify five items: (i) construct one exact history lift and the two selected
traces on it; (ii) prove the zero-fiber property of the scalar matcher;
(iii) justify the joint \(C^2\) graph--trace parameter jets, including the
pure and mixed derivatives in \cref{eq:abstract-gap-second-jets};
(iv) identify the adjoint pairings \(a_N,b_N\), including all phase and
endpoint terms, prove
\(\inf_N|a_N|>0\), and estimate their finite-section tails; and (v) prove
the four range-to-gap bounds
\cref{eq:abstract-gap-level,eq:abstract-gap-slope,%
eq:abstract-gap-response,eq:abstract-gap-second-jets}.  A nonzero response
additionally requires a model-specific bound on \(|b_N/a_N|\); it is not a
consequence of the transfer theorem.

\paragraph{Structural and model-specific inputs.}
The response coefficient should first be obtained from
\cref{thm:hidden-return-schur}; \cref{thm:abstract-root-response} then
persists that coefficient to an exact nonlinear root.  In the shared-resource
network take
\(\mathcal D_{\perp,N}=\mathcal Y_{\perp,N}=E_N\), with the network norm.
The first transverse graph variation and its collective return give the
following bounded maps; in particular, \(\mathsf B_Nh\) is the indicated
whole-line profile, not a single vector:
\begin{align}
 L_{\perp,N}&=A_N,&
 \mathsf S_N\mathbf R
 &=\frac{K}{2\alpha}P_{\perp,N}
      \left(\sum_k\theta_kR_k\right)\mathbf1,\notag\\
 \mathsf B_Nh(s)&=\left(\frac{s}{\alpha}
             \pi_N^T\diag(c_N)h,0\right)^T,&
 \mathfrak l_0f&=\int_{\R}e^{-s^2/2}(s,1)f(s)\,\dd s.   \label{eq:abstract-schur-dictionary}
\end{align}
Here
\begin{equation}
 \mathcal D_{\parallel,N}=\mathcal D_\infty^{c,M},\qquad
 \mathcal Y_{\parallel,N}=\mathcal Y_\infty^{c,M},\qquad
 L_{\parallel,N}=L_0,                                 \label{eq:abstract-collective-dictionary}
\end{equation}
and \cref{lem:fold-gaussian-cokernel} proves
\(\operatorname{Ran}L_{\parallel,N}=\ker\mathfrak l_0\).  The
matching-parameter profile is \(f_{\nu,N}=(0,1)^T\), so
\(\mathfrak l_0f_{\nu,N}=\sqrt{2\pi}\).  Hence
\cref{eq:hidden-return-schur} is exactly
\cref{eq:response-functional}.  The identities
\[
 X_0(s)-X_0(s-\theta_k)=-\frac{\theta_k}{2\alpha},
 \qquad
 \psi(s)=e^{-s^2/2}(s,1)^T
\]
and the stationary-row cancellation are specific to the affine singular
orbit and the chosen delay layers.  Row neutrality removes the direct
structural delay profile.  Separately, the transverse solution
\(h_{*,N}=L_{\perp,N}^{-1}\mathsf S_N\mathbf R\) is constant in \(s\), so
the unperturbed delay operator applied to
\(h_{*,N}(s)-h_{*,N}(s-\theta_k)\) also vanishes.  The remaining
curvature--resolvent return gives
\begin{align}
 f_{\zeta,N}
 &=\mathsf B_NA_N^{-1}\mathsf S_N\mathbf R_N
   \quad\text{in }\mathcal Y_\infty^{c,M},
   \qquad w_{N,R_N}=0,\notag\\
 a_N&=\sqrt{2\pi},\qquad
 b_N=\sqrt{2\pi}\frac{r_N}{\alpha},\qquad
 -\frac{b_N}{a_N}=-\frac{r_N}{\alpha}=\mathscr C_N.
 \label{eq:abstract-current-dictionary}
\end{align}
In particular, at \(\zeta=0\),
\begin{equation}
 \delta^{-1}\partial_\zeta\nu_{c,N}\longrightarrow\mathscr C_N,
 \qquad
 \delta^{-3}\partial_\zeta\mu_{c,N}\longrightarrow\mathscr C_N,
                                                               \label{eq:abstract-current-scaling}
\end{equation}
uniformly in \(N\).  Thus the Schur variable is normalized relative to the
physical parameter \(\mu=\delta^2\nu\), not to an arbitrary rescaling of the
matching row.
The exact history lift, selected trace range, zero-fiber implication,
parameter jets, and transverse inverse needed for this dictionary are
supplied in
\cref{lem:shared-graph-trace}; the exact finite-section passage is
\cref{prop:finite-gap-bridge}.  Together with
\cref{eq:shared-leading-gap,eq:shared-melnikov,%
eq:shared-gaussian-tail,eq:shared-gap-expansion,eq:shared-gap-nu,%
eq:shared-gap-zeta,eq:shared-gap-zz,eq:shared-gap-mixed}
verify the pairing and remainder inputs.  In another model, these affine,
Gaussian, and curvature identities must be recomputed; the abstract theorem
does not propagate them automatically.
