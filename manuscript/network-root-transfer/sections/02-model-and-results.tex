\section{Shared-resource networks and main results}\label{sec:model-results}

\subsection{Network class and fold scaling}

We consider a shared-resource RFDE whose node fields are heterogeneous and
whose slow recovery variable is shared.  For each \(N\ge2\), let
\(P_N\ge0\) be row stochastic and let \(\pi_N\) be a strictly positive
stationary probability vector.  For fixed \(D,\gamma>0\), assume
\begin{equation}
 P_N\mathbf1=\mathbf1,\qquad
 \pi_N^TP_N=\pi_N^T,\qquad \pi_N^T\mathbf1=1,\qquad
 \tau(P_N)\le1-\gamma.                                \label{eq:shared-markov}
\end{equation}
Set
\begin{equation}
 P_{c,N}=\mathbf1\pi_N^T,\qquad P_{\perp,N}=\Id-P_{c,N},\qquad
 E_N=\Ker\pi_N^T,\qquad A_N=D(P_N-\Id)|_{E_N},       \label{eq:shared-splitting}
\end{equation}
and equip \(\R^N\) with
\begin{equation}
 \norm{x}_N=\abs{\pi_N^Tx}+\osc(x).                   \label{eq:shared-norm}
\end{equation}
Let the curvature vectors satisfy, with constants independent of \(N\),
\begin{equation}
 0<c_-\le c_{i,N}\le c_+,\qquad \pi_N^Tc_N=\alpha>0. \label{eq:shared-curvature}
\end{equation}
Fix \(\beta>0\), \(K\ne0\), and finitely many delay locations
\(0\le\theta_0<\cdots<\theta_m\le\Theta_*\).  For
\begin{equation}
 B_{k,N}(\zeta)=B_{k,N}+\zeta R_{k,N},\qquad
 C_N(\zeta)=\sum_{k=0}^mB_{k,N}(\zeta),                \label{eq:shared-layers}
\end{equation}
assume the fixed-support operator bound and the atomwise full-row
neutrality
\begin{equation}
 \sup_N\sum_{k=0}^m
 \left(\norm{B_{k,N}}_{\cL(\R^N,\norm{\cdot}_N)}
       +\norm{R_{k,N}}_{\cL(\R^N,\norm{\cdot}_N)}\right)<\infty,
 \qquad \pi_N^TR_{k,N}=0.                             \label{eq:shared-layer-bounds}
\end{equation}
The latter is a row identity, not merely
\(\pi_N^TR_{k,N}\mathbf1=0\).

For \(\eps=\delta^2\), the RFDE is
\begin{align}
 \dot v(t)={}&\frac23\mathbf1-w(t)\mathbf1
 -c_N\circ(v(t)-\mathbf1)^{\circ2}
 -\frac\beta3(v(t)-\mathbf1)^{\circ3}+D(P_N-\Id)v(t)\notag\\
 &+\delta^2K\left\{C_N(\zeta)v(t)
 -\sum_{k=0}^mB_{k,N}(\zeta)
       v\left(t-\frac{\theta_k}{\delta}\right)\right\},
                                                               \label{eq:shared-rfde-v}\\
 \dot w(t)={}&\delta^2\{\pi_N^Tv(t)-a\},             \label{eq:shared-rfde-w}
\end{align}
where \(v\in\R^N\), \(w\in\R\), and the powers and products are
componentwise.  Its collective fold is
\((v,w,a)=(\mathbf1,2/3,1)\).  Write
\begin{equation}
 \begin{gathered}
  a=1+\delta^2\nu,\qquad \mu=a-1=\delta^2\nu,\\
  M_{1,N}(\zeta)=\sum_{k=0}^m\theta_kB_{k,N}(\zeta),\qquad
  \dot M_{1,N}:=\partial_\zeta M_{1,N}(0)
  =\sum_{k=0}^m\theta_kR_{k,N}.
 \end{gathered}                                       \label{eq:shared-parameters}
\end{equation}
For fixed \(\delta\), let
\[
 \mathcal H_{N,\delta}
 =C([-\theta_m/\delta,0];\R^N)\times\R
\]
be the phase space and denote the right-hand side of
\cref{eq:shared-rfde-v,eq:shared-rfde-w} by
\(\mathcal F_{N,\delta,\nu,\zeta}:\mathcal H_{N,\delta}
\to\R^N\times\R\).  The stationary output projection is
\[
 \Pi_N(x,y)=(\pi_N^Tx,y).
\]

\begin{proposition}[Exact stationary-projection blindness]
\label[proposition]{prop:projection-blindness}
For every full history
\(\Phi=(\phi,\omega)\in\mathcal H_{N,\delta}\) and every admitted parameter
value,
\begin{equation}
 \Pi_N\mathcal F_{N,\delta,\nu,\zeta}(\Phi)
 =\Pi_N\mathcal F_{N,\delta,\nu,0}(\Phi).              \label{eq:projection-blindness}
\end{equation}
Equivalently,
\(\partial_\zeta(\Pi_N\mathcal F_{N,\delta,\nu,\zeta})=0\)
as an identity of maps on the full history space.
\end{proposition}

\begin{proof}
The \(\zeta\)-dependent part of the voltage field is
\[
 \delta^2K\sum_{k=0}^mR_{k,N}
 \{\phi(0)-\phi(-\theta_k/\delta)\}.
\]
Left multiplication by \(\pi_N^T\) annihilates every summand by
\cref{eq:shared-layer-bounds}; the resource equation has no \(\zeta\)-term.
\end{proof}

\Cref{eq:projection-blindness} compares the projected right-hand side at the
same full vector history.  It does not close an autonomous equation for
\(\pi_N^T v\): heterogeneous nonlinearities can return a transverse history
to the stationary output.

Define
\begin{align}
 g_N&=A_N^{-1}P_{\perp,N}c_N,\notag\\
 \kappa_N&=\frac\beta3+2\pi_N^T\diag(c_N)g_N,\qquad
 \nu_{0,N}=-\frac{3\kappa_N}{8\alpha^3},             \label{eq:shared-center}\\
 r_N&=\frac{K}{2\alpha}\pi_N^T\diag(c_N)
 A_N^{-1}P_{\perp,N}\dot M_{1,N}\mathbf1.            \label{eq:shared-rn}
\end{align}
For a row-neutral delay direction
\(\mathbf R=(R_0,\ldots,R_m)\), put
\begin{equation}
 \Lambda_N(\mathbf R)
 =-\frac{K}{2\alpha^2}\pi_N^T\diag(c_N)
 A_N^{-1}P_{\perp,N}
 \left(\sum_{k=0}^m\theta_kR_k\right)\mathbf1.
                                                               \label{eq:response-functional}
\end{equation}
Thus the direction appearing in \cref{eq:shared-layers} has
\(\mathscr C_N=\Lambda_N(\mathbf R_N)=-r_N/\alpha\).

We now fix the selection entering the theorem.  In the fold chart
\[
 s=\delta t,\qquad w=\frac23-\delta^2Y,\qquad
 v=\mathbf1+\delta\mathbf1X
       +\delta^2\{g_NX^2+h\},\qquad h\in E_N,
\]
the singular critical field is
\(q_0(X,Y)=(Y-\alpha X^2,-X)\), with singular orbit
\[
 \gamma_0(s)=\left(-\frac{s}{2\alpha},
                    \frac{s^2-2}{4\alpha}\right).
\]

\begin{definition}[Selection convention]
\label[definition]{def:admissible-preparation}
Put
\[
 \chi=-2\alpha X,\qquad
 d=Y-\alpha X^2+\frac1{2\alpha},
\]
and denote the inverse coordinate map by \(\Gamma(\chi,d)\).  Fix
\(p>4\), \(B>2\Theta_*+2\), \(0<\kappa<1\), \(h_*>0\), and
\[
 0<b_0<\min\left\{1,\frac{B-2\Theta_*-2}{4}\right\}.
\]
For each \(0<\delta<1\), set
\begin{align*}
 S_\delta&=\sqrt{2p\log(1/\delta)},&
 R_\delta&=S_\delta+B,\\
 \mathcal U^{\rm pl}_{\rm ret}(\delta)
 &=\{\Gamma(\chi,d):|\chi|<S_\delta+1,
                       |d|<\delta^\kappa\},\\
 \mathfrak H^{[2]}_\delta
 &=\{\Phi_{q_0}^{-t}u:
       u\in\mathcal U^{\rm pl}_{\rm ret}(\delta),\
       0\le t\le2\Theta_*\}.
\end{align*}

An \emph{admissible fixed selection datum}
\(\mathbf p_{\rm sel}\) consists of extension rules
\((\mathfrak P_{\rm gr},\mathfrak P_{\rm pl})\), positive constants
\(C_{\rm prep},c_{\rm prep}\), an integer \(M_{\rm prep}\in\mathbb N\), and
\(\delta_{\rm prep}>0\).  One pair of rules is fixed simultaneously for all
\(N\) and all \(0<\delta<\delta_{\rm prep}\): it may depend on the frozen
target \(\delta\), but not on the differentiated parameters
\((\nu,\zeta)\).  We require the following four clauses.

\smallskip
\noindent\textup{(S1) Exact depth-two agreement.}
The graph rule agrees with the exact chart field
\cref{eq:shared-exact-X,eq:shared-exact-h} on the open \(b_0\)-neighborhood,
in the \((\chi,d)\)- and stable product norm, of
\(\mathfrak H^{[2]}_\delta\), in every current and delayed slot with
\(\|h\|_N<h_*\).

\smallskip
\noindent\textup{(S2) Complete uniform graph extension.}
The graph rule produces a globally bounded \(C^{12}\) field whose planar
flow is complete.  For the dummy chart amplitude \(\rho\), every derivative
\(D_u^a\partial_\rho^b\partial_\nu^i\partial_\zeta^j\) with
\(a+b+i+j\le12\), \(b\le3\), and \(i,j\le2\) is bounded in the product
norms by
\[
 C_{\rm prep}\langle R_\delta\rangle^{M_{\rm prep}}
 e^{c_{\rm prep}R_\delta},
\]
uniformly in \(N\).  Its fixed-time planar flow has the corresponding mixed
regularity through grade eleven required by the special-flow graph transform,
with the same \(N\)-uniform polynomial--exponential dependence on
\(R_\delta\).

\smallskip
\noindent\textup{(S3) Controlled planar completion.}
After graph reduction, \(\mathfrak P_{\rm pl}\) gives a globally bounded and
globally Lipschitz planar field \(\widetilde Q_{N,\delta,\nu,\zeta}\),
\(C^3\) in the state, with continuous derivatives
\(D_u^a\partial_\nu^i\partial_\zeta^j\) for
\(0\le a\le3\), \(0\le i,j\le2\), and \(i+j\le2\), all with the same
type of \(N\)-uniform tame bound.  It equals the reduced graph field on
\(\mathcal U^{\rm pl}_{\rm ret}(\delta)\), equals \(q_0\) on the
\(b_0\)-balls about \(\gamma_0(\pm R_\delta)\), and, on
\[
 \mathcal C_\delta
 =\{\Gamma(\chi,d):|\chi|\le R_\delta,\ |d|\le b_0/2\},
\]
satisfies, for those same indices,
\begin{equation}
 \left|D_u^a\partial_\nu^i\partial_\zeta^j
   (\widetilde Q_{N,\delta,\nu,\zeta}-q_0)
       (\Gamma(\chi,d))\right|
 \le C_{\rm prep}\delta^{\omega_{ij}}
       e^{c_{\rm prep}|\chi|}\langle\chi\rangle^{M_{\rm prep}},
 \quad
 \omega_{ij}=1+\mathbf1_{\{j\ge1\ {\rm or}\ i=2\}}.    \label{eq:planar-preparation-smallness}
\end{equation}
Both rules are held fixed under parameter differentiation.

\smallskip
\noindent\textup{(S4) Endpoint and phase selection.}
The selected attracting and repelling traces are the solutions satisfying
\[
 \mathscr H_\alpha(z^{a,\rm sel}_N(-R_\delta))=0,
 \qquad
 \mathscr H_\alpha(z^{r,\rm sel}_N(R_\delta))=0,
 \qquad
 X^{a,\rm sel}_N(0)=X^{r,\rm sel}_N(0)=0.
\]
We use ``admissible preparation'' as a synonym for a datum satisfying
\textup{(S1)}--\textup{(S4)}.  The explicit
cutoffs and the exact flow-jet implementation are recorded in the
supplementary material.
\end{definition}

\begin{lemma}[Existence of admissible preparations]
\label[lemma]{lem:preparation-existence}
For every choice of \(p,B,\kappa,b_0,h_*\) satisfying
\cref{def:admissible-preparation}, admissible extension rules exist.  More
precisely, first fix the target-frozen cutoff, saturation, and collar data,
then choose \(C_{\rm prep},c_{\rm prep},M_{\rm prep}\) sufficiently large,
and finally decrease \(\delta_{\rm prep}>0\).
\end{lemma}

\begin{proof}
A target-frozen graph cutoff and transverse saturation give
\textup{(S1)}--\textup{(S2)}; a weighted Whitney
completion on a fixed-width collar gives \textup{(S3)} without
losing powers of \(\delta\).  The construction and uniform estimates are in
\cref{subsec:supplementary-preparation-existence} of the supplementary
material; \textup{(S4)} then fixes the two traces.
\end{proof}

The normalized central gap of these traces is
\begin{align}
 \mathscr D^{\rm sel}_N(\delta,\nu,\zeta)
  &=\frac{2}{\alpha e}\left\{
    \mathscr H_\alpha(z^{a,\rm sel}_N(0))
   -\mathscr H_\alpha(z^{r,\rm sel}_N(0))\right\},
                                                               \label{eq:shared-gap}\\
 \mathscr H_\alpha(X,Y)
  &=\frac12e^{-2\alpha Y}
    \left(\alpha Y-\alpha^2X^2+\frac12\right).       \label{eq:shared-first-integral}
\end{align}
On the retained graph, a zero of \(\mathscr D^{\rm sel}_N\) is
equivalent to equality of the two selected complete RFDE histories.

\begin{theorem}[Uniform complete-history root and hidden response]
\label{thm:shared-resource}
Assume \cref{eq:shared-markov,eq:shared-curvature,eq:shared-layer-bounds}
with all displayed bounds common in \(N\).  Fix a compact interval
\(I_\nu\) containing every \(\nu_{0,N}\) in its
interior with one common positive boundary distance, and one fixed admissible
preparation datum \(\mathbf p_{\rm sel}\) as in
\cref{def:admissible-preparation}.  Then there
exist \(\delta_0,\zeta_0,c_0,C>0\), depending only on the common model
bounds, \(I_\nu\), its stated boundary distance, and
\(\mathbf p_{\rm sel}\), but independent
of \(N\), such that for every \(N\ge2\), every
\(0<\delta<\delta_0\), and every \(\abs\zeta<\zeta_0\), the gap
\(\mathscr D^{\rm sel}_N/\delta\) has exactly one root in the local window
\(\{\nu:\abs{\nu-\nu_{0,N}}<c_0\}\subset I_\nu\),
\begin{equation}
 \nu^{\rm sel}_{c,N}(\delta,\zeta)\in I_\nu,
 \qquad
 \abs{\nu^{\rm sel}_{c,N}(\delta,\zeta)-\nu_{0,N}}<c_0. \label{eq:shared-root}
\end{equation}
No assertion is made here about additional zeros elsewhere in \(I_\nu\).
The selected attracting and repelling complete histories agree at this
root.  The map \(\zeta\mapsto\nu^{\rm sel}_{c,N}(\delta,\zeta)\) is
twice continuously differentiable on the common parameter box.  With
\(\mu^{\rm sel}_{c,N}=\delta^2\nu^{\rm sel}_{c,N}\),
\begin{align}
 &\left|\mu^{\rm sel}_{c,N}(\delta,\zeta)
       -\mu^{\rm sel}_{c,N}(\delta,0)
       -\mathscr C_N\delta^3\zeta\right|
 \le C\{\delta^4\abs\zeta+\delta^3\zeta^2\},        \label{eq:shared-response}\\
 &\mathscr C_N=-\frac{r_N}{\alpha}
 =\Lambda_N(\mathbf R_N).                              \label{eq:shared-coeff}
\end{align}
At \(\zeta=0\),
\begin{equation}
 \left|\delta^{-3}\partial_\zeta
 \mu^{\rm sel}_{c,N}(\delta,0)-\mathscr C_N\right|
 \le C\delta.                                          \label{eq:shared-response-derivative}
\end{equation}
Consequently, if \(c_*:=\inf_N|\mathscr C_N|>0\), then, after decreasing
\(\delta_0\),
\(|\partial_\zeta\mu^{\rm sel}_{c,N}(\delta,0)|
\ge(c_*/2)\delta^3\) for every \(N\).  The exact finite-\(\delta\) root is
the object selected by \(\mathbf p_{\rm sel}\).
\end{theorem}

\begin{corollary}[Non-recovery from the projected functional]
\label[corollary]{cor:projection-nonidentifiability}
Assume the hypotheses and retain the fixed preparation datum of
\cref{thm:shared-resource}.  Suppose in addition that
\(c_*=\inf_N|\mathscr C_N|>0\).  There are common
\(\widehat\delta,\widehat\zeta>0\) such that, whenever
\(0<\delta<\widehat\delta\) and
\(0<|\zeta|<\widehat\zeta\),
\begin{equation}
 \operatorname{sgn}(\mathscr C_N\zeta)
 \{\mu^{\rm sel}_{c,N}(\delta,\zeta)
       -\mu^{\rm sel}_{c,N}(\delta,0)\}
 \ge\frac{c_*}{2}\delta^3|\zeta|                     \label{eq:root-separation}
\end{equation}
for every \(N\).  For every common \(\nu\), the two \(\nu\)-parametrized
systems have exactly the same stationary-projected vector-field functional by
\cref{prop:projection-blindness}, while the roots of their matching problems
differ.  In particular, no rule whose only input is the projected functional
family
\(\nu\mapsto\Pi_N\mathcal F_{N,\delta,\nu,\zeta}\) can determine the selected
root throughout this class.
\end{corollary}

\begin{theorem}[Classification of leading hidden-response directions]
\label{thm:response-directions}
For fixed \(N\), define
\begin{align*}
 \mathfrak T_N^{\rm sr}
 &=\{\mathbf R=(R_0,\ldots,R_m),\ R_k\in\R^{N\times N}:
       \pi_N^TR_k=0
       \text{ for every }k\},\\
 \mathfrak T_N^{\rm red}
 &=\{\mathbf R\in\mathfrak T_N^{\rm sr}:\textstyle\sum_kR_k=0\},
 \qquad
 \|\mathbf R\|_{\mathfrak T_N}=\sum_k
 \|R_k\|_{\mathcal L(\R^N,\|\cdot\|_N)}.
\end{align*}
The functional \(\Lambda_N\) in \cref{eq:response-functional} is bounded,
with the network-size-independent estimate
\begin{equation}
 |\Lambda_N(\mathbf R)|
 \le \frac{|K|c_+\Theta_*}{2\alpha^2D\gamma}
       \|\mathbf R\|_{\mathfrak T_N}.                 \label{eq:response-functional-bound}
\end{equation}
If at least one delay location is positive, the restriction of \(\Lambda_N\)
to \(\mathfrak T_N^{\rm sr}\) is nonzero if and only if
\(c_N\ne\alpha\mathbf1\).  If two delay locations are distinct, the same
equivalence holds on the pure-redistribution space
\(\mathfrak T_N^{\rm red}\).  Whenever the restriction is nonzero, its
zero set is a closed codimension-one hyperplane, so the nonzero-response
directions are open and dense in the corresponding finite-dimensional
direction space.

More explicitly, for distinct \(\theta_j,\theta_\ell\), set
\(z_N=c_N-\alpha\mathbf1\), \(u_N=A_Nz_N\), and
\begin{equation}
 R_j=\frac{u_N\pi_N^T}{\theta_j-\theta_\ell},\qquad
 R_\ell=-R_j,\qquad R_k=0\quad(k\notin\{j,\ell\}).  \label{eq:variance-witness}
\end{equation}
Then \(\mathbf R\in\mathfrak T_N^{\rm red}\) and
\begin{equation}
 \Lambda_N(\mathbf R)
 =-\frac{K}{2\alpha^2}
   \left\{\pi_N^T(c_N^{\circ2})-\alpha^2\right\}.    \label{eq:variance-response}
\end{equation}
Hence heterogeneity, measured by the stationary weighted variance of the
node curvatures, is exactly the obstruction to a nonzero leading-order
projection-invisible response in this tangent class.
\end{theorem}

The preceding classification detects whether one particular return channel is
zero.  We now prove a stronger statement: projection-blind delay probes span
the whole transverse forcing space, so their scalar responses determine the
entire transverse return covector compressed by the collective matching
equation.

For \(\mathbf R\in\mathfrak T_N^{\rm red}\), define the blind source and its
transverse lift by
\begin{equation}
 \mathsf S_N\mathbf R
 =\frac{K}{2\alpha}P_{\perp,N}
       \left(\sum_{k=0}^m\theta_kR_k\right)\mathbf1,
 \qquad
 \mathsf T_N\mathbf R=A_N^{-1}\mathsf S_N\mathbf R\in E_N.
                                                               \label{eq:tomography-source}
\end{equation}
Thus the curvature model has the return covector
\begin{equation}
 \mathfrak r_N^{\rm curv}(h)
 =-\frac1\alpha\pi_N^T\diag(c_N)h,
 \qquad
 \Lambda_N=\mathfrak r_N^{\rm curv}\mathsf T_N.      \label{eq:curvature-return-covector}
\end{equation}

This source is an exact Markov realization of
\cref{prop:abstract-two-atom-source}.  On
\[
 \mathcal A_N^{\rm row}
 =\{R\in\R^{N\times N}:\pi_N^TR=0\}
\]
take
\(\mathsf v_N(R)=P_{\perp,N}R\mathbf1\),
\(\mathsf j_N(y)=y\pi_N^T\), and \(c=K/(2\alpha)\).  In the inherited
operator norm, \(\|\mathsf v_N\|=\|\mathsf j_N\|=1\), uniformly in
\(N\), and \(\mathfrak T_N^{\rm red}\) is isometric to the corresponding
sum-zero tuple space over \(\mathcal A_N^{\rm row}\).  Thus the source and
the displayed right inverse have norm product one; the condition loss below
comes only from the transverse dynamics \(A_N^{-1}\).

\begin{theorem}[Two-delay blind controllability and dual reconstruction]
\label[theorem]{thm:hidden-return-tomography}
Suppose \(m\ge1\), set
\(\Delta_\theta=\theta_m-\theta_0>0\), and retain the norm in
\cref{eq:shared-norm}.  The source
\(\mathsf S_N:\mathfrak T_N^{\rm red}\to E_N\) is onto.  A right inverse is
\begin{equation}
 \begin{aligned}
  (\mathsf Q_Ny)_0
    &=-\frac{2\alpha}{K\Delta_\theta}\,y\pi_N^T,&
  (\mathsf Q_Ny)_m
    &= \frac{2\alpha}{K\Delta_\theta}\,y\pi_N^T,\\
  (\mathsf Q_Ny)_k&=0&& (k\notin\{0,m\}),
 \end{aligned}
 \qquad y\in E_N,                                    \label{eq:tomography-source-right-inverse}
\end{equation}
and the sharp operator bounds are
\begin{equation}
 \|\mathsf S_N\|
   =\frac{|K|\Delta_\theta}{4\alpha},
 \qquad
 \|\mathsf Q_N\|
   =\frac{4\alpha}{|K|\Delta_\theta}.               \label{eq:tomography-source-condition}
\end{equation}
More generally, after coincident delay atoms have been merged, the source
vanishes on \(\mathfrak T_N^{\rm red}\) when the delay support consists of
a single point.  Hence, in the merged representation, at least two distinct
delay locations are necessary and sufficient for this blind controllability.

For any bounded return covector \(\mathfrak r_N\in E_N^*\), put
\begin{equation}
 \mathcal O_N\mathfrak r_N
 :=\mathfrak r_N\mathsf T_N
 \in(\mathfrak T_N^{\rm red})^*.                    \label{eq:tomography-dual-map}
\end{equation}
Then \(\mathcal O_N\) is injective.  If
\(\Lambda_N=\mathcal O_N\mathfrak r_N\), the reconstruction formula is
\begin{equation}
 \mathfrak r_N(z)
 =\Lambda_N(\mathsf Q_NA_Nz),
 \qquad z\in E_N,                                    \label{eq:tomography-reconstruction}
\end{equation}
and
\begin{equation}
 \frac{|K|\Delta_\theta}{4\alpha D(2-\gamma)}
       \|\mathfrak r_N\|
 \le \|\Lambda_N\|
 \le
 \frac{|K|\Delta_\theta}{4\alpha D\gamma}
       \|\mathfrak r_N\|.                           \label{eq:tomography-dual-condition}
\end{equation}
Thus \(\mathcal O_N\), regarded as an isomorphism onto its range, has
condition number at most \((2-\gamma)/\gamma\), uniformly in \(N\), when
the range carries the inherited dual norm; and \(\Lambda_N=0\) if and only
if \(\mathfrak r_N=0\).

There is also a generator-supported realization that preserves the support of
the chosen base layers.
If \(P_N\) is irreducible, define
\begin{equation}
 Q_N(d)=(P_N-\Id)\diag(d),\qquad
 \mathsf G_N(d)_0=Q_N(d),\qquad
 \mathsf G_N(d)_m=-Q_N(d),                           \label{eq:tomography-generator-probe}
\end{equation}
with all other layers zero.  For \(d\in E_N\),
\begin{align}
 \mathsf T_N\mathsf G_N(d)
 &=-\frac{K\Delta_\theta}{2\alpha D}\,d,            \label{eq:tomography-generator-identity}\\
 \mathfrak r_N(d)
 &=-\frac{2\alpha D}{K\Delta_\theta}
       (\mathcal O_N\mathfrak r_N)(\mathsf G_N(d)), \label{eq:tomography-generator-reconstruction}\\
 2\gamma\|d\|_N
 &\le \|\mathsf G_N(d)\|_{\mathfrak T_N}
 \le6(2-\gamma)\|d\|_N.                            \label{eq:tomography-generator-condition}
\end{align}
With base layers
\(B_{0,N}=B_{m,N}=(\Id+P_N)/2\), both perturbed layers are entrywise
nonnegative with exactly the support of the chosen base layers whenever
\(2|\zeta|\|d\|_\infty<1\).  For the open dense set of
\(d\in E_N\) with no zero entry, each nonzero probe layer has rank \(N-1\).
\end{theorem}

\begin{corollary}[Finite-scale recovery of curvature pairings]
\label[corollary]{cor:curvature-tomography}
Assume the common network, curvature, base-layer, and interval hypotheses of
\cref{thm:shared-resource}, let \(m\ge1\), suppose \(P_N\) is irreducible,
and fix one admissible preparation rule common to all
\(\mathsf G_N(d_N)\) with \(\|d_N\|_N\le1\).  Write
\(z_N=c_N-\alpha\mathbf1\in E_N\) and
\(\langle x,y\rangle_{\pi_N}=\pi_N^T(x\circ y)\).  On the
generator-supported probes in
\cref{eq:tomography-generator-probe},
\begin{equation}
 \Lambda_N(\mathsf G_N(d))
 =\frac{K\Delta_\theta}{2\alpha^2D}
       \langle z_N,d\rangle_{\pi_N}.                 \label{eq:curvature-tomography-pairing}
\end{equation}
Hence the leading responses on any \(N-1\) probes spanning \(E_N\)
determine the entire centered curvature profile \(z_N\), not merely whether
it is heterogeneous.

More quantitatively, there are
\(\delta_{\rm tom},\zeta_{\rm tom},C_{\rm tom}>0\), depending only on the
common data in \cref{thm:shared-resource} and that fixed preparation rule,
but independent of \(N\) and of the choice of \(d_N\in E_N\) with
\(\|d_N\|_N\le1\), such that the following holds.  Let
\(\mu^{\mathbf p,d}_{c,N}\) denote the selected root for the direction
\(\mathsf G_N(d_N)\).  For every \(N\ge2\),
\(0<\delta<\delta_{\rm tom}\), and
\(0<|\zeta|<\zeta_{\rm tom}\),
\begin{equation}
 \left|
  \frac{2\alpha^2D}{K\Delta_\theta}
  \frac{\mu^{\mathbf p,d}_{c,N}(\delta,\zeta)
        -\mu^{\mathbf p,d}_{c,N}(\delta,0)}
       {\delta^3\zeta}
  -\langle z_N,d_N\rangle_{\pi_N}
 \right|
 \le C_{\rm tom}(\delta+|\zeta|).                    \label{eq:curvature-finite-tomography}
\end{equation}
The network skeleton \((P_N,\pi_N,D)\), delay locations, \(K\), and
\(\alpha\) are regarded as known; the return profile \(z_N\) is the object
identified by the root responses.  The estimate is uniform for the weighted
response pairings.  It does not, without a uniform lower frame bound for the
chosen probes, assert an \(N\)-uniform condition number for recovering the
coordinate vector \(z_N\) from finitely many noisy pairings.
\end{corollary}

\begin{remark}[Additional structural constraints]
\label[remark]{rem:constrained-directions}
The equivalence in \cref{thm:response-directions} concerns the full
row-neutral matrix spaces displayed there.  If an application prescribes a
smaller tangent space---for example by fixing a sparse support, a Laplacian
sign pattern, or another conservation law---the applicable functional is
\(\Lambda_N\) restricted to that space.  Curvature heterogeneity remains
necessary, but it is sufficient only when this restricted functional is
nonzero; in that case its kernel is again a codimension-one hyperplane in the
admissible tangent space.  The rank-one witness
\cref{eq:variance-witness} need not respect extra constraints.
\end{remark}

The next result supplies a large constrained class which preserves the
chosen base-layer support.  It also shows that the codimension-one conclusion is not an
artifact of allowing arbitrary full matrices.

\begin{proposition}[Generator-supported hidden-response directions]
\label[proposition]{prop:generator-supported-directions}
Fix \(N\ge2\), suppose \(P_N\) is irreducible, and choose two delay atoms
\(\theta_j<\theta_\ell\).  For \(d\in\R^N\), put
\begin{equation}
 Q_N(d)=(P_N-\Id)\diag(d),\qquad
 R_j(d)=Q_N(d),\qquad R_\ell(d)=-Q_N(d),               \label{eq:generator-direction}
\end{equation}
with all other \(R_k(d)=0\), and denote the resulting direction space by
\(\mathfrak G_N\).  Then
\(d\mapsto\mathbf R(d)\) is a linear isomorphism from \(\R^N\) onto
\(\mathfrak G_N\subset\mathfrak T_N^{\rm red}\), and
\begin{equation}
 \Lambda_N(\mathbf R(d))
 =\frac{K(\theta_\ell-\theta_j)}{2\alpha^2D}
   \left\{\pi_N^T(c_N\circ d)
             -(\pi_N^Tc_N)(\pi_N^Td)\right\}.         \label{eq:generator-covariance}
\end{equation}
Consequently, the restriction of \(\Lambda_N\) to \(\mathfrak G_N\) is
nonzero if and only if \(c_N\ne\alpha\mathbf1\).  In that case its kernel is
a closed codimension-one hyperplane in this support-constrained,
pure-redistribution direction space.

For the positive choice \(d=c_N\),
\begin{equation}
 \Lambda_N(\mathbf R(c_N))
 =\frac{K(\theta_\ell-\theta_j)}{2\alpha^2D}
   \left\{\pi_N^T(c_N^{\circ2})-\alpha^2\right\}.     \label{eq:generator-variance}
\end{equation}
Moreover, choose the compatible base layers
\(B_{j,N}=B_{\ell,N}=(\Id+P_N)/2\), with every other layer zero.
If \(\|d\|_\infty\le d_+\), then
\(B_{j,N}+\zeta Q_N(d)\) and
\(B_{\ell,N}-\zeta Q_N(d)\) remain entrywise nonnegative, with their original
base-layer support unchanged, whenever \(2|\zeta|d_+<1\).  For \(d=c_N\), the perturbation
matrix has rank \(N-1\).
\end{proposition}

\begin{corollary}[Leading response is preparation universal]
\label[corollary]{cor:preparation-universality}
Assume the common physical-model, direction-family, and interval hypotheses
of \cref{thm:shared-resource}.  Fix two admissible preparation data
\(\mathbf p_1,\mathbf p_2\) satisfying
\cref{def:admissible-preparation}, and let
\(\mu^{(j)}_{c,N}\) be their selected roots.  On a common parameter box,
with a constant independent of \(N\),
\begin{align}
 &\left|\{\mu^{(1)}_{c,N}(\delta,\zeta)
             -\mu^{(1)}_{c,N}(\delta,0)\}
          -\{\mu^{(2)}_{c,N}(\delta,\zeta)
             -\mu^{(2)}_{c,N}(\delta,0)\}\right|\notag\\
 &\hspace{35mm}\le
 C_{12}\{\delta^4|\zeta|+\delta^3\zeta^2\}.          \label{eq:preparation-comparison}
\end{align}
In particular, for \(\zeta\ne0\), each normalized response differs from the
same physical coefficient \(\mathscr C_N\):
\begin{equation}
 \left|\frac{\mu^{(j)}_{c,N}(\delta,\zeta)
                 -\mu^{(j)}_{c,N}(\delta,0)}{\delta^3\zeta}
       -\mathscr C_N\right|
 \le C_{12}(\delta+|\zeta|),\qquad j=1,2.              \label{eq:normalized-preparation-response}
\end{equation}
Here \(C_{12}\) may depend on the two fixed preparation data but is
independent of \(N\).  This compares response
increments from the two preparation-specific baselines; it does not assert
equality of the two finite-\(\delta\) baseline roots.
\end{corollary}

\begin{definition}[Response-germ equivalence]
\label[definition]{def:response-germ}
Let \(F=(F_N)_{N\ge2}\) and \(G=(G_N)_{N\ge2}\) be scalar-valued families,
each defined on an \(N\)-uniform punctured neighborhood of
\((\delta,\zeta)=(0,0)\), with
\(F_N(\delta,0)=G_N(\delta,0)=0\).  We write \(F\sim_{\rm resp}G\) if
\begin{equation}
 \lim_{\rho\downarrow0}
 \sup_{\substack{N\ge2,\ 0<\delta<\rho\\0<|\zeta|<\rho}}
 \frac{|F_N(\delta,\zeta)-G_N(\delta,\zeta)|}
      {\delta^3|\zeta|}=0.                             \label{eq:response-germ-equivalence}
\end{equation}
where the supremum is taken after restricting the representatives to the
intersection of their neighborhoods.  This is an equivalence relation; its
classes under further restriction are called dimension-uniform response
germs.
\end{definition}

\begin{theorem}[Canonical leading response germ and covector]
\label[theorem]{thm:canonical-response-germ}
Fix one physical model family satisfying the hypotheses of
\cref{thm:shared-resource}, and fix a stationary-row-neutral direction family
\(\mathbf R=(\mathbf R_N)_{N\ge2}\) with a common
\(\mathfrak T_N\)-norm bound.  Throughout the comparison, hold fixed the
stationary projection \(\Pi_N\), the phase condition
\(X_N^a(0)=X_N^r(0)=0\), the endpoint equation \(\mathscr H_\alpha=0\),
and the physical-parameter normalization \(\mu=\delta^2\nu\).  For an
admissible preparation \(\mathbf p\), set
\[
 \Delta^{\mathbf p}_{N,\mathbf R}(\delta,\zeta)
 =\mu^{\mathbf p}_{c,N}(\delta,\zeta)
   -\mu^{\mathbf p}_{c,N}(\delta,0).
\]
The response germ
\begin{equation}
 \mathfrak C(\mathbf R)
 :=[\Delta^{\mathbf p}_{\mathbf R}]_{\rm resp}
 =[(\delta,\zeta)\mapsto
      \delta^3\zeta\,\Lambda_N(\mathbf R_N)]_{\rm resp} \label{eq:canonical-response-germ}
\end{equation}
is the same for any two fixed admissible preparation data, after restricting
the representatives to the intersection of their parameter boxes.  Moreover,
the same coefficient is recovered by the limit
\begin{equation}
 \mathfrak R_N(\mathbf R_N)
 :=\lim_{\delta\downarrow0}\delta^{-3}
   \left.\partial_\zeta
      \mu^{\mathbf p}_{c,N}(\delta,\zeta)\right|_{\zeta=0}
 =\Lambda_N(\mathbf R_N)                               \label{eq:canonical-response-covector}
\end{equation}
which has the same value for every fixed admissible preparation in this
comparison class; for each fixed admissible preparation and each uniformly
bounded direction family the convergence is uniform in
\(N\).  Thus each \(\mathfrak R_N\) is a bounded covector on the
stationary-row-neutral direction space, and the covector norms obey the
uniform estimate
\cref{eq:response-functional-bound}.  Its restriction to each of the
direction spaces in
\cref{thm:response-directions,prop:generator-supported-directions}
has a codimension-one kernel whenever it is nonzero.
\end{theorem}

\begin{remark}[What is canonical]
\label[remark]{rem:canonical-boundary}
The canonicity in \cref{thm:canonical-response-germ} is relative to the fixed
physical model family, stationary projection, phase and endpoint conditions,
and physical-parameter normalization specified above.  Within this comparison
class, replacing one fixed admissible preparation datum by another leaves the
baseline-subtracted leading response germ, and hence its limiting covector,
unchanged.  The assertion is pairwise on common parameter boxes and does not
provide one remainder constant over the entire preparation class.  No
invariance is claimed under changes of the fixed projection, matching
conditions, or parameter normalization.  The theorem neither identifies the
two exact finite-\(\delta\) roots nor produces a preparation-independent
maximal canard.
\end{remark}

\begin{proposition}[Finite-section noncanonicity]
\label[proposition]{prop:finite-section-noncanonicity}
Fix the physical RFDE, the graph preparation, the phase section, and the
endpoint equation \(\mathscr H_\alpha=0\).  The remaining planar completion
allowed by \cref{def:admissible-preparation} does not determine an intrinsic
finite-\(\delta\) root.  More precisely, for any fixed network size \(N_0\)
and every sufficiently small fixed target \(\delta_*>0\), there is an
admissible one-parameter family of target-frozen planar completion rules
\(\mathfrak P_{{\rm pl},\lambda}\), defined for all network sizes and all
targets and equal on the retained physical tubes and endpoint balls, such
that
\begin{equation}
 \left.\partial_\lambda
   \mu^{\lambda}_{c,N_0}(\delta_*,0)\right|_{\lambda=0}\ne0.    \label{eq:completion-root-sensitivity}
\end{equation}
Thus the exact selected root cannot be a function only of the physical RFDE
and its retained local fold data under the present finite-section axioms.
This does not preclude an intrinsic canard selected by additional physical
outer invariant manifolds or asymptotic boundary conditions.
\end{proposition}

The nondegeneracy class is nonempty without a hidden synchrony quotient.
Take \(\pi_N=N^{-1}\mathbf1\), \(P_N=P_{c,N}\), and
\[
 s_{i,N}=\frac{\sqrt3(2i-N-1)}{\sqrt{(N-1)(N+1)}},
 \qquad c_N=\mathbf1+\sigma s_N,\qquad 0<\sigma<1/\sqrt3.
\]
These choices satisfy
\[
 \pi_N^Ts_N=0,\qquad \pi_N^Ts_N^{\circ2}=1,\qquad
 \norm{s_N}_\infty<\sqrt3,\qquad \alpha=1.
\]
For two delays \(\theta_0<\theta_1\), set
\[
 B_{0,N}=B_{1,N}=bP_{c,N},\qquad
 R_{0,N}=s_N\pi_N^T,\qquad R_{1,N}=-s_N\pi_N^T,
 \qquad b>0.
\]
The layers remain entrywise positive for
\(\abs\zeta<b/\sqrt3\), all entries of
\(c_N\) are distinct, and
\begin{equation}
 \mathscr C_N=\frac{K\sigma(\theta_0-\theta_1)}{2D}\ne0
                                                               \label{eq:shared-witness-coeff}
\end{equation}
for every \(N\ge2\).  The distinct curvatures destroy every nontrivial
synchrony polydiagonal.

\begin{corollary}[Robust response coefficient and an asynchronous witness orbit]
\label[corollary]{cor:robust-asynchronous}
Let \(\mathbf R_N^*\) denote the two-layer direction above, put
\(\Delta\theta=\theta_1-\theta_0\), and define
\begin{equation}
 \rho_*=\frac{\sigma\Delta\theta}
 {2(1+\sigma\sqrt3)\theta_1}.                          \label{eq:robust-radius}
\end{equation}
Every sequence \(\widetilde{\mathbf R}_N\in\mathfrak T_N^{\rm sr}\)
satisfying
\[
 \sup_{N\ge2}\|\widetilde{\mathbf R}_N-\mathbf R_N^*\|_{\mathfrak T_N}
 <\rho_*
\]
has
\begin{equation}
 \inf_{N\ge2}|\Lambda_N(\widetilde{\mathbf R}_N)|
 \ge\frac{|K|\sigma\Delta\theta}{4D}.                 \label{eq:robust-response}
\end{equation}
The constants in \cref{thm:shared-resource} may be chosen uniformly over this
ball, because it supplies one common operator bound.  This operator-norm ball
need not preserve entrywise positivity of the perturbed layers; positivity is
not an assumption of \cref{thm:shared-resource}.

For the reference direction \(\mathbf R_N^*\), let
\(v^{\rm sel}_{c,N}\) be the voltage component of the matched selected root
orbit.  Uniformly in \(N\) and in the common \(\zeta\)-interval,
\begin{equation}
 \left\|\delta^{-2}P_{\perp,N}
 v^{\rm sel}_{c,N}(-\delta^{-1})
       +\frac{\sigma}{4D}s_N\right\|_N\le C_a\delta.  \label{eq:asynchronous-profile}
\end{equation}
After decreasing \(\delta_0\), this gives the genuine asynchronous lower
bound
\begin{equation}
 \osc\!\left(P_{\perp,N}v^{\rm sel}_{c,N}(-\delta^{-1})\right)
 \ge\frac{\sigma}{4D}\delta^2                         \label{eq:asynchronous-lower}
\end{equation}
for every \(N\ge2\).  The response is therefore both robust in the
dimension-uniform direction norm and realized on a root orbit outside the
collective subspace.
\end{corollary}

This instance is not an arbitrary-size replication of a lower-dimensional
quotient.  It realizes the projection-blind response on heterogeneous root
orbits with a quantitative transverse amplitude.

The preceding example maximizes mixing and uses rank-one delay directions.
The following family realizes the same obstruction on a sparse directed
topology with high-rank, support-preserving redistribution.

\begin{corollary}[Sparse directed high-rank response]
\label[corollary]{cor:reset-cycle-response}
Fix \(q\in(0,1)\), put \(\varrho=1-q\), and let \(S_N\) be the cyclic forward
shift, \((S_Nx)_i=x_{i+1}\), with cyclic indices.  For \(N\ge3\), set
\begin{equation}
 P_N=\varrho S_N+q\mathbf1e_1^T,\qquad
 \pi_{i,N}=\frac{q\varrho^{i-1}}{1-\varrho^N}.         \label{eq:reset-cycle-network}
\end{equation}
Choose \(0<\eta<1/4\) and define
\begin{equation}
 f_{i,N}=(-1)^i+\eta\frac{i}{N},\qquad
 s_N=f_N-(\pi_N^Tf_N)\mathbf1,\qquad
 c_N=\mathbf1+\sigma s_N,\qquad
 0<\sigma<(2+\eta)^{-1}.                              \label{eq:reset-cycle-curvature}
\end{equation}
Take two delays \(0=\theta_0<\theta_1=\theta\), the base layers
\(B_{0,N}=B_{1,N}=(\Id+P_N)/2\), and the generator-supported directions
\begin{equation}
 R_{0,N}=\lambda(P_N-\Id)\diag(c_N),\qquad
 R_{1,N}=-R_{0,N},\qquad \lambda\ne0.                 \label{eq:reset-cycle-layers}
\end{equation}
Then the common physical-model hypotheses hold with constants independent of
\(N\), \(\tau(P_N)=1-q\), and every \(P_N\) has only \(O(N)\) nonzero
entries.  After fixing the common interval \(I_\nu\) constructed below and an
admissible preparation supplied by \cref{lem:preparation-existence},
\cref{thm:shared-resource} applies with constants independent of \(N\).
The perturbed delay layers retain the same nonnegative
sparse support for
\(2|\zeta\lambda|[1+\sigma(2+\eta)]<1\), while
\(\operatorname{rank}R_{0,N}=N-1\).  Their response satisfies
\begin{align}
 \mathscr C_N
 &=\frac{K\lambda\theta}{2D}
       \left\{\pi_N^T(c_N^{\circ2})-1\right\},
                                                               \label{eq:reset-cycle-coefficient}\\
 \inf_{N\ge3}|\mathscr C_N|
 &\ge \frac{2|K\lambda|\theta\sigma^2q^2(1-q)}{D}>0. \label{eq:reset-cycle-lower}
\end{align}
All entries of \(c_N\) are distinct, so the RFDE has no nontrivial synchrony
polydiagonal.  For its matched root orbit, uniformly on the common
\(\zeta\)-box of \cref{thm:shared-resource} after a common decrease of
\(\delta_0\),
\begin{equation}
 \osc\!\left(P_{\perp,N}v^{\rm sel}_{c,N}(-\delta^{-1})\right)
 \ge\frac{\sigma}{8D}\delta^2,\qquad N\ge3.           \label{eq:reset-cycle-asynchrony}
\end{equation}
\end{corollary}
